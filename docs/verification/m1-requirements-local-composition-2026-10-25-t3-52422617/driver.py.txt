def require(value):

def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'),

def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def unique(pairs):
    value = {}
    for key, item in pairs:
        require(key not in value)
        value[key] = item
    return value


def finite(text):
    value = float(text)
    require(math.isfinite(value))
    return value


def nonfinite(text):
    raise ValueError('nonfinite JSON')


def load(raw):
    return json.loads(raw.decode('ascii'), object_pairs_hook=unique,
                      parse_float=finite, parse_constant=nonfinite)


def keys(value, names):
    require(type(value) is dict and set(value) == set(names))


def equal(actual, expected):
    require(type(actual) is type(expected))
    if type(expected) is dict:
        keys(actual, expected)
        for key in expected:
            equal(actual[key], expected[key])
    elif type(expected) is list:
        require(len(actual) == len(expected))
        for left, right in zip(actual, expected):
            equal(left, right)
    else:
        require(actual == expected)


def integer(value, low=1, high=9223372036854775807):
    require(type(value) is int and low <= value <= high)
    return value


def hexhash(value, width=64):
    require(type(value) is str and len(value) == width
            and all(c in '0123456789abcdef' for c in value))

def token_span(raw, node):
    lines = raw.splitlines(keepends=True)
    return (sum(map(len, lines[:node.lineno - 1])) + node.col_offset,
            sum(map(len, lines[:node.end_lineno - 1])) + node.end_col_offset)


def literal_assignment(tree, name):
    matches = [node for node in tree.body if isinstance(node, ast.Assign)
               and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name)
               and node.targets[0].id == name]
    require(len(matches) == 1 and isinstance(matches[0].value, ast.Constant))
    return matches[0].value


def config_method(raw):
    # AST is DATA ONLY. No compile/eval/exec/import of supplied source.
    tree = ast.parse(raw.decode('ascii'))
    node = literal_assignment(tree, 'CONFIG_TEXT')
    require(type(node.value) is str)
    left, right = token_span(raw, node)
    strings = []
    lines = raw.splitlines(keepends=True)
    for tok in tokenize.tokenize(io.BytesIO(raw).readline):
        if tok.type == tokenize.STRING:
            start = sum(map(len, lines[:tok.start[0] - 1])) + tok.start[1]
            end = sum(map(len, lines[:tok.end[0] - 1])) + tok.end[1]
            if start == left and end == right:
                strings.append(tok)
    require(len(strings) == 1)  # No concatenation or source-expression substitution.
    node.value = '<CONFIG_TEXT>'
    normalized = ast.dump(tree, annotate_fields=True, include_attributes=False).encode('ascii')
    return {'schema': 'openergy.single-config-render-method.v1',
            'template': {'bytes': len(raw), 'sha256': digest(raw)},
            'token_start': left, 'token_end': right, 'token_sha256': digest(raw[left:right]),
            'prefix_sha256': digest(raw[:left]), 'suffix_sha256': digest(raw[right:]),
            'normalized_ast_sha256': digest(normalized),
            'normalization': 'ast.dump annotate_fields=True include_attributes=False; CONFIG_TEXT value=<CONFIG_TEXT>',
            'render': 'one ASCII Python STRING via json.dumps(JSON-text,ensure_ascii=True); exact restore',
            'excludes': ['own reference', 'bound config', 'future full rendered source hash']}


def render_assessor(template, config, method):
    text = canonical(config).decode('ascii').rstrip('\n')
    token = json.dumps(text, ensure_ascii=True).encode('ascii')
    left, right = method['token_start'], method['token_end']
    rendered = template[:left] + token + template[right:]
    require(len(rendered) <= 65536)
    tree = ast.parse(rendered.decode('ascii'))
    node = literal_assignment(tree, 'CONFIG_TEXT')
    equal(node.value, text)
    new_left, new_right = token_span(rendered, node)
    require(new_left == left and rendered[:new_left] == template[:left]
            and rendered[new_right:] == template[right:])
    require(rendered[:new_left] + template[left:right] + rendered[new_right:] == template)
    node.value = '<CONFIG_TEXT>'
    equal(digest(ast.dump(tree, annotate_fields=True, include_attributes=False).encode('ascii')),
          method['normalized_ast_sha256'])
    tick()
    return rendered, text


def verify_observer(template, actual, config):
    tree = ast.parse(template.decode('ascii'))
    source = ast.parse(actual.decode('ascii'))
    edits = []
    for name, expected in [('PAYLOAD_BYTES', config['payload']['bytes']),
                           ('PAYLOAD_SHA256', config['payload']['sha256'])]:
        old = literal_assignment(tree, name)
        new = literal_assignment(source, name)
        require(old.value is None)
        equal(new.value, expected)
        left, right = token_span(actual, new)
        lines = actual.splitlines(keepends=True)
        matches = []
        for token in tokenize.tokenize(io.BytesIO(actual).readline):
            start = sum(map(len, lines[:token.start[0] - 1])) + token.start[1]
            end = sum(map(len, lines[:token.end[0] - 1])) + token.end[1]
            if start == left and end == right:
                matches.append(token)
        require(len(matches) == 1 and matches[0].type ==
                (tokenize.NUMBER if type(expected) is int else tokenize.STRING))
        edits.append((*token_span(actual, new), template[slice(*token_span(template, old))]))
        new.value = None
    restored = actual
    for left, right, original in sorted(edits, reverse=True):
        restored = restored[:left] + original + restored[right:]
    equal(restored, template)
    equal(ast.dump(source, annotate_fields=True, include_attributes=False),
          ast.dump(tree, annotate_fields=True, include_attributes=False))
    equal({'bytes': len(actual), 'sha256': digest(actual)}, config['observer_rendered'])
    tick()

def capture(handle, record):
    # Used only by the two literal Popen call sites below. No phase/argv/cap input.
    selector = None
    buffers = {'stdout': b'', 'stderr': b''}
    caps = {'stdout': 512, 'stderr': 2048}
    try:
        tick()
        selector = selectors.DefaultSelector()
        tick()
        for name in ['stdout', 'stderr']:
            pipe = getattr(handle, name)
            require(pipe is not None and stat.S_ISFIFO(os.fstat(pipe.fileno()).st_mode))
            require(os.fpathconf(pipe.fileno(), 'PC_PIPE_BUF') >= 512)
            os.set_blocking(pipe.fileno(), False)
            tick()
            selector.register(pipe, selectors.EVENT_READ, name)
            tick()
        while True:
            now = time.monotonic_ns()
            if now >= END:
                record['timeout'] = True
                raise TimeoutError('original outer end')
            tick()
            if record['exit_observed_ns'] is None:
                code = handle.poll()  # Actual WNOHANG/reap result, never inferred.
                observed = time.monotonic_ns()  # Immediate sample AFTER stored code.
                record['exit_code'] = code
                if code is not None:
                    record['exit_observed_ns'] = observed
                    record['actual_reaped'] = True
                tick()
                if code is not None:
                    require(type(code) is int and code == 0)
            if (record['exit_observed_ns'] is not None
                    and record['stdout_eof_ns'] is not None
                    and record['stderr_eof_ns'] is not None):
                break
            remaining = (END - tick()) / SECOND
            require(remaining > 0)
            ready = selector.select(timeout=min(0.05, remaining))
            tick()
            for key, events in ready:
                name = key.data
                try:
                    chunk = os.read(key.fileobj.fileno(), caps[name] + 1 - len(buffers[name]))
                    read_ns = time.monotonic_ns()
                except BlockingIOError:
                    tick()
                    continue
                if chunk:
                    record[name + '_observed_bytes'] += len(chunk)
                    buffers[name] = (buffers[name] + chunk)[:caps[name]]
                    record[name + '_hex'] = buffers[name].hex()
                    if record[name + '_observed_bytes'] > caps[name]:
                        record['cap_exceeded'] = True
                        raise ValueError('console cap; unread remainder unknown')
                    if name == 'stdout':
                        require(record['stdout_complete_ns'] is None)
                        if b'\n' in buffers[name]:
                            require(buffers[name].count(b'\n') == 1 and buffers[name].endswith(b'\n'))
                            record['stdout_complete_ns'] = read_ns
                else:
                    record[name + '_eof_ns'] = read_ns
                    if record['stdout_eof_ns'] is not None and record['stderr_eof_ns'] is not None:
                        record['unread_remainder_unknown'] = False
                    selector.unregister(key.fileobj)
                tick()
        require(record['stdout_complete_ns'] is not None and buffers['stderr'] == b'')
        parsed = load(buffers['stdout'])
        equal(canonical(parsed), buffers['stdout'])
        record['unread_remainder_unknown'] = False
    except BaseException as error:
        latch(error)  # Latch BEFORE any subsequent close or possible publication.
        raise
    finally:
        for name in ['stdout', 'stderr']:
            pipe = getattr(handle, name)
            if pipe is not None:
                try:
                    pipe.close()  # No drain, signal or indefinite wait.
                    record[name + '_closed_ns'] = time.monotonic_ns()
                    tick()
                except BaseException as error:
                    record['close_errors'].append(name)
                    latch(error)
        if selector is not None:
            try:
                selector.close()
                tick()
            except BaseException as error:
                record['close_errors'].append('selector')
                latch(error)
    require(FIRST_FAILURE is None and not record['close_errors'])
    record['capture_complete_ns'] = tick()
    return buffers['stdout']

def main():
    global CONFIG, START, END, GUARD_DURABLE, STAGE
    global OBSERVER_PROCESS, ASSESSOR_PROCESS, OBSERVER_RECORD, ASSESSOR_RECORD, ASSESSOR_CONFIG
    CONFIG = bind_config()
    STAGE = 'runtime-preflight'
    qualify_runtime(CONFIG)
    assessor_template, method = preflight(CONFIG)
    # Conservative OUTER ceiling, selected once BEFORE guard and observer Popen.
    # It never replaces or renews the observer-selected original S/A.
    START = time.monotonic_ns()
    END = START + 135 * SECOND
    STAGE = 'once-guard'
    publish('once', canonical({
        'schema': 'openergy.requirements-source-local-parent-once.v1', 'driver_start_ns': START,
        'driver_end_ns': END, 'driver_source': REFS['driver'],
        'config_sha256': digest(CONFIG_TEXT.encode('ascii')), 'retries': 0,
        'observer_argv': [PYTHON, '-I', '-S', '-B', OBSERVER],
        'assessor_argv': [PYTHON, '-I', '-S', '-B', ASSESSOR],
        'requirements_accepted': False, 'r9_reserved_publisher': False}))
    GUARD_DURABLE = True
    STAGE = 'observer'
    OBSERVER_RECORD = child_record(OBSERVER, REFS['observer_source'])
    OBSERVER_RECORD['spawn_started_ns'] = tick()
    OBSERVER_RECORD['spawn_attempted'] = True
    OBSERVER_PROCESS = subprocess.Popen(
        [PYTHON, '-I', '-S', '-B', OBSERVER], stdin=subprocess.DEVNULL,
        stdout=subprocess.PIPE, stderr=subprocess.PIPE, shell=False, close_fds=True,
        cwd='/', env=ENV, bufsize=0)
    returned = time.monotonic_ns()
    OBSERVER_RECORD['spawn_returned_ns'] = returned
    OBSERVER_RECORD['pid_from_retained_handle'] = OBSERVER_PROCESS.pid
    tick()
    observer_console = capture(OBSERVER_PROCESS, OBSERVER_RECORD)
    STAGE = 'observer-parent-record'
    T = observer_parent(CONFIG, observer_console)
    retained_inputs()
    handle_ref = publish('handle', canonical({
        'schema': 'openergy.requirements-source-observer-handle-source.v1',
        'observer': OBSERVER_RECORD, 'runtime': CONFIG['runtime'], 'clock': CLOCK,
        'source_stamps': {name: list(STAMPS[name]) for name in
                          ['driver', 'observer_source', 'payload', 'collector']},
        'driver_source': REFS['driver'], 'root_source_association': REFS['root_source_association'],
        'driver_review': REFS['driver_review'], 'observer_render': REFS['observer_render'],
        'payload_render': REFS['payload_render'], 'parent': REFS['parent'],
        'request': REFS['request'], 'collector_T_ns': T, 'driver_start_ns': START,
        'driver_end_ns': END, 'original_started_ns': S, 'original_absolute_until_ns': A,
        'own_exit_observed': False, 'requirements_accepted': False}))
    # Direct retained Popen topology is actual local causal evidence. Descriptor
    # alone is NOT a host ID. Original collector topology remains a dependency.

        'schema': 'openergy.requirements-source-local-direct-topology.v1', 'driver_pid': os.getpid(),
        'observer_pid_from_retained_handle': OBSERVER_PROCESS.pid,
        'observer_handle_source': handle_ref, 'root_source_association': REFS['root_source_association'],
        'runtime': CONFIG['runtime'], 'clock': CLOCK, 'clock_sha256': CLOCK_HASH,
        'observed_local_path': 'this running driver directly created and reaped the retained observer; next child is the same local Python assessor',
        'remote_host_identity_proved': False, 'own_exit_observed': False}))
    STAGE = 'assessor-render'
    method_ref = publish('method', canonical(method))
    equal(method_ref['sha256'], CONFIG['assessor_method_sha256'])
    ASSESSOR_CONFIG = make_assessor_config(CONFIG, observer_console, topology_ref, handle_ref)
    rendered, config_text = render_assessor(assessor_template, ASSESSOR_CONFIG, method)
    rendered_ref = publish('assessor', rendered)
    actual_rendered = read_fixed('assessor_source')
    equal(actual_rendered, rendered)
    equal(render_assessor(assessor_template, ASSESSOR_CONFIG, method)[0], actual_rendered)

        'config_sha256': digest(config_text.encode('ascii')), 'rendered': rendered_ref,

                                   ASSESSOR_CONFIG, config_text)

def finish(outcome):
    global STAGE
    if not GUARD_DURABLE:
        return 1  # Null/preflight/partial guard: no fallback output or child.
    try:
        STAGE = 'result'
        tick()
        retained_inputs()
        # Earlier complete observer raw bytes live in the handle/source record.
        # If that record was never durable, retain them here instead.
        observer = ({'record': {key: PUBLICATIONS['requirements-source-observer-handle-source.json'][key]
                               for key in ['bytes', 'sha256']}}
                    if 'requirements-source-observer-handle-source.json' in PUBLICATIONS else OBSERVER_RECORD)
        value = {'schema': 'openergy.requirements-source-local-parent-result.v1',
                 'status': ('LOCAL_TRANSPORT_ONLY_PENDING_FULL_REQUIREMENTS'
                            if FIRST_FAILURE is None and outcome is not None else 'FAILED_UNVERIFIED'),
                 'first_observed_failure': FIRST_FAILURE, 'driver_start_ns': START,
                 'driver_end_ns': END, 'original_started_ns': S, 'original_absolute_until_ns': A,
                 'result_prewrite_ns': tick(), 'driver_source': REFS['driver'],
                 'config_sha256': digest(CONFIG_TEXT.encode('ascii')), 'observer': observer,
                 'assessor': ASSESSOR_RECORD, 'completed': outcome,
                 'retained_input_refs': REFS,
                 'retained_input_stamps': {name: list(value) for name, value in STAMPS.items()},
                 'pending': PENDING,
                 'remote_child_exit': None, 'remote_current_state': 'UNVERIFIED',
                 'requirements_accepted': False, 'copy_accepted': False, 'artifact_accepted': False,
                 'product_accepted': False, 'r9_reserved_publisher': False,
                 'own_exit_observed': False, 'external_root_actual_exit_and_eof_required': True}
        ref = publish('result', canonical(value))
        retained_inputs()
        for label in ['run', 'tools', 'launch', 'base']:
            if label in DIRS:
                fd = DIRS.pop(label)
                close_fd(fd)
        STAGE = 'console'
        raw = canonical({'v': 'requirements-source-local-parent.v1', 'state': value['status'],
                         'r': ref, 'requirements_accepted': False, 'own_exit_observed': False})
        require(len(raw) <= 512)
        tick()
        info = os.fstat(1)
        tick()
        require(stat.S_ISFIFO(info.st_mode) and os.fpathconf(1, 'PC_PIPE_BUF') >= 512)
        os.set_blocking(1, False)
        tick()
        require(os.write(1, raw) == len(raw))
        tick()
        return 0 if FIRST_FAILURE is None and outcome is not None else 1
    except BaseException as error:
        latch(error)  # Already exposed result may be stale; NEVER replace it.
        return 1


try:
    OUTCOME = main()
except BaseException as ERROR:
    latch(ERROR)
    OUTCOME = None
RESULT = finish(OUTCOME)
os._exit(RESULT)  # No destructor poll, implicit wait, buffered output or self-exit claim.
