"""Shared strict ownership policy; normal sampling and recovery have separate callers."""
from common import *
from macproc import key, targeted
from coverage import controls
import fcntl


def budget(runtime, deadline, row_count, actor):
    """One cross-process budget: 6000 rounds, 24000 retained process rows."""
    with io_until(deadline):
        with (runtime / 'budget.lock').open('a+b') as lock:
            while True:
                try:
                    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                    break
                except BlockingIOError:
                    pause(deadline)
            path = runtime / 'budget.json'
            value = load(path, deadline) if path.exists() else dict(rounds=0, rows=0)
            proposed = dict(rounds=value['rounds'] + 1, rows=value['rows'] + row_count)
            if actor == 'main' and (proposed['rounds'] > 4976 or proposed['rows'] > 19904):
                raise RuntimeError('main observation share exhausted; cleanup reserve untouched')
            if proposed['rounds'] > 6000 or proposed['rows'] > 24000:
                raise RuntimeError('global observation budget exhausted')
            value = proposed
            # Reserve cleanup coverage inside the existing total, rather than
            # consuming the whole budget in the failed main sampler.
            value['main_budget_exhausted'] = actor == 'main' and (
                value['rounds'] >= 4976 or value['rows'] >= 19904)
            temporary = runtime / ('budget-next-' + str(os.getpid()))
            with temporary.open('w') as stream:
                json.dump(value, stream)
            os.replace(temporary, path)
            return value


def expected(spec):
    root, bundle = Path(spec['private_root']), Path(spec['bundle'])
    app, daemon = str(bundle / 'Contents/MacOS/Openergy'), str(bundle / 'Contents/Helpers/openergy')
    return {
        'peer': [daemon, 'daemon', '--platform', 'virtual', '--config', str(root / 'peer/config.toml'),
                 '--socket', str(root / 'peer/control.sock'), '--foreground'],
        'app': [app, '--e2e', str(root / 'app'), '--join', f'127.0.0.1:{PORTS[0]}',
                '--host-port', str(PORTS[1]), '--require-peer'],
        'peer-status': [daemon, 'status', '--socket', str(root / 'peer/control.sock')],
        'peer-stop': [daemon, 'ctl', 'daemon.stop', '--socket', str(root / 'peer/control.sock'), '--raw'],
    }


def wrapper_argv(spec, runtime, label):
    return [spec['python_executable'], str(HERE / 'exec_owned.py'), str(runtime), label]


def exact_argv(actual, wanted):
    return (len(actual) == len(wanted) and actual[1:] == wanted[1:]
            and Path(actual[0]).resolve() == Path(wanted[0]).resolve())


class Inventory:
    def __init__(self, spec, runtime, actor):
        self.spec, self.runtime, self.actor = spec, Path(runtime), actor
        self.root = Path(spec['private_root'])
        self.commands = expected(spec)
        self.records = {}
        self.pending = {}
        self.unresolved_private = {}  # PID/start metadata only; never ownership.
        self.private_overflow = False
        self.handoff_ready = self.handoff_lost = False
        self.uncertainty_complete = False
        self.checkpoint_sequence = None
        self.sequence = 0
        self.last = None
        self.evidence = Evidence()
        self.deliveries = {}
        self.direct, self.launch_labels = {}, set()
        self.reload(min(spec['flow_deadline'] + CLEANUP_SECONDS, time.monotonic() + 2))

    def reload(self, deadline):
        with io_until(deadline):
            for path in sorted(self.runtime.glob('identity-*.json')):
                record = self.evidence.attempt(load, path, min(deadline, time.monotonic() + 0.05))
                if record is not None:
                    self.records[record['identity_id']] = record

    def known_pids(self):
        return {record['pid'] for record in self.records.values()}

    def uncertainty_state(self, sequence):
        registered = {key(row): {field: row[field] for field in ('pid', 'start_sec', 'start_usec')}
                      for row in self.records.values()}
        return dict(sequence=sequence, unresolved_private=list(self.unresolved_private.values()),
                    registered_starts=list(registered.values()),
                    pending=[dict(pid=pid, started=value[0], attempts=value[1])
                             for pid, value in self.pending.items()],
                    private_overflow=self.private_overflow, handoff_lost=self.handoff_lost)

    def initialize_uncertainty(self, deadline):
        # Controller only, before the watchdog or any product. Never recreate loss.
        write_bytes(self.runtime / 'uncertainty.lock', b'', deadline)
        self.uncertainty_append(self.uncertainty_state(0), deadline)

    def uncertainty_append(self, value, deadline):
        # Payload uses the unchanged byte ledger, file cap and cleanup allocation.
        append(self.runtime / 'uncertainty.jsonl', value, deadline, critical=self.actor != 'main')
        with (self.runtime / 'uncertainty.jsonl').open('rb') as stream:
            os.fsync(stream.fileno())  # A begin must complete before any probe.

    @contextlib.contextmanager
    def absence_guard(self, deadline):
        # Bind permission/final decisions to their full snapshot. A newer or
        # active observer must not be skipped between observation and publication.
        self.handoff_ready = False
        try:
            with io_until(deadline):
                with self.uncertainty_lock(deadline):
                    self.reload(deadline)
                    sequence = self.restore_uncertainty(deadline)
                    if sequence != self.checkpoint_sequence:
                        raise RuntimeError('newer observer state; stale absence decision prohibited')
                    yield
        except BaseException:
            self.handoff_ready = False
            raise

    @contextlib.contextmanager
    def uncertainty_lock(self, deadline):
        # r+b must not create an empty replacement after evidence loss.
        with (self.runtime / 'uncertainty.lock').open('r+b') as lock:
            while True:
                try:
                    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                    break
                except BlockingIOError:
                    pause(deadline)
            yield

    def restore_uncertainty(self, deadline):
        raw = read(self.runtime / 'uncertainty.jsonl', deadline)
        lines = raw.splitlines()
        if not raw.endswith(b'\n') or not 1 <= len(lines) <= 12001:
            raise RuntimeError('missing/truncated/oversized uncertainty handoff')
        sequence, waiting, state = 0, False, None
        for line in lines:
            value = json.loads(line)
            if set(value) == {'begin'}:
                if (state is None or waiting or type(value['begin']) is not int or
                        value['begin'] != sequence + 1):
                    raise RuntimeError('uncertainty transaction sequence mismatch')
                waiting = True
                continue
            if (set(value) != {'sequence', 'unresolved_private', 'registered_starts', 'pending', 'private_overflow', 'handoff_lost'} or
                    type(value['sequence']) is not int or
                    value['sequence'] != sequence + int(waiting) or
                    (state is not None and not waiting) or not 0 <= value['sequence'] <= 6000 or
                    type(value['unresolved_private']) is not list or len(value['unresolved_private']) > 24000 or
                    type(value['registered_starts']) is not list or len(value['registered_starts']) > 24000 or
                    type(value['pending']) is not list or len(value['pending']) > 24000 or
                    type(value['private_overflow']) is not bool or type(value['handoff_lost']) is not bool):
                raise RuntimeError('invalid uncertainty checkpoint')
            unresolved, pending = {}, {}
            registered = {}
            for row in value['unresolved_private'] + value['registered_starts']:
                if (set(row) != {'pid', 'start_sec', 'start_usec'} or
                        any(type(row[k]) is not int for k in row) or row['pid'] <= 0 or
                        row['start_sec'] < 0 or not 0 <= row['start_usec'] < 1000000):
                    raise RuntimeError('invalid metadata-only uncertainty token')
            for row in value['unresolved_private']:
                if row['pid'] in unresolved:
                    raise RuntimeError('duplicate unresolved PID')
                unresolved[row['pid']] = row
            for row in value['registered_starts']:
                if key(row) in registered:
                    raise RuntimeError('duplicate registered start')
                registered[key(row)] = row
            for row in value['pending']:
                if (set(row) != {'pid', 'started', 'attempts'} or type(row['pid']) is not int or row['pid'] <= 0 or
                        type(row['started']) not in (int, float) or not 0 <= row['started'] <= time.monotonic() or
                        type(row['attempts']) is not int or not 0 <= row['attempts'] <= 24000 or row['pid'] in pending):
                    raise RuntimeError('invalid original uncertainty allowance')
                pending[row['pid']] = (row['started'], row['attempts'])
            if len(unresolved.keys() | pending.keys() | {r['pid'] for r in registered.values()}) > 24000:
                raise RuntimeError('uncertainty metadata cap exceeded')
            sequence, waiting = value['sequence'], False
            state = (unresolved, pending, value['private_overflow'], value['handoff_lost'], registered)
        if state is None:
            raise RuntimeError('uncertainty checkpoint unavailable')
        self.unresolved_private, self.pending = state[:2]
        self.private_overflow |= state[2]
        self.handoff_lost |= state[3] or waiting
        # This negative index cannot adopt a start. If its ownership receipt/cache
        # is unavailable, do not turn that lost evidence into app-tree absence.
        self.handoff_lost |= not set(state[4]).issubset({key(row) for row in self.records.values()})
        if sequence:
            # Only sequence zero may precede the first budget file. Loss later
            # cannot invoke budget()'s initial zero counters in a new observer.
            counts = load(self.runtime / 'budget.json', deadline)
            if (set(counts) != {'rounds', 'rows', 'main_budget_exhausted'} or
                    type(counts['rounds']) is not int or not 1 <= counts['rounds'] <= 6000 or
                    type(counts['rows']) is not int or not 0 <= counts['rows'] <= 24000 or
                    type(counts['main_budget_exhausted']) is not bool):
                raise RuntimeError('observation counters unavailable; no allowance reset')
        self.handoff_ready = not self.handoff_lost
        if not self.handoff_ready:
            raise RuntimeError('interrupted/lost uncertainty handoff; absence prohibited')
        return sequence

    def retain_uncertainty(self, row):
        pid = row['pid']
        if pid not in self.unresolved_private:
            if pid not in self.pending and len(self.unresolved_private.keys() | self.pending.keys()) >= 24000:
                self.private_overflow = True
            else:
                self.unresolved_private[pid] = {field: row[field] for field in ('pid', 'start_sec', 'start_usec')}

    def note_pending(self, pid):
        # Pending owned starts also need metadata queries if their identity receipt
        # is lost. These tokens still confer no detail, adoption or signal authority.
        for row in list(self.records.values()) + list(self.direct.values()):
            if row['pid'] == pid:
                self.retain_uncertainty(row)
                break
        if pid not in self.pending and pid not in self.unresolved_private and len(
                self.unresolved_private.keys() | self.pending.keys()) >= 24000:
            self.private_overflow = True
            return time.monotonic(), 8  # Exhaustion blocks; never evict an entry.
        started, attempts = self.pending.get(pid, (time.monotonic(), 0))
        self.pending[pid] = (started, attempts + 1)
        return started, attempts + 1

    def observe(self, deadline, validate=True, focus=None):
        self.handoff_ready = False
        try:
            with io_until(deadline):
                with self.uncertainty_lock(deadline):
                    self.reload(deadline)
                    sequence = self.restore_uncertainty(deadline) + 1
                    if sequence > 6000:
                        raise RuntimeError('uncertainty observation ceiling exhausted')
                    self.handoff_ready = self.uncertainty_complete = False
                    self.uncertainty_append(dict(begin=sequence), deadline)
                    try:
                        return self._observe(deadline, validate, focus)
                    finally:
                        self.handoff_lost |= not self.uncertainty_complete
                        self.uncertainty_append(self.uncertainty_state(sequence), deadline)
                        self.checkpoint_sequence = sequence
                        self.handoff_ready = not self.handoff_lost
        except BaseException:
            # A completed validation error preserves its checkpoint and allowance.
            # Any incomplete transaction/read/write leaves absence blocked locally
            # and a missing, dirty or lost checkpoint blocks every successor.
            if not self.handoff_ready:
                self.handoff_lost = True
            raise

    def acquire_product(self, label, direct, deadline):
        """Prompt same-start acquisition inside observe's existing journal lock.

        Positive rows use identify(); only the later FULL snapshot can grant
        absence. Repeated wrapper reads are observations, never product retries.
        """
        receipt, owner = direct[label], self.spec['owner']
        while True:
            # Reserve two retained rows and one round before this extra query.
            counts = budget(self.runtime, deadline, 2, self.actor)
            started = time.monotonic()
            data = targeted([key(receipt), key(owner)])
            packet = dict(monotonic=time.monotonic(), started_monotonic=started,
                          actor=self.actor, label=label, rows=data['rows'],
                          races=data['races'], absence=data['absence'], counts=counts)
            self.evidence.attempt(append, self.runtime / (self.actor + '-prompt-observations.jsonl'),
                                  packet, deadline, critical=self.actor != 'main')
            rows = {r['pid']: r for r in data['rows']}
            row = rows.get(receipt['pid'])
            if row is None and data['absence'].get(str(receipt['pid'])) == 'ESRCH':
                return  # A missed stage stays missing; no fabricated product record.
            if row is None or row.get('read_race') or row.get('identity_reused') or 'argv' not in row:
                started, attempts = self.note_pending(receipt['pid'])
                if attempts >= 8 or time.monotonic() - started >= 2:
                    raise RuntimeError('finite prompt detail observation budget')
            else:
                record = self.identify(row, rows, direct, deadline, data['absence'])
                if record is None:
                    raise RuntimeError('prompt direct stage has no exact ownership')
                identity_id = key(row) + '-' + record['stage']
                record.update(row, identity_id=identity_id, observed_monotonic=packet['monotonic'])
                if identity_id not in self.records:
                    self.records[identity_id] = record
                    self.evidence.attempt(first, self.runtime / ('identity-' + identity_id + '.json'),
                                          record, deadline, critical=self.actor != 'main')
                if row['nice'] != 15 or row['nice_after'] != 15:
                    raise RuntimeError('prompt observed NI != 15')
                self.pending.pop(row['pid'], None)
                self.unresolved_private.pop(row['pid'], None)
                if record['stage'] == 'product':
                    return
            # Parent-only yield, bounded by the unchanged gate/CLI/flow deadline.
            # The unchanged wrapper/product remains free to exec and run.
            pause(deadline, 0.001)

    def _observe(self, deadline, validate, focus):
        """Raw partial rows are published before ownership/NI assertions can fail."""
        self.sequence += 1
        groups = {record['pgid'] for record in self.records.values()}
        pids = self.known_pids()
        owner = self.spec['owner']
        direct = dict(self.direct)
        self.launch_labels |= {p.name.removesuffix('-command.json') for p in self.runtime.glob('*-command.json')}
        for label in self.commands:
            path = self.runtime / (label + '-exec.json')
            if path.exists():
                receipt = self.evidence.attempt(load, path, deadline)
                if receipt is not None:
                    direct[label] = receipt
            if label in direct:
                pids.add(direct[label]['pid'])
        self.direct = direct
        if focus is not None:
            self.acquire_product(focus, direct, deadline)
        control_tokens = {key(owner)}
        guard_path = self.runtime / 'watchdog-ready.json'
        if guard_path.exists():
            control_tokens.add(key(load(guard_path, deadline)['identity']))
        data = json.loads(command([PYTHON, HERE / 'probe.py', str(deadline), self.root,
                                   *sorted({key(r) for r in self.records.values()} |
                                           {key(r) for r in direct.values()} | control_tokens),
                                   '--private-candidates',
                                   *sorted(key(r) for r in self.unresolved_private.values())], deadline,
                                  coverage=(self.runtime, self.actor, self.spec)))
        selected = [r for r in data['rows'] if r['pid'] in pids or r['ppid'] in pids
                    or r['pid'] in self.unresolved_private or r.get('private_candidate')
                    or r['pgid'] in groups or (r.get('executable') and
                        Path(r['executable']).is_relative_to(self.root))]
        self.last = data  # Fresh rows survive failed publication or accounting.
        for row in selected:
            detected = (row.get('private_candidate') or (row['ppid'] in pids and
                        (row.get('read_race') or 'argv' not in row)))
            if detected and not any(key(r) == key(row) for r in self.records.values()):
                self.retain_uncertainty(row)
        control_rows = controls(self.spec, self.runtime, data, deadline)
        budget_error = None
        try:
            retained_rows = len(selected) + sum(r['row'] is not None for r in control_rows)
            counts = budget(self.runtime, min(deadline, time.monotonic() + 0.05), retained_rows, self.actor)
        except Exception as error:
            budget_error, counts = repr(error), None
        packet = dict(monotonic=time.monotonic(), actor=self.actor, sequence=self.sequence,
                      rows=selected, races=[r for r in data['races'] if r.get('known') or r['pid'] in pids],
                      absence=data['absence'], counts=counts, controls=control_rows,
                      phase=getattr(self, 'phase', 'flow' if self.actor == 'main' else 'cleanup'))
        if counts is not None:
            self.evidence.attempt(append, self.runtime / (self.actor + '-observations.jsonl'),
                                  packet, min(deadline, time.monotonic() + 0.05), critical=self.actor != 'main')
        errors, seen_pending = [], set()
        all_rows = {r['pid']: r for r in data['rows']}
        direct_pids = {r['pid'] for r in direct.values()}
        for row in sorted(selected, key=lambda r: r['pid'] not in direct_pids):
            # Group membership alone never establishes ownership.
            candidate = (row['pid'] in pids or row['ppid'] in pids or
                         row['pid'] in self.unresolved_private or row.get('private_candidate') or
                         (row.get('executable') and Path(row['executable']).is_relative_to(self.root)))
            if not candidate:
                continue
            if (row.get('identity_reused') or (row['pid'] in self.unresolved_private and
                    key(row) != key(self.unresolved_private[row['pid']]))):
                errors.append(dict(row=row, error='known PID reused; no argv read or signal'))
                continue
            if row.get('read_race') or 'argv' not in row:
                # Zombies can lack an executable/argv. A known start remains visible
                # for absence checks, but it cannot be signalled from a partial row.
                if row['status'] == 5 and any(key(r) == key(row) for r in self.records.values()):
                    continue
                seen_pending.add(row['pid'])
                started, attempts = self.note_pending(row['pid'])
                if attempts >= 8 or time.monotonic() - started >= 2:
                    errors.append('finite detail observation budget: ' + str(row['pid']))
                continue
            try:
                record = self.identify(row, all_rows, direct, deadline, data['absence'])
                if record is None:
                    raise RuntimeError('private candidate has no exact ownership chain')
                identity_id = key(row) + '-' + record['stage']
                record.update(row, identity_id=identity_id, observed_monotonic=time.monotonic())
                if identity_id not in self.records:
                    # Save ownership even if NI validation fails next.
                    self.records[identity_id] = record
                    self.evidence.attempt(first, self.runtime / ('identity-' + identity_id + '.json'), record, min(deadline, time.monotonic() + 0.05))
                if row['uid'] != owner['uid'] or row['ruid'] != owner['uid']:
                    raise RuntimeError('UID mismatch')
                if validate and (row['nice'] != 15 or row['nice_after'] != 15):
                    raise RuntimeError('observed product/wrapper NI != 15')
                # Only complete existing ownership validation resolves detection.
                self.unresolved_private.pop(row['pid'], None)
            except Exception as error:
                errors.append(dict(row=row, error=repr(error)))
        for race in packet['races']:
            if race['pid'] in self.unresolved_private and race['pid'] in seen_pending:
                continue  # One unresolved attempt per full observation.
            seen_pending.add(race['pid'])
            started, attempts = self.note_pending(race['pid'])
            if attempts >= 8 or time.monotonic() - started >= 2:
                errors.append(dict(race=race, error='finite known-PID read budget'))
        for pid in list(self.unresolved_private):
            if pid not in all_rows and data['absence'].get(str(pid)) == 'ESRCH':
                del self.unresolved_private[pid]
                continue
            if pid not in seen_pending:
                seen_pending.add(pid)
                started, attempts = self.note_pending(pid)
                if attempts >= 8 or time.monotonic() - started >= 2:
                    errors.append('finite unresolved-private observation budget: ' + str(pid))
        for pid in list(self.pending):
            if pid not in seen_pending:
                del self.pending[pid]
        for control in control_rows:
            row = control['row']
            if row is not None and (not control['valid'] or row.get('nice') != 15 or row.get('nice_after') != 15):
                errors.append('control identity/NI15 detail gap: ' + control['label'])
            elif row is None and control['absent'] != 'ESRCH':
                errors.append('control absence uncertain: ' + control['label'])
        if self.private_overflow:
            errors.append('unresolved-private metadata cap exhausted; absence prohibited')
        if budget_error:
            errors.append(dict(observation_budget_error=budget_error))
        if counts is not None and counts['main_budget_exhausted']:
            errors.append('main observation budget ended; cleanup reserve retained')
        self.uncertainty_complete = True  # Includes expected finite validation failures.
        if errors:
            raise RuntimeError(repr(errors))
        return data

    def identify(self, row, rows, direct, deadline, absence):
        if row['uid'] != self.spec['owner']['uid'] or row['ruid'] != row['uid']:
            raise RuntimeError('candidate UID mismatch')
        existing = [r for r in self.records.values() if key(r) == key(row)
                    and r['executable'] == row['executable'] and r['argv'] == row['argv']]
        if existing:
            record = dict(existing[0])
            parent_pid = int(record['parent_identity'].split('-')[0])
            if row['ppid'] == parent_pid:
                parent = rows.get(parent_pid)
                if parent is None or key(parent) != record['parent_identity']:
                    raise RuntimeError('verified parent absent, uncertain, or reused')
            elif not (row['ppid'] == 1 and parent_pid not in rows and
                      absence.get(str(parent_pid)) == 'ESRCH'):
                raise RuntimeError('unverified parent transition')
            else:
                record['orphaned_from_verified_parent'] = parent_pid
            if row['pgid'] != record['pgid']:
                raise RuntimeError('group changed after identity registration')
            return record
        for label, receipt in direct.items():
            if key(receipt) != key(row):
                continue
            if row['pgid'] != receipt['pgid']:
                raise RuntimeError('direct child group changed across exec')
            owner = self.spec['owner']
            parent = rows.get(owner['pid'])
            live_parent = (row['ppid'] == owner['pid'] and parent is not None and
                           key(parent) == key(owner) and not parent.get('read_race') and
                           parent.get('executable') == owner['executable'] and
                           parent.get('argv') == owner['argv'] and
                           parent['uid'] == owner['uid'] and parent['ruid'] == owner['uid'])
            if (row['executable'] == self.spec['python_executable'] and
                    exact_argv(row['argv'], wrapper_argv(self.spec, self.runtime, label))):
                if not live_parent:
                    raise RuntimeError('new wrapper requires live matched controller')
                return dict(role=label, stage='wrapper', parent_identity=key(owner))
            wanted = self.commands[label]
            if row['executable'] == str(Path(wanted[0]).resolve()) and exact_argv(row['argv'], wanted):
                registered = load(self.runtime / (label + '-registered.json'), deadline)
                if (key(registered) != key(row) or registered['stage'] != 'wrapper' or
                        registered['role'] != label or registered['pgid'] != row['pgid'] or
                        registered['ppid'] != owner['pid'] or registered['nice'] != 15 or
                        registered['nice_after'] != 15 or
                        registered['uid'] != owner['uid'] or registered['ruid'] != owner['uid'] or
                        registered['parent_identity'] != key(owner) or
                        registered['executable'] != self.spec['python_executable'] or
                        not exact_argv(registered['argv'], wrapper_argv(self.spec, self.runtime, label))):
                    raise RuntimeError('same-PID exec verified-wrapper gate mismatch')
                orphan_transition = (row['ppid'] == 1 and parent is None and
                                     absence.get(str(owner['pid'])) == 'ESRCH')
                if not (live_parent or orphan_transition):
                    raise RuntimeError('direct product requires live controller or explicit controller ESRCH')
                return dict(role=label, stage='product', parent_identity=key(owner),
                            controller_absent_exec_transition=orphan_transition)
        app_records = [r for r in self.records.values() if r['role'] == 'app' and r['stage'] == 'product']
        for app in app_records:
            parent = rows.get(app['pid'])
            if parent is None or key(parent) != key(app) or row['ppid'] != app['pid']:
                continue
            if parent.get('executable') != app['executable'] or parent.get('argv') != app['argv']:
                raise RuntimeError('supervised parent details uncertain')
            if parent['uid'] != self.spec['owner']['uid'] or parent['ruid'] != parent['uid']:
                raise RuntimeError('supervised parent UID changed')
            wanted = [str(Path(self.spec['bundle']) / 'Contents/Helpers/openergy'), 'daemon',
                      '--parent-pid', str(app['pid']), '--socket', str(self.root / 'app/config/control.sock'),
                      '--config', str(self.root / 'app/config/config.toml'), '--platform', 'virtual']
            if row['executable'] != wanted[0] or not exact_argv(row['argv'], wanted):
                continue
            if (row['start_sec'], row['start_usec']) < (app['start_sec'], app['start_usec']):
                raise RuntimeError('child started before matched parent')
            prior = {key(r) for r in self.records.values() if r['role'].endswith('supervised-daemon')}
            role = 'initial-supervised-daemon' if not prior else 'recovered-supervised-daemon'
            return dict(role=role, stage='product', parent_identity=key(app))
        return None

    def signal(self, identity, sig, deadline):
        with io_until(deadline):
            return self._signal(identity, sig, deadline)

    def _signal(self, identity, sig, deadline):
        # Fresh independent read immediately before each single-PID signal.
        data = json.loads(command([PYTHON, HERE / 'probe.py', str(deadline), self.root,
                                   '--targeted', key(identity), identity['parent_identity']], deadline,
                                  coverage=(self.runtime, self.actor, self.spec)))
        rows = {r['pid']: r for r in data['rows']}
        row = rows.get(identity['pid'])
        if row is None:
            if data['absence'].get(str(identity['pid'])) == 'ESRCH':
                return dict(delivered=False, result='ESRCH')
            raise RuntimeError('signal target absence uncertain')
        if (key(row) != key(identity) or row.get('read_race') or
                row.get('executable') != identity['executable'] or row.get('argv') != identity['argv'] or
                row['uid'] != self.spec['owner']['uid'] or row['ruid'] != row['uid'] or
                row['pgid'] != identity['pgid']):
            raise RuntimeError('refuse changed/reused/uncertain signal target')
        parent_pid = int(identity['parent_identity'].split('-')[0])
        parent = rows.get(parent_pid)
        if row['ppid'] == parent_pid:
            if parent is None or key(parent) != identity['parent_identity']:
                raise RuntimeError('refuse uncertain signal parent')
            parents = [r for r in self.records.values() if key(r) == identity['parent_identity']]
            parents.sort(key=lambda r: r['stage'] == 'product', reverse=True)
            expected_parent = self.spec['owner'] if key(self.spec['owner']) == identity['parent_identity'] else (
                parents[0] if parents else None)
            if (expected_parent is None or parent.get('read_race') or
                    parent.get('executable') != expected_parent['executable'] or
                    parent.get('argv') != expected_parent['argv'] or
                    parent['uid'] != self.spec['owner']['uid'] or parent['ruid'] != parent['uid']):
                raise RuntimeError('refuse changed/uncertain signal parent details')
        elif not (row['ppid'] == 1 and data['absence'].get(str(parent_pid)) == 'ESRCH'):
            raise RuntimeError('refuse unmatched signal parent')
        self.evidence.attempt(append, self.runtime / (self.actor + '-signals.jsonl'),
               dict(identity=identity, current=row, signal=int(sig), before_monotonic=time.monotonic()),
               min(deadline, time.monotonic() + 0.05), critical=True)
        # Verification, never evidence publication, authorizes this single PID.
        # The final read-to-signal interval is not an atomic macOS identity lease.
        # Keep a delivered signal and its in-memory time together across our
        # own I/O alarm. Only this constant-size, nonblocking action is masked.
        previous_mask = signal.pthread_sigmask(signal.SIG_BLOCK, {signal.SIGALRM})
        try:
            try:
                os.kill(row['pid'], sig)
            except ProcessLookupError:
                return dict(delivered=False, result='ESRCH after verification')
            receipt = dict(identity_id=key(identity), signal=int(sig), delivered=True,
                           delivered_monotonic=time.monotonic())
            self.deliveries.setdefault((key(identity), int(sig)), receipt)
        finally:
            signal.pthread_sigmask(signal.SIG_SETMASK, previous_mask)
        self.evidence.attempt(append, self.runtime / (self.actor + '-signals.jsonl'),
                              receipt, min(deadline, time.monotonic() + 0.05), critical=True)
        return receipt

    def absent(self, data, roles=None):
        records = [r for r in self.records.values() if roles is None or roles(r['role'])]
        # A launch intent with no verified identity is uncertainty, never absence.
        if any((roles is None or roles(label)) and not any(r['role'] == label for r in records)
               for label in self.launch_labels):
            return False
        rows = {r['pid']: r for r in data['rows']}
        # PID reuse is a recorded cleanup failure, never treated as disappearance.
        return (self.handoff_ready and not self.handoff_lost and
                not self.private_overflow and not self.unresolved_private and not self.pending and
                all(r['pid'] not in rows and data['absence'].get(str(r['pid'])) == 'ESRCH' for r in records))

    def final(self, data, deadline):
        with self.absence_guard(deadline):
            return self._final(data)

    def _final(self, data):
        pids = self.known_pids()
        groups = {r['pgid'] for r in self.records.values()}
        leftovers = [r for r in data['rows'] if r['pid'] in pids or r['pgid'] in groups
                     or r['pid'] in self.unresolved_private or r.get('private_candidate')
                     or (r.get('executable') and Path(r['executable']).is_relative_to(self.root))]
        sockets = []
        for path in (self.root / 'app/config/control.sock', self.root / 'peer/control.sock'):
            sockets.append(dict(path=str(path), exists=os.path.lexists(path)))
        return dict(known_pids=sorted(pids), actual_groups=sorted(groups),
                    unresolved_private=list(self.unresolved_private.values()),
                    private_overflow=self.private_overflow,
                    handoff_ready=self.handoff_ready, handoff_lost=self.handoff_lost,
                    absence=data['absence'], remaining=leftovers, sockets=sockets,
                    absent=self.absent(data) and not leftovers and not any(s['exists'] for s in sockets))
