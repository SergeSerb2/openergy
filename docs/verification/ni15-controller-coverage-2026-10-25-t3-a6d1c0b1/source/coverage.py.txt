"""Sample coverage, never continuous NI15 proof or ownership/signal authority."""
from common import *
from macproc import api, bsd, detail, key


def same(row, identity):
    return (not row.get('read_race') and not row.get('identity_reused') and
            all(row.get(k) == identity[k] for k in
                ('pid', 'start_sec', 'start_usec', 'uid', 'ruid', 'pgid', 'executable', 'argv')))


def controls(spec, runtime, data, deadline):
    """Query/coverage identities only. These never enter Inventory.records."""
    identities = {'controller': spec['owner']}
    path = Path(runtime) / 'watchdog-ready.json'
    if path.exists():
        identities['watchdog'] = load(path, deadline)['identity']
    rows = {r['pid']: r for r in data['rows']}
    result = []
    for label, identity in identities.items():
        row = rows.get(identity['pid'])
        valid = row is not None and same(row, identity)
        if valid:
            valid = (row['ppid'] == identity['ppid'] or
                     (label == 'watchdog' and row['ppid'] == 1 and
                      data['absence'].get(str(spec['owner']['pid'])) == 'ESRCH'))
        result.append(dict(label=label, identity=key(identity), row=row,
                           valid=valid, absent=data['absence'].get(str(identity['pid']))))
    return result


class CommandCoverage:
    """Observe a real retained helper child. No added child, gate, retry or signal."""
    def __init__(self, runtime, actor, spec, argv, deadline):
        self.runtime, self.actor, self.argv, self.deadline = Path(runtime), actor, argv, deadline
        # R9 distinguishes the selected launcher from Python's actual executable.
        self.wanted = [spec['python_executable'], *argv[1:]] if argv[0] == PYTHON else argv
        self.token = str(os.getpid()) + '-' + str(time.monotonic_ns())
        self.disabled = False
        self.identity = None
        self.record(dict(event='intent', argv=argv, expected_actual_argv=self.wanted))

    def error(self, message):
        if len(COVERAGE_ERRORS) < 32:
            COVERAGE_ERRORS.append(message[:2048])

    def record(self, value):
        try:
            append(self.runtime / 'helper-coverage.jsonl',
                   dict(value, token=self.token, actor=self.actor, monotonic=time.monotonic()),
                   self.deadline, critical=self.actor != 'main')
        except Exception as error:
            self.error('helper coverage publication: ' + repr(error))

    def spawn(self, process):
        self.record(dict(event='spawn', pid=process.pid))

    def sample(self, process):
        if self.disabled:
            return
        if process.returncode is not None:
            self.record(dict(event='gap', error='helper already reaped; no further PID query'))
            return
        # Reserve BEFORE these extra reads. Loss cannot reset R9's counters.
        try:
            from ownership import budget, exact_argv
            if not (self.runtime / 'budget.json').exists():
                raise RuntimeError('helper recorder requires initialized observation counters')
            with io_until(self.deadline):
                counts = budget(self.runtime, self.deadline, 2, self.actor)
        except Exception as error:
            self.disabled = True
            self.error('helper coverage budget: ' + repr(error))
            self.record(dict(event='gap', error=repr(error)))
            return
        rows, error = [], None
        try:
            with io_until(self.deadline):
                lib = api()
                parent = detail(lib, bsd(lib, os.getpid()))
                rows.append(parent)
                child = bsd(lib, process.pid)
                if child is None:
                    raise RuntimeError('helper ESRCH before product detail')
                rows.append(child)
                if self.identity is None:
                    self.identity = key(child)
                if key(child) != self.identity:
                    raise RuntimeError('helper PID/start changed; no detail read')
                child = detail(lib, child)
                rows[-1] = child
                if (child['ppid'] != parent['pid'] or child['uid'] != parent['uid'] or
                        child['ruid'] != parent['ruid'] or child['pgid'] != parent['pgid'] or
                        child['executable'] != str(Path(self.wanted[0]).resolve()) or
                        not exact_argv(child['argv'], self.wanted) or
                        any(r['nice'] != 15 or r['nice_after'] != 15 for r in rows)):
                    raise RuntimeError('helper identity/argv/NI15 mismatch')
        except Exception as caught:
            error = repr(caught)
        self.record(dict(event='sample', pid=process.pid, rows=rows, error=error, counts=counts))

    def finish(self, process):
        self.record(dict(event='exit', pid=process.pid, code=process.returncode,
                         complete_os_exit_status=process.returncode is not None))


def audit(runtime, deadline, required=(), final=False):
    """Closed finite policy. Gaps stay UNVERIFIED; samples cannot prove continuity."""
    runtime = Path(runtime)
    records = {}
    for p in sorted(runtime.glob('identity-*.json')):
        record = load(p, deadline)
        records.setdefault(key(record), []).append(record)
    spec = load(runtime / 'session.json', deadline)
    guard = load(runtime / 'watchdog-ready.json', deadline)['identity']
    stages, control_samples, gaps = {}, {}, []
    for actor in ('main', 'watchdog', 'fallback', 'cleanup-launch'):
        path = runtime / (actor + '-observations.jsonl')
        if not path.exists():
            continue
        for line in read(path, deadline).splitlines():
            packet = json.loads(line)
            stamp = packet['monotonic']
            for observed, row in [(stamp, r) for r in packet['rows']]:
                for record in records.get(key(row), []):
                    if same(row, record) and row.get('nice') == 15 and row.get('nice_after') == 15:
                        token = record['role'] + ':' + record['stage']
                        stages.setdefault(token, []).append((observed, key(row), actor))
            for control in packet.get('controls', []):
                if control['valid']:
                    row = control['row']
                    if row['nice'] != 15 or row['nice_after'] != 15:
                        gaps.append('control NI15 mismatch: ' + control['label'])
                    else:
                        token = control['label'] + ':' + packet['phase']
                        control_samples.setdefault(token, []).append((stamp, key(row), actor))
                elif control['row'] is not None or control['absent'] != 'ESRCH':
                    gaps.append('control detail missing/changed: ' + control['label'])
    for actor in ('main', 'watchdog', 'fallback', 'cleanup-launch'):
        path = runtime / (actor + '-prompt-observations.jsonl')
        if not path.exists():
            continue
        for line in read(path, deadline).splitlines():
            packet = json.loads(line)
            for row in packet['rows']:
                for record in records.get(key(row), []):
                    if same(row, record) and row.get('nice') == 15 and row.get('nice_after') == 15:
                        token = record['role'] + ':' + record['stage']
                        stages.setdefault(token, []).append((packet['monotonic'], key(row), actor))
    helpers = {}
    path = runtime / 'helper-coverage.jsonl'
    if path.exists():
        for line in read(path, deadline).splitlines():
            event = json.loads(line)
            helpers.setdefault(event['token'], []).append(event)
    else:
        gaps.append('helper ledger missing')
    for token, events in helpers.items():
        intents = [r for r in events if r['event'] == 'intent']
        spawns = [r for r in events if r['event'] == 'spawn']
        exits = [r for r in events if r['event'] == 'exit']
        samples = [r for r in events if r['event'] == 'sample' and not r['error']]
        if (len(intents) != 1 or len(spawns) != 1 or len(exits) != 1 or not samples or
                exits[0].get('complete_os_exit_status') is not True or
                spawns[0]['pid'] != exits[0]['pid'] or
                any(r['pid'] != spawns[0]['pid'] for r in samples) or
                any(r['event'] == 'gap' or (r['event'] == 'sample' and r['error']) for r in events)):
            gaps.append('helper coverage/exit gap: ' + token)
        if samples:
            for sample in samples:
                parent = sample['rows'][0]
                label = 'controller' if same(parent, spec['owner']) else 'watchdog' if same(parent, guard) else None
                if label is None:
                    gaps.append('helper collector identity changed: ' + token)
                elif sample['actor'] == 'postflight':
                    control_samples.setdefault(label + ':postflight', []).append(
                        (sample['monotonic'], key(parent), sample['actor']))
    for token in required:
        samples = stages.get(token, [])
        if not samples:
            gaps.append('missing actual stage: ' + token)
    if final:
        for label in ('controller', 'watchdog'):
            for phase in ('flow', 'flow-complete', 'cleanup:all-exits'):
                if not control_samples.get(label + ':' + phase):
                    gaps.append('missing control phase: ' + label + ':' + phase)
        for role in ('peer', 'app', 'initial-supervised-daemon', 'recovered-supervised-daemon'):
            samples = stages.get(role + ':product', [])
            if not samples or max(r[0] for r in samples) - min(r[0] for r in samples) < 0.02:
                gaps.append('missing separated late product sample: ' + role)
        recovered = stages.get('recovered-supervised-daemon:product', [])
        app = stages.get('app:product', [])
        if not recovered or not any(r[0] >= min(x[0] for x in recovered) for r in app):
            gaps.append('missing late app sample with recovered daemon')
        if not control_samples.get('controller:postflight'):
            gaps.append('missing late controller postflight sample')
        if not any(r[2] != 'main' for r in stages.get('peer:product', [])):
            gaps.append('missing late peer cleanup sample')
    stages = {k: sorted(v) for k, v in stages.items()}
    control_samples = {k: sorted(v) for k, v in control_samples.items()}
    gaps.extend(COVERAGE_ERRORS)
    return dict(schema='openergy.ni15-sample-coverage.v1',
                status='UNVERIFIED' if gaps else 'REQUIRED_SAMPLES_PRESENT',
                continuous_priority_proved=False, gaps=gaps[:128],
                stages={k: dict(rows=len(v), first=v[0], last=v[-1]) for k, v in sorted(stages.items())},
                controls={k: dict(rows=len(v), first=v[0], last=v[-1]) for k, v in sorted(control_samples.items())},
                helper_commands=len(helpers))


def require(runtime, deadline, required=(), final=False):
    result = audit(runtime, deadline, required, final)
    if result['status'] != 'REQUIRED_SAMPLES_PRESENT':
        first(Path(runtime) / 'first-coverage-gap.json', result, deadline, critical='terminal')
        raise RuntimeError('required NI15 samples UNVERIFIED: ' + repr(result['gaps']))
    return result
