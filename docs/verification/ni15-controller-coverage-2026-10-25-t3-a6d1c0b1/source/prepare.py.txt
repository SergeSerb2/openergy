def bind_sources(deadline):
    # Mechanical data admission only. Parent owns the real qualification gates.
    operation = load(HERE / 'operation.bound.json', deadline)
    receipt_path = HERE / 'artifact-receipt.json'
    if (set(operation) != {'schema', 'execution', 'artifact_receipt_sha256'} or
            operation['schema'] != 'openergy.m1-ordinary-operation.v1' or operation['execution'] is not True or
            sha(receipt_path, deadline) != operation['artifact_receipt_sha256']):
        raise RuntimeError('fresh artifact is not concretely bound; no preparation/flow')
    receipt = load(receipt_path, deadline)
    if (receipt['schema'] != 'openergy.m1-ordinary-artifact.v1' or
            receipt['build_candidate'] != '7f20c9cdd97c1dc313f007b879babc726acbb914' or
            receipt['build_review'] != 'fb7b0d97ba4842021d8315956800c9aa8ec20dfe' or
            receipt['source_commit'] != 'ed7d96bf53da1a539fa2acde3b7133613221fd4b' or
            receipt['app_sha256'] != '094d02b3831ee6565ae1628407756ad48053a0268f2a489c65bbde28b768d775' or
            receipt['daemon_sha256'] != '17065c334290b34c7a1b69483f409655e072eb0505ff2a0cc0bf6ed5f77a7bb2'):
        raise RuntimeError('wrong current build/source/review binding')
    required = ('successful_current_build', 'source_702_raw_git_bytes_modes', 'runtime_artifacts_bytes_modes_symlinks', 'arm64_app_daemon_and_embedded_slice', 'signatures_entitlements_unchanged', 'ordinary_info_no_fixture_no_launcher', 'sparkle_pinned_updates_disabled', 'qualified_private_python', 'm1_process_abi', 'archive_transfer_integrity')
    if (set(receipt['qualification_gates']) != set(required) or
            any(receipt['qualification_gates'][name] is not True for name in required)):
        raise RuntimeError('parent artifact qualification incomplete')
    records = receipt['evidence']
    if set(records) != set(('build_result', 'build_worker', 'build_bundle_exit', 'build_operation', 'bundle_files', 'artifact_qualification', 'python_qualification', 'abi_qualification', 'transfer_qualification')):
        raise RuntimeError('missing fresh build/qualification evidence')
    current_records = {
        'build_result': '8129b6c6874fd56e1a4dab83c84114bdafa7a519923efeefbe0c7b733e8ee6f3',
        'build_worker': 'd169b2bf627d906f0df714d4300457e4cd0b8b7239c7db9ab602d58a97f40053',
        'build_bundle_exit': 'be1977b47f6eec5c82700af3087008509b19920e7038191ce6db6b69ea441acb',
        'build_operation': '99759a37cf8fa251d1f400c2db15b6f62bfd387bc9a99fd32e0cb50c6186a039',
        'bundle_files': 'e8255fcb45e50be4ea11710e921f64d8a60368fde2f217a7044e07b0d490e78f',
    }
    if any(records[name]['sha256'] != digest for name, digest in current_records.items()):
        raise RuntimeError('not the exact retained current build metadata')
    for record in list(records.values()) + receipt['supporting_evidence']:
        path = Path(record['path'])
        if (not path.is_absolute() or '..' in path.parts or path.is_symlink() or not path.is_file() or
                path.stat().st_size != record['bytes'] or record['bytes'] > LOG_CAP or
                sha(path, deadline) != record['sha256']):
            raise RuntimeError('fresh evidence bytes changed/missing')
    result = load(Path(records['build_result']['path']), deadline)
    worker = load(Path(records['build_worker']['path']), deadline)
    recovery = result['recovery']
    status = 'BUNDLE_COMPLETED_ACCEPTANCE_UNVERIFIED'
    if (result['status'] != status or result['bundle_completed'] is not True or
            result['before_total_deadline'] is not True or result['dropped_output_bytes'] != 0 or
            recovery['uncertain'] is not False or recovery['stop_requested'] is not False or
            recovery['keeper_reaped'] is not True or recovery['keeper_wait_status'] != 0 or
            recovery['remaining_sampled_identities'] != [] or worker['status'] != status or
            load(Path(records['build_bundle_exit']['path']), deadline)['returncode'] != 0):
        raise RuntimeError('current build did not complete acceptably')
    built = load(Path(records['build_operation']['path']), deadline)
    if (built['execution'] is not True or built['source_commit'] != receipt['source_commit'] or
            built['recipe_sha256'] != {
                'controller.py': 'cd5dc620e6adbec09798e600a99f55c0368b8eeff10d1117e1f7793ca1baf4ee',
                'source-bindings.json': '20e011251151cec81f69c4b80b0072d9fd0a98c1eaaf0e6c74d83ff752546617',
                'swift-entry.sh': 'eda808a9e24b9041afbfaaf4f1febcaa01e0f76c4a56c7502da7eb602472eb08'}):
        raise RuntimeError('build operation does not bind reviewed recipe')
    for value in (receipt['source_root'], receipt['source_bundle'], receipt['python']['launch_path']):
        path = Path(value)
        if not path.is_absolute() or '..' in path.parts or str(path.resolve(strict=True)) != value:
            raise RuntimeError('parent must bind canonical existing machine paths')
    python = receipt['python']
    if (str(REPO) != receipt['source_root'] or PYTHON != python['launch_path'] or
            python['prefix_sha256'] != 'c46c9234c07785e90b66774ca161f20c7ef72c0a87323f52583fa4edb1186e67' or
            python['executable_sha256'] != 'd81701904d7a58f6abb3ab339c1d38d48cd24a39b37645cf7338c07a1e8e70e6' or
            sha(Path(PYTHON), deadline) != python['executable_sha256']):
        raise RuntimeError('private source/Python binding changed')
    bindings_path = HERE / 'build-source-bindings.json'
    if (sha(bindings_path, deadline) != '20e011251151cec81f69c4b80b0072d9fd0a98c1eaaf0e6c74d83ff752546617' or
            sha(HERE / 'limits.json', deadline) != 'cadd06e19607f5dbe0ad070072673c8e518d961b1554958d243a0d028243d733'):
        raise RuntimeError('pinned new source inventory or unchanged limits changed')
    bound = load(bindings_path, deadline)
    rows = bound['source_files']
    if bound['source_commit'] != receipt['source_commit'] or len(rows) != 702:
        raise RuntimeError('wrong raw Git source closure')
    with io_until(deadline):
        actual = {str(p.relative_to(REPO)) for p in REPO.rglob('*') if p.is_file() or p.is_symlink()}
        if actual != {r['path'] for r in rows}:
            raise RuntimeError('fresh 702-file source set changed')
        for record in rows:
            path = REPO / record['path']
            if (path.is_symlink() or not path.is_file() or path.stat().st_size != record['bytes'] or
                    oct(path.stat().st_mode & 0o7777) != record['mode'] or sha(path, deadline) != record['sha256']):
                raise RuntimeError('fresh raw Git source bytes/mode changed: ' + record['path'])
    inventory = load(Path(records['bundle_files']['path']), deadline)
    if inventory['app'] != receipt['build_bundle_path'] or tree(Path(receipt['source_bundle']), deadline) != inventory['files']:
        raise RuntimeError('fresh full bundle tree differs from successful build')
    return dict(artifact=receipt, artifact_receipt_sha256=operation['artifact_receipt_sha256'],
                source_bindings_sha256='20e011251151cec81f69c4b80b0072d9fd0a98c1eaaf0e6c74d83ff752546617', limits_sha256='cadd06e19607f5dbe0ad070072673c8e518d961b1554958d243a0d028243d733')


def main():
    nice15()
    deadline = time.monotonic() + 120  # File I/O budget only; no flow started.
    with io_until(deadline):
        initialize_bytes(deadline)
        output = HERE / 'prepared.json'
        if output.exists():
            raise RuntimeError('one file-only fixture already prepared; do not retry')
        # A separate exclusive guard survives partial preparation failures.
        save(HERE / 'prepare-guard.json', dict(monotonic=time.monotonic(), runtime_invocations=0), deadline)
        bindings = bind_sources(deadline)
        old = bindings['artifact']
        expected_tree = load(Path(old['evidence']['bundle_files']['path']), deadline)['files']
        source_bundle = Path(old['source_bundle']).resolve()
        if tree(source_bundle, deadline) != expected_tree:
            raise RuntimeError('validated source bundle tree changed/missing')
        private = Path(tempfile.mkdtemp(prefix='oe-m1-ordinary-fresh-', dir='/private/tmp')).resolve()
        private.chmod(0o700)
        for name in ('home', 'tmp', 'app', 'peer', 'cache'):
            (private / name).mkdir(mode=0o700)
        if private.stat().st_dev != HERE.stat().st_dev:
            raise RuntimeError('reserved archival requires one filesystem; no copy fallback')
        bundle = private / 'Openergy.app'
        shutil.copytree(source_bundle, bundle, symlinks=True, copy_function=shutil.copy2)
        if tree(bundle, deadline) != expected_tree or tree(source_bundle, deadline) != expected_tree:
            raise RuntimeError('bundle copy/source tree differs')
        info = plistlib.loads(read(bundle / 'Contents/Info.plist', deadline))
        if (info['CFBundleIdentifier'] != 'dev.openergy.app' or info['CFBundleExecutable'] != 'Openergy'
                or 'OpenergyPrivateUpdateFixture' in info or
                (bundle / 'Contents/MacOS/OpenergyPrivateLauncher').exists()):
            raise RuntimeError('not ordinary Info.plist/bundle')
        app_hash = sha(bundle / 'Contents/MacOS/Openergy', deadline)
        daemon_hash = sha(bundle / 'Contents/Helpers/openergy', deadline)
        if app_hash != old['app_sha256'] or daemon_hash != old['daemon_sha256']:
            raise RuntimeError('copied executable differs')
        environment = dict(PATH='/usr/bin:/bin', LANG='C', HOME=str(private / 'home'),
                           CFFIXED_USER_HOME=str(private / 'home'), TMPDIR=str(private / 'tmp') + '/',
                           XDG_CONFIG_HOME=str(private / 'home'), XDG_CACHE_HOME=str(private / 'cache'),
                           MACOSX_DEPLOYMENT_TARGET='15.0', PYTHONDONTWRITEBYTECODE='1',
                           PYTHONNOUSERSITE='1', OPENERGY_DISABLE_UPDATES='1',
                           OPENERGY_ORDINARY_SOURCE_ROOT=str(REPO), OPENERGY_ORDINARY_PYTHON=PYTHON)
        inherited_limits = load(HERE / 'limits.json', deadline)
        limit_names = ('flows', 'routine', 'flow_watchdog_seconds', 'owned_cleanup_seconds',
                       'peer_readiness_seconds', 'cli_seconds', 'sample_interval_seconds',
                       'max_process_samples', 'max_sample_rounds', 'max_log_bytes', 'max_evidence_bytes',
                       'ipc_max_frame_bytes', 'routine_stalled_padding_bytes', 'routine_stalled_request_ms',
                       'routine_responsiveness_seconds', 'routine_stalled_bound_seconds',
                       'routine_step_wait_seconds', 'routine_supervisor_stop_seconds', 'exit_contract', 'no_retry', 'scope')
        value = dict(private_root=str(private), bundle=str(bundle), environment=environment,
                     bundle_tree=expected_tree, app_sha256=app_hash, daemon_sha256=daemon_hash,
                     source_bindings=bindings, original_commit=ORIGINAL,
                     source_bundle=str(source_bundle), product_invocations=0, ports=list(PORTS),
                     baseline='fresh-m1', source_commit=old['source_commit'],
                     limits={name: inherited_limits[name] for name in limit_names})
        save(output, value, deadline)


if __name__ == '__main__':
    main()
