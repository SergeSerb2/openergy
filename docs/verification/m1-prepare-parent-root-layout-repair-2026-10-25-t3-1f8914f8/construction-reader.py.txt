def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=True, allow_nan=False).encode('ascii')

def write(name, value):
    (OUT / name).write_bytes((json.dumps(value, sort_keys=True, indent=2, ensure_ascii=True, allow_nan=False) + '\n').encode('ascii'))

def pairs(items):
    result = {}
    for key, value in items:
        assert key not in result, key
        result[key] = value
    return result

def strict(raw):
    def reject(value):
        raise ValueError(value)
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=reject)

def check(label, predicate):
    CHECKS.append(dict(check=label, passed=bool(predicate)))
    assert predicate, label

def git(*args):
    return subprocess.check_output(['git', *args], env=dict(os.environ, GIT_OPTIONAL_LOCKS='0'))

def binding(raw):
    return dict(bytes=len(raw), sha256=sha(raw))

def tree(raw):
    return ast.parse(raw, feature_version=(3, 9))

def literals(raw):
    result = {}
    for node in tree(raw).body:
        if isinstance(node, ast.Assign) and len(node.targets) == 1 and isinstance(node.targets[0], ast.Name):
            try:
                result[node.targets[0].id] = ast.literal_eval(node.value)
            except (ValueError, TypeError):
                pass
    return result

def declarations(raw):
    lines = raw.splitlines(keepends=True)
    result = {}
    def visit(nodes, prefix=''):
        for node in nodes:
            if isinstance(node, (ast.FunctionDef, ast.ClassDef, ast.AsyncFunctionDef)):
                name = prefix + node.name
                start = min([node.lineno] + [d.lineno for d in getattr(node, 'decorator_list', [])])
                end = node.end_lineno
                code = b''.join(lines[start - 1:end])
                result[name] = dict(start_line=start, end_line=end, **binding(code), code=code, node=node)
                visit(node.body, name + '.')
            else:
                for child in ast.iter_child_nodes(node):
                    if hasattr(child, 'body') and isinstance(child.body, list):
                        visit(child.body, prefix)
    visit(tree(raw).body)
    return result
