class Stop(RuntimeError):
    pass


def require(ok, label):
    if not ok:
        raise Stop(label)


def clock_ns():
    require(sys.platform == 'darwin' and getattr(time, 'CLOCK_MONOTONIC', None) == 6
            and hasattr(time, 'clock_gettime_ns'), 'closed SYSTEM clock unavailable')
    value = time.clock_gettime_ns(time.CLOCK_MONOTONIC)
    require(type(value) is int and 0 <= value < 10 ** 20, 'shared clock integer')
    return value


def remaining(end):
    left = end - clock_ns()
    require(left > 0, 'shared phase expired')
    return math.nextafter(left / NS, -math.inf)


@contextlib.contextmanager
def until(end):
    # Absolute ceilings only. No restoration of an expired budget as new time.
    duration = remaining(end)
    old_handler = signal.getsignal(signal.SIGALRM)
    old_timer = signal.getitimer(signal.ITIMER_REAL)
    require(old_timer == (0.0, 0.0), 'no inherited/nested alarm')
    def expired(_signum, _frame):
        raise TimeoutError('shared absolute ceiling')
    signal.signal(signal.SIGALRM, expired)
    try:
        signal.setitimer(signal.ITIMER_REAL, duration)
        yield
        remaining(end)
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
        signal.signal(signal.SIGALRM, old_handler)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def encoded(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'),
                       ensure_ascii=True, allow_nan=False) + '\n').encode('ascii')


def hex_string(value, width):
    return (type(value) is str and len(value) == width
            and all(c in '0123456789abcdef' for c in value))


def same_data(left, right):
    # Python bool/int equality must not weaken closed JSON binding types.
    if type(left) is not type(right):
        return False
    if type(right) is dict:
        return set(left) == set(right) and all(same_data(left[k], right[k]) for k in right)
    if type(right) is list:
        return len(left) == len(right) and all(same_data(a, b) for a, b in zip(left, right))
    return left == right


def stamp(st):
    return (st.st_dev, st.st_ino, st.st_mode, st.st_uid, st.st_gid, st.st_nlink,
            st.st_size, st.st_mtime_ns, st.st_ctime_ns)


def open_root():
    # Exact path only. No root creation, fallback, search or canonical rewrite.
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        for component in ROOT.split('/')[1:]:
            child = os.open(component, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW,
                            dir_fd=fd)
            os.close(fd)
            fd = child
        info = os.fstat(fd)
        require(info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o700
                and info.st_dev == os.stat('/private/tmp').st_dev, 'exact root owner/mode/device')
        require(stamp(info) == stamp(os.stat(ROOT, follow_symlinks=False)), 'root linked identity')
        return fd
    except BaseException:
        os.close(fd)
        raise


def read_leaf(rootfd, name, binding):
    require('/' not in name and name not in ('', '.', '..'), 'fixed initial leaf')
    fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=rootfd)
    try:
        before = os.fstat(fd)
        require(stat.S_ISREG(before.st_mode) and before.st_uid == os.getuid()
                and stat.S_IMODE(before.st_mode) == 0o600 and before.st_nlink == 1
                and before.st_size == binding['bytes'] <= FILE_CAP, 'initial leaf metadata')
        data = bytearray()
        while len(data) <= binding['bytes']:
            chunk = os.read(fd, min(65536, binding['bytes'] + 1 - len(data)))
            if not chunk:
                break
            data.extend(chunk)
        require(len(data) == binding['bytes'] and sha(data) == binding['sha256'], 'initial raw bytes/hash')
        require(stamp(before) == stamp(os.fstat(fd))
                == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)), 'stable linked initial leaf')
        return bytes(data)
    finally:
        os.close(fd)

def inventory(rootfd):
    # Finite complete count below ROOT only. No followed links; no output listing.
    count, total = 0, 0
    def walk(fd, depth):
        nonlocal count, total
        require(depth <= DEPTH_CAP, 'initial directory depth cap')
        before = os.fstat(fd)
        require(stat.S_ISDIR(before.st_mode) and before.st_uid == os.getuid()
                and stat.S_IMODE(before.st_mode) == 0o700, 'initial directory owner/mode')
        with os.scandir(fd) as entries:
            for entry in entries:
                count += 1
                require(count <= ENTRY_CAP and len(entry.name.encode('utf-8')) <= 255,
                        'initial entry/name cap')
                require(entry.name not in ('bytes.json', 'bytes.lock'), 'SOURCE before ledger')
                info = entry.stat(follow_symlinks=False)
                require(info.st_uid == os.getuid() and info.st_dev == before.st_dev,
                        'initial owner/device')
                if stat.S_ISDIR(info.st_mode):
                    child = os.open(entry.name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
                    try:
                        require(stamp(info) == stamp(os.fstat(child)), 'linked directory')
                        walk(child, depth + 1)
                        require(stamp(info) == stamp(os.stat(entry.name, dir_fd=fd, follow_symlinks=False)),
                                'stable linked directory')
                    finally:
                        os.close(child)
                else:
                    require(stat.S_ISREG(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o600
                            and info.st_nlink == 1 and info.st_size <= FILE_CAP,
                            'initial regular file/mode/cap')
                    total += info.st_size
                    require(total <= INITIAL_CAP, 'initial aggregate reservation')
        require(stamp(before) == stamp(os.fstat(fd)), 'no concurrent directory mutation')
    walk(rootfd, 0)
    return dict(entries=count, bytes=total)


def closed_root_layout(rootfd, entry_outputs=False):
    # Reject presence; no recursion, suffix admission, deletion or root repair.
    # Exact initial260, then only the four already-durable INITIAL outputs264.
    require(type(entry_outputs) is bool and len(ROOT_INITIAL_NAMES) == 260,
            'closed pre-prepare layout phase/names')
    extra = OUTPUTS[:4] if entry_outputs else ()
    require(set(PUBLICATIONS) == set(extra), 'no future output publication before private attempt')
    for name in extra:
        item = PUBLICATIONS[name]
        require(item['state'] == 'durable' and item['ref'] is not None,
                'only already-durable entry outputs')
        read_leaf(rootfd, name, item['ref'])
    expected = ROOT_INITIAL_NAMES | frozenset(extra)
    before = os.fstat(rootfd)
    require(stat.S_ISDIR(before.st_mode) and before.st_uid == os.getuid()
            and stat.S_IMODE(before.st_mode) == 0o700, 'flat root owner/mode')
    seen, total = {}, 0
    with os.scandir(rootfd) as entries:
        for entry in entries:
            remaining(REMOTE_END)
            name = entry.name
            require(name in expected and name not in seen and len(seen) < len(expected)
                    and len(name.encode('utf-8')) <= 255, 'exact flat ROOT name set')
            info = os.stat(name, dir_fd=rootfd, follow_symlinks=False)
            require(stat.S_ISREG(info.st_mode) and info.st_uid == before.st_uid
                    and info.st_dev == before.st_dev and stat.S_IMODE(info.st_mode) == 0o600
                    and info.st_nlink == 1 and 0 <= info.st_size <= FILE_CAP,
                    'flat nofollow regular owner0600 nlink1 file cap')
            seen[name] = stamp(info)
            total += info.st_size
            require(len(seen) <= ENTRY_CAP and total <= INITIAL_CAP, 'flat initial caps')
    require(set(seen) == expected, 'no missing or extra ROOT names')
    for name, identity in seen.items():
        remaining(REMOTE_END)
        require(identity == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)),
                'stable flat linked leaf')
    require(stamp(before) == stamp(os.fstat(rootfd))
            == stamp(os.stat(ROOT, follow_symlinks=False)), 'stable linked flat root')
    # Exclusive-root duty continues through exit, including permitted later writes.
    return dict(entries=len(seen), bytes=total)

def observe_exit(proc, status, code):
    # Called immediately after each successful actual poll/wait, before any writes.
    if code is not None and status['actual_exit'] is None:
        status['actual_exit'] = code
        status['reaped'] = True
        status['actual_exit_observed_ns'] = clock_ns()
        return status['actual_exit_observed_ns']
    return clock_ns()  # Sample even a real poll result of None; do not infer exit.

            and type(status['actual_exit_observed_ns']) is int

def closed_json(data):
    def pairs(items):
        out = {}
        for key, value in items:
            require(key not in out, 'duplicate JSON key')
            out[key] = value
        return out
    def nonfinite(_value):
        raise Stop('nonfinite JSON constant')
    def finite_float(value):
        result = float(value)
        require(math.isfinite(result), 'nonfinite JSON number')
        return result
    return json.loads(data, object_pairs_hook=pairs, parse_constant=nonfinite, parse_float=finite_float)

def initial_put(rootfd, name, data, budget):
    require(name in OUTPUTS[:4] and len(data) <= FILE_CAP, 'only pre-primary INITIAL outputs')
    item = dict(path=ROOT + '/' + name, state='attempted', ref=None)
    PUBLICATIONS[name] = item
    fd = None
    try:
        with until(REMOTE_END):
            absent(rootfd, 'bytes.json')
            absent(rootfd, 'bytes.lock')
            # Reserve the exact growth before exclusive create; never refund a partial.
            require(budget['bytes'] + len(data) <= INITIAL_CAP
                    and budget['entries'] + 1 <= ENTRY_CAP, 'INITIAL pre-write reservation')
            budget['bytes'] += len(data)
            budget['entries'] += 1
            fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW,
                         0o600, dir_fd=rootfd)
            os.fchmod(fd, 0o600)
            require(os.write(fd, data) == len(data), 'short raw INITIAL write; no retry')
            os.fsync(fd)
            closing, fd = fd, None
            os.close(closing)
            os.fsync(rootfd)
        item.update(state='durable', mode='0o600', ref=dict(path=ROOT + '/' + name,
                                             bytes=len(data), sha256=sha(data)))
        return item['ref']
    except BaseException as exc:
        fail('initial-publication', exc)
        item['state'] = 'failed_or_partial_unverified'
        PUBLICATION_ERRORS.append(dict(name=name, error=type(exc).__name__))
        raise
    finally:
        if fd is not None:
            closing, fd = fd, None
            try:
                os.close(closing)
            except BaseException as exc:
                fail('initial-close', exc)
                PUBLICATION_ERRORS.append(dict(name=name, error=type(exc).__name__, stage='close'))

def common_deadline(common, deadline, conversions):
    # Cleared R1 method: common local FIRST, shared AFTER; discard the gap.
    shared_before = clock_ns()
    local_before = common.time.monotonic()
    shared_after = clock_ns()
    budget = deadline - shared_after - 1000000
    require(shared_before <= shared_after and budget > 0 and math.isfinite(local_before),
            'conservative common before/after bounds')
    duration = math.nextafter(budget / NS, -math.inf)
    local_end = math.nextafter(local_before + duration, -math.inf)
    require(math.isfinite(local_end) and local_end > local_before, 'common safe local deadline')
    remaining(deadline)
    conversions.append(dict(shared_before_ns=shared_before, local_before=local_before,
                            shared_after_ns=shared_after, shared_end_ns=deadline,
                            discarded_ns=1000000, local_end=local_end))
    return local_end

            and ledger['product_reserved'] == 16777216 and ledger['cleanup_reserved'] == 4194304
            and ledger['infrastructure_reserved'] == 65536, 'only valid prepare-created ledger')
    return dict(ledger=ref, lock=lock_ref, observed=ledger)

def canonical_prepared_directory(value, rootfd):
    require(type(value) is str and value.startswith('/') and '..' not in Path(value).parts,
            'new prepared canonical path')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        for part in value.split('/')[1:]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = child
        info = os.fstat(fd)
        require(info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o700
                and info.st_dev == os.fstat(rootfd).st_dev
                and stamp(info) == stamp(os.stat(value, follow_symlinks=False)),
                'new prepared directory owner/mode/filesystem')
        require(str(Path(value).resolve(strict=True)) == value, 'canonical prepared directory')
    finally:
        os.close(fd)

def collect(label):
    # Only these two vectors exist. Helpers have10s including capture/reap/close.
    # The prepare child keeps its own unchanged120s file budget. No renewed phase.
    started = clock_ns()
    require(label in ('entry', 'prepare'), 'closed file-only child kind')
    if label == 'entry':
        argv = [SYSTEM, '-I', '-S', '-B', ROOT + '/file-qualification.recipe.py.txt', ROOT, 'entry']
        end = min(started + 10 * NS, REMOTE_END - 2 * NS)
        work_end, reap_end = end - 500000000, end
    else:
        argv = [PRIVATE_PYTHON, '-B', ROOT + '/prepare.py']
        end = REMOTE_END
        work_end, reap_end = end - 2 * NS, end - 1500000000
    buffers = {'stdout': bytearray(), 'stderr': bytearray()}
    status = dict(schema='openergy.prepare-file-child.v1', label=label, clock=CLOCK,
        argv=argv, cwd=ROOT, environment=startup(), phase_started_ns=started,
        phase_deadline_ns=end, remote_started_ns=REMOTE_START, remote_deadline_ns=REMOTE_END,
        collection_end_ns=work_end, reap_end_ns=reap_end, launch_started_ns=None,
        launch_returned_ns=None, pid_from_retained_handle=None, handle_retained=False,
        actual_exit=None, actual_exit_observed_ns=None, reaped=False,
        eof={'stdout': False, 'stderr': False}, observed_bytes=0, dropped_known_bytes=0,
        unread_or_dropped_unknown=False, overflow=False, timed_out=False, deadline_loss=False,
        kill_attempted=False, kill_syscall_returned=False, kill_or_reap_error=None,
        kill_eligibility_exit=None, kill_eligibility_observed_ns=None,
        first_error=None, close_errors=[], finished_ns=None)
    proc, selector = None, None
    retained = dict(label=label, handle=None, status=status)
    HANDLES.append(retained)
    try:
        with until(work_end):
            if label == 'entry':
                require(started + 750000000 < work_end, 'helper remaining admission')
            status['launch_started_ns'] = clock_ns()
            if label == 'prepare':
                require(status['launch_started_ns'] + 120 * NS + 500000000 < work_end,
                        'late admission cannot fit original prepare120; no spawn')
            proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL,
                stdout=subprocess.PIPE, stderr=subprocess.PIPE, cwd=ROOT,
                env=startup(), shell=False, close_fds=True, bufsize=0)
            retained['handle'] = proc
            status['handle_retained'] = True
            status['pid_from_retained_handle'] = proc.pid
            status['launch_returned_ns'] = clock_ns()
            require(status['launch_returned_ns'] < work_end, 'late spawn return')
            selector = selectors.DefaultSelector()
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                os.set_blocking(stream.fileno(), False)
                selector.register(stream, selectors.EVENT_READ, name)
            while True:
                code = proc.poll()
                observe_exit(proc, status, code)
                if code is not None and all(status['eof'].values()):
                    break
                require(clock_ns() < work_end, 'original child work ceiling')
                for key, _ in selector.select(min(0.05, remaining(work_end))):
                    allowance = RAW_CAP - sum(len(v) for v in buffers.values())
                    try:
                        chunk = os.read(key.fd, min(65536, allowance + 1))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        status['eof'][key.data] = True
                        selector.unregister(key.fileobj)
                        continue
                    status['observed_bytes'] += len(chunk)
                    buffers[key.data].extend(chunk[:allowance])
                    if len(chunk) > allowance:
                        status['overflow'] = True
                        status['dropped_known_bytes'] += len(chunk) - allowance
                        status['unread_or_dropped_unknown'] = True
                        raise Stop('combined1MiB while-read cap; no drain')
    except BaseException as exc:
        latch(status, 'collect', exc)
        fail(label + '-collection', exc)  # FIRST actual failure BEFORE any publication.
        if isinstance(exc, (TimeoutError, subprocess.TimeoutExpired)):
            status['timed_out'] = True
        try:
            if clock_ns() >= work_end:
                status['timed_out'] = True
        except BaseException:
            status['deadline_loss'] = True
    finally:
        if proc is not None and not status['reaped']:
            try:
                # Actual code is retained even if late. No prior timeout assertion
                # can skip this final nonblocking poll or manufacture reap.
                code = proc.poll()
                observed_ns = observe_exit(proc, status, code)
                status['kill_eligibility_exit'] = code
                status['kill_eligibility_observed_ns'] = observed_ns
                if code is None:
                    if not (observed_ns < reap_end and observed_ns < REMOTE_END):
                        status['timed_out'] = True
                        status['unread_or_dropped_unknown'] = True
                        raise TimeoutError('late post-poll sample; no signal/wait extension')
                    if status['overflow'] or status['timed_out']:
                        # Fresh actual None and its immediate clock are both retained.
                        # No intervening OS call before the sole direct KILL.
                        status['kill_attempted'] = True
                        os.kill(proc.pid, signal.SIGKILL)  # PID only from this still-unreaped handle.
                        status['kill_syscall_returned'] = True  # Return only; no delivery/reap claim.
                    with until(reap_end):
                        code = proc.wait(timeout=remaining(reap_end))
                        observe_exit(proc, status, code)
            except BaseException as exc:
                status['kill_or_reap_error'] = type(exc).__name__
                latch(status, 'kill-or-reap', exc)
                fail(label + '-kill-or-reap', exc)
        if proc is not None:
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                if stream is not None:
                    try:
                        stream.close()
                    except BaseException as exc:
                        status['close_errors'].append(dict(stream=name, error=type(exc).__name__))
                        latch(status, 'pipe-close', exc)
                        fail(label + '-pipe-close', exc)
        if selector is not None:
            try:
                selector.close()
            except BaseException as exc:
                status['close_errors'].append(dict(stream='selector', error=type(exc).__name__))
                latch(status, 'selector-close', exc)
                fail(label + '-selector-close', exc)
    status['unread_or_dropped_unknown'] |= not all(status['eof'].values())
    status['retained_bytes'] = {name: len(data) for name, data in buffers.items()}
    status['finished_ns'] = clock_ns()
    status['deadline_loss'] |= status['finished_ns'] >= end
    if not collection_complete(status):
        latch(status, 'incomplete-actual-collection', Stop('incomplete'))
        fail(label + '-actual-status', Stop('incomplete'))
    # Completed status/raw buffers live on retained handle record BEFORE writers.
    return status, bytes(buffers['stdout']), bytes(buffers['stderr'])


def emit_once(receipt_ref, complete):
    # One <=512 atomic nonblocking raw write; no retry, flush or shutdown output.
    value = dict(v='prepare-parent.v1', state=('PREPARED_DATA_COMPLETE' if complete else 'FAILED_UNVERIFIED'),
                 r=([receipt_ref['bytes'], receipt_ref['sha256']] if receipt_ref else None))
    data = encoded(value)
    require(len(data) <= CONSOLE_CAP, 'compact console cap')
    with until(REMOTE_END):
        require(stat.S_ISFIFO(os.fstat(1).st_mode) and os.get_blocking(1) is False
                and os.fpathconf(1, 'PC_PIPE_BUF') >= CONSOLE_CAP, 'same admitted console FIFO')
        remaining(REMOTE_END)
        require(os.write(1, data) == len(data), 'single console write failed; no retry')

        with until(REMOTE_END):
            ledger = existing_ledger(rootfd)
        reserved_put(rootfd, common, OUTPUTS[4], stdout, conversions)
        reserved_put(rootfd, common, OUTPUTS[5], stderr, conversions)
        reserved_put(rootfd, common, OUTPUTS[6], encoded(primary), conversions)
        if collection_complete(primary) and FIRST_ERROR is None:
            STAGE = 'new-prepared-data-binding'
            try:
                with until(REMOTE_END):
                    prepared = inspect_prepared(rootfd, artifact, bundle_tree)
            except BaseException as exc:
                fail(STAGE, exc)
        with until(REMOTE_END):
            final_nice = os.getpriority(os.PRIO_PROCESS, 0)
            require(final_nice == 15, 'own NI15 final observation')
        complete = (collection_complete(primary) and FIRST_ERROR is None and prepared is not None
                    and not PUBLICATION_ERRORS)
        STAGE = 'full-R9-receipt'
        receipt = dict(schema='openergy.prepare-parent-receipt.v1', identity=identity,
            evidence_root=ROOT, clock=CLOCK, remote_started_ns=REMOTE_START,
            caps=dict(file=FILE_CAP, aggregate=TOTAL_CAP, INITIAL=INITIAL_CAP,
                      combined_each_child=RAW_CAP, console=CONSOLE_CAP, full_receipt=RECEIPT_CAP,
                      product_reserve=16777216, cleanup_reserve=4194304, accounting_reserve=65536),
            remote_deadline_ns=REMOTE_END, remote_collection_end_ns=REMOTE_END - 2 * NS,
            remote_reap_end_ns=REMOTE_END - 1500000000, helper=auth,
            primary=primary, primary_attempted=primary_attempted, actual_handles=[
                dict(label=e['label'], pid_from_retained_handle=(e['handle'].pid if e['handle'] else None),
                     actual_exit=(e['handle'].returncode if e['handle'] else None), status=e['status'])
                for e in HANDLES],
            first_error=FIRST_ERROR, publication_errors=PUBLICATION_ERRORS,
            publications=PUBLICATIONS, prepared=prepared, initial_tally=initial,
            preprimary_INITIAL_charged_by_unchanged_prepare=budget, ledger_before_parent_writes=ledger,
            common_clock_conversions_before_receipt=list(conversions),
            receipt_writer_boundary='same conservative conversion at reserved_put; outer end unchanged',
            fixed_artifact_files=EIGHT, frozen17=PINS,
            source_result=SOURCE_RESULT, source_raw_manifest_sha256=sha(encoded(SOURCE_RAWS)),
            source_boundaries=CONFIG['completed_SOURCE'], artifact_publication=CONFIG['artifact_publication'],
            own_nice_before=NICE_BEFORE, own_nice_entry=OWN_NICE, own_nice_final=final_nice,
            console_admission=console, state=('PREPARED_DATA_COMPLETE' if complete else 'FAILED_UNVERIFIED'),
            execution=False, flow_started=False, copy_qualification='STILL_REQUIRED',
            artifact_runtime_acceptance=False, runtime='UNVERIFIED',
            own_exit_EOF_SSH_collector_D4_and_final_barriers='external actual parent duty')
        receipt_ref = reserved_put(rootfd, common, OUTPUTS[7], receipt, conversions)
    except BaseException as exc:
        fail(STAGE, exc)
        complete = False
    # No second writer or failure-path file publication. Existing partial outputs
    # remain. Missing receipt/raw/ledger/exit is declared FAILED_UNVERIFIED below.
    if rootfd is not None:
        try:
            closing, rootfd = rootfd, None
            with until(REMOTE_END):
                os.close(closing)
        except BaseException as exc:
            fail('root-close', exc)
            complete = False
    try:
        require(CONFIG is not None and console is not None, 'no admitted console')
        emit_once(receipt_ref, complete and FIRST_ERROR is None)
        remaining(REMOTE_END)
    except BaseException:
        return 1
    return 0 if complete and FIRST_ERROR is None else 1
