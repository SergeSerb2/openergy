### import_common
def import_common(rootfd, authentication):
    require(authentication is not None and FIRST_ERROR is None, 'authenticate before module use')
    # Repeat exact pins immediately before the only supplied-module import.
    for name, binding in PINS.items():
        read_leaf(rootfd, name, binding)
    # -B suppresses cache writes, not reads. Keep ROOT exclusive through exit.
    for name in ('__pycache__', 'common.pyc', 'common.pyo'):
        absent(rootfd, name)  # Reject only; no cache deletion, root repair or retry.
    require('common' not in sys.modules, 'no prior common module')
    os.environ['OPENERGY_ORDINARY_SOURCE_ROOT'] = SOURCE_ROOT
    os.environ['OPENERGY_ORDINARY_PYTHON'] = PRIVATE_PYTHON
    with until(REMOTE_END):
        spec = importlib.util.spec_from_file_location('common', ROOT + '/common.py')
        module = importlib.util.module_from_spec(spec)
        sys.modules['common'] = module
        spec.loader.exec_module(module)  # Exact pinned common ONLY, after SYSTEM authentication.
        require(module.HERE == Path(ROOT) and str(module.REPO) == SOURCE_ROOT
                and module.PYTHON == PRIVATE_PYTHON
                and module.LOG_CAP == FILE_CAP and module.EVIDENCE_CAP == TOTAL_CAP
                and module.PRODUCT_RESERVE == 16777216 and module.CLEANUP_RESERVE == 4194304
                and module.ACCOUNTING_RESERVE == 65536, 'unchanged SYSTEM common binding/constants')
    return module

### collect
def collect(label):
    # Only these two vectors exist. Helpers have10s including capture/reap/close.
    # The prepare child keeps its own unchanged120s file budget. No renewed phase.
    started = clock_ns()
    require(label in ('entry', 'prepare'), 'closed file-only child kind')
    if label == 'entry':
        argv = [SYSTEM, '-I', '-S', '-B', ROOT + '/file-qualification.recipe.py.txt', ROOT, 'entry']
        end = min(started + 10 * NS, REMOTE_END - 2 * NS)
        work_end, reap_end = end - 500000000, end
    else:
        argv = [PRIVATE_PYTHON, '-B', ROOT + '/prepare.py']
        end = REMOTE_END
        work_end, reap_end = end - 2 * NS, end - 1500000000
    buffers = {'stdout': bytearray(), 'stderr': bytearray()}
    status = dict(schema='openergy.prepare-file-child.v1', label=label, clock=CLOCK,
        argv=argv, cwd=ROOT, environment=startup(), phase_started_ns=started,
        phase_deadline_ns=end, remote_started_ns=REMOTE_START, remote_deadline_ns=REMOTE_END,
        collection_end_ns=work_end, reap_end_ns=reap_end, launch_started_ns=None,
        launch_returned_ns=None, pid_from_retained_handle=None, handle_retained=False,
        actual_exit=None, actual_exit_observed_ns=None, reaped=False,
        eof={'stdout': False, 'stderr': False}, observed_bytes=0, dropped_known_bytes=0,
        unread_or_dropped_unknown=False, overflow=False, timed_out=False, deadline_loss=False,
        kill_attempted=False, kill_delivered=False, kill_or_reap_error=None,
        kill_eligibility_exit=None, kill_eligibility_observed_ns=None,
        first_error=None, close_errors=[], finished_ns=None)
    proc, selector = None, None
    retained = dict(label=label, handle=None, status=status)
    HANDLES.append(retained)
    try:
        with until(work_end):
            if label == 'entry':
                require(started + 750000000 < work_end, 'helper remaining admission')
            status['launch_started_ns'] = clock_ns()
            if label == 'prepare':
                require(status['launch_started_ns'] + 120 * NS + 500000000 < work_end,
                        'late admission cannot fit original prepare120; no spawn')
            proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL,
                stdout=subprocess.PIPE, stderr=subprocess.PIPE, cwd=ROOT,
                env=startup(), shell=False, close_fds=True, bufsize=0)
            retained['handle'] = proc
            status['handle_retained'] = True
            status['pid_from_retained_handle'] = proc.pid
            status['launch_returned_ns'] = clock_ns()
            require(status['launch_returned_ns'] < work_end, 'late spawn return')
            selector = selectors.DefaultSelector()
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                os.set_blocking(stream.fileno(), False)
                selector.register(stream, selectors.EVENT_READ, name)
            while True:
                code = proc.poll()
                observe_exit(proc, status, code)
                if code is not None and all(status['eof'].values()):
                    break
                require(clock_ns() < work_end, 'original child work ceiling')
                for key, _ in selector.select(min(0.05, remaining(work_end))):
                    allowance = RAW_CAP - sum(len(v) for v in buffers.values())
                    try:
                        chunk = os.read(key.fd, min(65536, allowance + 1))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        status['eof'][key.data] = True
                        selector.unregister(key.fileobj)
                        continue
                    status['observed_bytes'] += len(chunk)
                    buffers[key.data].extend(chunk[:allowance])
                    if len(chunk) > allowance:
                        status['overflow'] = True
                        status['dropped_known_bytes'] += len(chunk) - allowance
                        status['unread_or_dropped_unknown'] = True
                        raise Stop('combined1MiB while-read cap; no drain')
    except BaseException as exc:
        latch(status, 'collect', exc)
        fail(label + '-collection', exc)  # FIRST actual failure BEFORE any publication.
        if isinstance(exc, (TimeoutError, subprocess.TimeoutExpired)):
            status['timed_out'] = True
        try:
            if clock_ns() >= work_end:
                status['timed_out'] = True
        except BaseException:
            status['deadline_loss'] = True
    finally:
        if proc is not None and not status['reaped']:
            try:
                # Actual code is retained even if late. No prior timeout assertion
                # can skip this final nonblocking poll or manufacture reap.
                code = proc.poll()
                observed_ns = observe_exit(proc, status, code)
                status['kill_eligibility_exit'] = code
                status['kill_eligibility_observed_ns'] = observed_ns
                if code is None:
                    if not (observed_ns < reap_end and observed_ns < REMOTE_END):
                        status['timed_out'] = True
                        status['unread_or_dropped_unknown'] = True
                        raise TimeoutError('late post-poll sample; no signal/wait extension')
                    if status['overflow'] or status['timed_out']:
                        # Fresh actual None and its immediate clock are both retained.
                        # No intervening OS call before the sole direct KILL.
                        status['kill_attempted'] = True
                        proc.kill()
                        status['kill_delivered'] = True
                    with until(reap_end):
                        code = proc.wait(timeout=remaining(reap_end))
                        observe_exit(proc, status, code)
            except BaseException as exc:
                status['kill_or_reap_error'] = type(exc).__name__
                latch(status, 'kill-or-reap', exc)
                fail(label + '-kill-or-reap', exc)
        if proc is not None:
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                if stream is not None:
                    try:
                        stream.close()
                    except BaseException as exc:
                        status['close_errors'].append(dict(stream=name, error=type(exc).__name__))
                        latch(status, 'pipe-close', exc)
                        fail(label + '-pipe-close', exc)
        if selector is not None:
            try:
                selector.close()
            except BaseException as exc:
                status['close_errors'].append(dict(stream='selector', error=type(exc).__name__))
                latch(status, 'selector-close', exc)
                fail(label + '-selector-close', exc)
    status['unread_or_dropped_unknown'] |= not all(status['eof'].values())
    status['retained_bytes'] = {name: len(data) for name, data in buffers.items()}
    status['finished_ns'] = clock_ns()
    status['deadline_loss'] |= status['finished_ns'] >= end
    if not collection_complete(status):
        latch(status, 'incomplete-actual-collection', Stop('incomplete'))
        fail(label + '-actual-status', Stop('incomplete'))
    # Completed status/raw buffers live on retained handle record BEFORE writers.
    return status, bytes(buffers['stdout']), bytes(buffers['stderr'])

### main
def main():
    global CONFIG, STAGE
    rootfd, common, identity, initial, budget = None, None, None, None, None
    auth, primary, prepared, receipt_ref = None, None, None, None
    conversions, ledger = [], None
    final_nice, console = None, None
    primary_attempted = False
    complete = False
    try:
        require(sys.argv == ['-'] and sys.version_info[:3] == (3, 9, 6)
                and sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode
                and not sys.flags.optimize, 'fixed trusted SYSTEM3.9.6 stdin -I -S -B')
        require(os.getuid() == os.geteuid(), 'same real/effective owner')
        os.umask(0o077)
        with until(REMOTE_END):
            CONFIG, identity = bind_config()  # Null-closed BEFORE private root access.
            console = console_admission()  # BEFORE ANY target writes.
            rootfd = open_root()
            STAGE = 'fixed-source-artifact-boundaries'
            artifact, bundle_tree = inspect_frozen(rootfd)
            # Exclude every ordinary ROOT common cache before supplied module/private entry.
            for name in OUTPUTS + ('__pycache__', 'common.pyc', 'common.pyo',
                                    'bytes.json', 'bytes.lock', 'prepared.json', 'prepare-guard.json'):
                absent(rootfd, name)
            initial = inventory(rootfd)  # Complete count BEFORE first growth.
            budget = dict(initial)
            guard_data = encoded(dict(schema='openergy.prepare-parent-once.v1', identity=identity,
                clock=CLOCK, remote_started_ns=REMOTE_START, remote_deadline_ns=REMOTE_END,
                guard_before_all_spawns=True, helper_maximum=1, primary_maximum=1,
                primary_argv=[PRIVATE_PYTHON, '-B', ROOT + '/prepare.py'],
                environment=startup(), nice_before=NICE_BEFORE, own_nice=OWN_NICE,
                execution=False, flow_started=False))
            require(len(guard_data) <= STATUS_CAP and initial['entries'] + 4 <= ENTRY_CAP
                    and initial['bytes'] + len(guard_data) + RAW_CAP + STATUS_CAP <= INITIAL_CAP,
                    'reserve guard+combined helper raw+status BEFORE first write')
        STAGE = 'exclusive-once-guard'
        initial_put(rootfd, OUTPUTS[0], guard_data, budget)
        STAGE = 'SYSTEM-entry-helper'
        status, stdout, stderr = collect('entry')
        require(status['reaped'] is True, 'unreaped helper: no later writes')
        # Even a failed helper keeps exact bounded raw/status if safe. Its first
        # actual failure is already latched. No private entry follows any error.
        initial_put(rootfd, OUTPUTS[1], stdout, budget)
        initial_put(rootfd, OUTPUTS[2], stderr, budget)
        require(len(encoded(status)) <= STATUS_CAP, 'helper status reserved bound')
        initial_put(rootfd, OUTPUTS[3], encoded(status), budget)
        with until(REMOTE_END):
            auth = inspect_entry(status, stdout, stderr)
            require(FIRST_ERROR is None, 'failed helper is not authentication')
        STAGE = 'authenticated-SYSTEM-common'
        common = import_common(rootfd, auth)
        with until(REMOTE_END):
            # Rebind fixed operation/artifact and modules just before private entry.
            inspect_artifact(rootfd)
            for name, binding in PINS.items():
                read_leaf(rootfd, name, binding)
            absent(rootfd, 'bytes.json')
            absent(rootfd, 'bytes.lock')
            # Recheck before the irreversible attempt; ownership stays exclusive through exit.
            for name in ('__pycache__', 'common.pyc', 'common.pyo'):
                absent(rootfd, name)
        STAGE = 'one-private-prepare'
        primary_attempted = True  # Irreversible mode boundary, even spawn failure.
        primary, stdout, stderr = collect('prepare')
        # No post-child writer touches ROOT until exact actual status is closed.
        require(primary['reaped'] is True and type(primary['actual_exit']) is int,
                'unreaped prepare: raw unpublished/unknown; no writer')
        # FIRST_ERROR and the full primary record were fixed BEFORE all calls below.
        STAGE = 'existing-R9-reserved-publication'
        with until(REMOTE_END):
            ledger = existing_ledger(rootfd)
        reserved_put(rootfd, common, OUTPUTS[4], stdout, conversions)
        reserved_put(rootfd, common, OUTPUTS[5], stderr, conversions)
        reserved_put(rootfd, common, OUTPUTS[6], encoded(primary), conversions)
        if collection_complete(primary) and FIRST_ERROR is None:
            STAGE = 'new-prepared-data-binding'
            try:
                with until(REMOTE_END):
                    prepared = inspect_prepared(rootfd, artifact, bundle_tree)
            except BaseException as exc:
                fail(STAGE, exc)
        with until(REMOTE_END):
            final_nice = os.getpriority(os.PRIO_PROCESS, 0)
            require(final_nice == 15, 'own NI15 final observation')
        complete = (collection_complete(primary) and FIRST_ERROR is None and prepared is not None
                    and not PUBLICATION_ERRORS)
        STAGE = 'full-R9-receipt'
        receipt = dict(schema='openergy.prepare-parent-receipt.v1', identity=identity,
            evidence_root=ROOT, clock=CLOCK, remote_started_ns=REMOTE_START,
            caps=dict(file=FILE_CAP, aggregate=TOTAL_CAP, INITIAL=INITIAL_CAP,
                      combined_each_child=RAW_CAP, console=CONSOLE_CAP, full_receipt=RECEIPT_CAP,
                      product_reserve=16777216, cleanup_reserve=4194304, accounting_reserve=65536),
            remote_deadline_ns=REMOTE_END, remote_collection_end_ns=REMOTE_END - 2 * NS,
            remote_reap_end_ns=REMOTE_END - 1500000000, helper=auth,
            primary=primary, primary_attempted=primary_attempted, actual_handles=[
                dict(label=e['label'], pid_from_retained_handle=(e['handle'].pid if e['handle'] else None),
                     actual_exit=(e['handle'].returncode if e['handle'] else None), status=e['status'])
                for e in HANDLES],
            first_error=FIRST_ERROR, publication_errors=PUBLICATION_ERRORS,
            publications=PUBLICATIONS, prepared=prepared, initial_tally=initial,
            preprimary_INITIAL_charged_by_unchanged_prepare=budget, ledger_before_parent_writes=ledger,
            common_clock_conversions_before_receipt=list(conversions),
            receipt_writer_boundary='same conservative conversion at reserved_put; outer end unchanged',
            fixed_artifact_files=EIGHT, frozen17=PINS,
            source_result=SOURCE_RESULT, source_raw_manifest_sha256=sha(encoded(SOURCE_RAWS)),
            source_boundaries=CONFIG['completed_SOURCE'], artifact_publication=CONFIG['artifact_publication'],
            own_nice_before=NICE_BEFORE, own_nice_entry=OWN_NICE, own_nice_final=final_nice,
            console_admission=console, state=('PREPARED_DATA_COMPLETE' if complete else 'FAILED_UNVERIFIED'),
            execution=False, flow_started=False, copy_qualification='STILL_REQUIRED',
            artifact_runtime_acceptance=False, runtime='UNVERIFIED',
            own_exit_EOF_SSH_collector_D4_and_final_barriers='external actual parent duty')
        receipt_ref = reserved_put(rootfd, common, OUTPUTS[7], receipt, conversions)
    except BaseException as exc:
        fail(STAGE, exc)
        complete = False
    # No second writer or failure-path file publication. Existing partial outputs
    # remain. Missing receipt/raw/ledger/exit is declared FAILED_UNVERIFIED below.
    if rootfd is not None:
        try:
            closing, rootfd = rootfd, None
            with until(REMOTE_END):
                os.close(closing)
        except BaseException as exc:
            fail('root-close', exc)
            complete = False
    try:
        require(CONFIG is not None and console is not None, 'no admitted console')
        emit_once(receipt_ref, complete and FIRST_ERROR is None)
        remaining(REMOTE_END)
    except BaseException:
        return 1
    return 0 if complete and FIRST_ERROR is None else 1
