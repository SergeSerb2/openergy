# observe_exit -- new source
def observe_exit(proc, status, code):
    # Called immediately after each successful actual poll/wait, before any writes.
    if code is not None and status['actual_exit'] is None:
        status['actual_exit'] = code
        status['reaped'] = True
        if code != 0:
            # Preserve the actual failed COMMAND before a clock/close/publisher
            # exception can become the first cause. Signed code is already saved.
            latch(status, 'command-exit', Stop('actual nonzero child exit'))
            fail(status['label'] + '-command-exit', Stop('actual nonzero child exit'))
        status['actual_exit_observed_ns'] = clock_ns()
        return status['actual_exit_observed_ns']
    return clock_ns()  # Sample even a real poll result of None; do not infer exit.

# closed_root_layout -- new source
def closed_root_layout(rootfd):
    # R7 flat admission, extended ONLY by this operation's own durable DATA leaves.
    require(len(ROOT_INITIAL_NAMES) == 260 and not (set(ROOT_PARENT_ONLY) & ROOT_INITIAL_NAMES),
            'remote five / local two initial set')
    require(set(PUBLICATIONS) <= set(OUTPUTS), 'four fixed DATA outputs only')
    for name, item in PUBLICATIONS.items():
        require(item['state'] == 'durable' and item['ref'] is not None, 'durable outputs only')
        read_leaf(rootfd, name, item['ref'])
    expected = ROOT_INITIAL_NAMES | frozenset(PUBLICATIONS)
    before = os.fstat(rootfd)
    require(stat.S_ISDIR(before.st_mode) and before.st_uid == os.getuid()
            and stat.S_IMODE(before.st_mode) == 0o700, 'flat owned ROOT700')
    seen, total = {}, 0
    with os.scandir(rootfd) as entries:
        for entry in entries:
            remaining(PHASE_END)
            name = entry.name
            require(name in expected and name not in seen and len(seen) < len(expected)
                    and len(name.encode('utf-8')) <= 255, 'exact flat name set')
            info = os.stat(name, dir_fd=rootfd, follow_symlinks=False)
            require(stat.S_ISREG(info.st_mode) and info.st_uid == before.st_uid
                    and info.st_dev == before.st_dev and stat.S_IMODE(info.st_mode) == 0o600
                    and info.st_nlink == 1 and 0 <= info.st_size <= FILE_CAP,
                    'sameFS regular0600 nlink1; no directories/links/caches/packages')
            seen[name] = stamp(info)
            total += info.st_size
            require(len(seen) <= ENTRY_CAP and total <= INITIAL_CAP, 'INITIAL count/actual byte tally')
    require(set(seen) == expected, 'no missing or additional leaf')
    for name, identity in seen.items():
        remaining(PHASE_END)
        if name in ROOT_OBSERVED_DATA:
            require(identity == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)),
                    'observed DATA enumerated identity')
            read_leaf(rootfd, name, ROOT_OBSERVED_DATA[name])
        require(identity == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)),
                'stable linked flat leaf')
    require(stamp(before) == stamp(os.fstat(rootfd))
            == stamp(os.stat(ROOT, follow_symlinks=False)), 'stable linked ROOT')
    return dict(entries=len(seen), bytes=total)

# initial_put -- new source
def initial_put(rootfd, name, data, budget):
    # Existing INITIAL exclusive writer closure. No ledger/common writer at any stage.
    require(name in OUTPUTS and name not in PUBLICATIONS and type(data) is bytes
            and len(data) <= OUTPUT_CAPS[OUTPUTS.index(name)] <= FILE_CAP, 'fixed DATA cap')
    require(all(e['status']['reaped'] is True and type(e['status']['actual_exit']) is int
                for e in HANDLES), 'unknown/unreaped child: no subsequent target publication')
    item = dict(path=ROOT + '/' + name, state='attempted', ref=None)
    PUBLICATIONS[name] = item
    fd = None
    try:
        with until(PHASE_END):
            for leaf in NO_LEDGER:
                absent(rootfd, leaf)
            # Charge actual bytes before exclusive growth; never reset/refund.
            require(budget['bytes'] + len(data) <= INITIAL_CAP
                    and budget['entries'] + 1 <= ENTRY_CAP, 'INITIAL pre-write reservation')
            budget['bytes'] += len(data)
            budget['entries'] += 1
            fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW,
                         0o600, dir_fd=rootfd)
            os.fchmod(fd, 0o600)
            before = os.fstat(fd)
            require(stat.S_ISREG(before.st_mode) and before.st_uid == os.getuid()
                    and stat.S_IMODE(before.st_mode) == 0o600 and before.st_nlink == 1
                    and before.st_dev == os.fstat(rootfd).st_dev and before.st_size == 0,
                    'exclusive INITIAL file identity')
            require(os.write(fd, data) == len(data), 'full single INITIAL write; no retry')
            os.fsync(fd)
            after = os.fstat(fd)
            require(after.st_size == len(data) and after.st_ino == before.st_ino
                    and after.st_dev == before.st_dev and after.st_uid == before.st_uid
                    and after.st_nlink == 1 and stat.S_IMODE(after.st_mode) == 0o600
                    and stamp(after) == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)),
                    'full INITIAL linked stamp before close')
            closing, fd = fd, None
            os.close(closing)
            os.fsync(rootfd)
            require(stamp(after) == stamp(os.stat(name, dir_fd=rootfd, follow_symlinks=False)),
                    'stable INITIAL linked stamp after directory barrier')
        ref = dict(path=ROOT + '/' + name, bytes=len(data), sha256=sha(data))
        with until(PHASE_END):
            read_leaf(rootfd, name, ref)
        item.update(state='durable', ref=ref)
        return ref
    except BaseException as exc:
        fail('initial-publication', exc)
        item['state'] = 'failed_or_partial_unverified'
        PUBLICATION_ERRORS.append(dict(name=name, stage='publication', error=type(exc).__name__))
        raise
    finally:
        if fd is not None:
            closing, fd = fd, None
            try:
                with until(PHASE_END):
                    os.close(closing)
            except BaseException as exc:
                fail('initial-close', exc)
                PUBLICATION_ERRORS.append(dict(name=name, stage='close', error=type(exc).__name__))


# collect -- new source
def collect():
    # R11 retained-handle/nonblocking combined1MiB collector, fixed nine-row interface.
    global FIRST_COMMAND_FAILURE
    require(PUBLICATIONS.get(OUTPUTS[0], {}).get('state') == 'durable'
            and PUBLICATIONS[OUTPUTS[0]]['ref'] is not None, 'durable exclusive guard BEFORE any Popen')
    require(len(OBSERVATIONS) < 9, 'fixed nine observations; no retry')
    spec = sequence()[len(OBSERVATIONS)]
    label, argv = spec['label'], spec['argv']
    started = clock_ns()
    end = min(started + 10 * NS, PHASE_END - NS)
    work_end = end - 500000000
    status = dict(schema='openergy.requirements-source-child.v1', label=label, argv=argv,
        cwd=ROOT, environment=spec['environment'], clock=CLOCK, phase_started_ns=started,
        phase_deadline_ns=end, collection_end_ns=work_end, reap_end_ns=end,
        remote_started_ns=REMOTE_START, remote_deadline_ns=REMOTE_END,
        launch_started_ns=None, launch_returned_ns=None, pid_from_retained_handle=None,
        handle_retained=False, actual_exit=None, actual_exit_observed_ns=None, reaped=False,
        eof={'stdout': False, 'stderr': False}, observed_bytes=0, dropped_known_bytes=0,
        retained_bytes={'stdout': 0, 'stderr': 0}, unread_or_dropped_unknown=False,
        overflow=False, timed_out=False, deadline_loss=False, kill_attempted=False,
        kill_syscall_returned=False, kill_eligibility_exit=None, kill_eligibility_observed_ns=None,
        kill_or_reap_error=None, first_error=None, close_errors=[], finished_ns=None)
    buffers = {'stdout': bytearray(), 'stderr': bytearray()}
    proc, selector = None, None
    retained = dict(label=label, argv=argv, handle=None, status=status, buffers=buffers)
    HANDLES.append(retained)  # Also records a failed spawn with actual_exit null.
    OBSERVATIONS.append(retained)
    try:
        with until(work_end):
            require(end - started > 750000000, 'original command/reap allowance')
            status['launch_started_ns'] = clock_ns()
            proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                stderr=subprocess.PIPE, cwd=ROOT, env=spec['environment'], shell=False,
                close_fds=True, bufsize=0)
            retained['handle'] = proc
            status['handle_retained'] = True
            status['pid_from_retained_handle'] = proc.pid
            status['launch_returned_ns'] = clock_ns()
            require(status['launch_returned_ns'] < work_end, 'late spawn return')
            selector = selectors.DefaultSelector()
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                os.set_blocking(stream.fileno(), False)
                selector.register(stream, selectors.EVENT_READ, name)
            while True:
                code = proc.poll()
                observe_exit(proc, status, code)  # Real signed code/reap BEFORE clock.
                if code is not None and all(status['eof'].values()):
                    break
                if clock_ns() >= work_end:
                    status['timed_out'] = status['deadline_loss'] = True
                    latch(status, 'work-deadline', TimeoutError('original work end'))
                    fail(label + '-work-deadline', TimeoutError('original work end'))
                    break
                if not selector.get_map():
                    try:
                        code = proc.wait(timeout=remaining(work_end))
                        observe_exit(proc, status, code)
                    except subprocess.TimeoutExpired:
                        status['timed_out'] = status['deadline_loss'] = True
                        latch(status, 'work-wait-deadline', TimeoutError('original work end'))
                        fail(label + '-work-wait-deadline', TimeoutError('original work end'))
                    continue
                for key, _mask in selector.select(min(0.05, remaining(work_end))):
                    allowance = RAW_CAP - sum(len(x) for x in buffers.values())
                    try:
                        chunk = os.read(key.fd, min(65536, allowance + 1))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        status['eof'][key.data] = True
                        selector.unregister(key.fileobj)
                        continue
                    status['observed_bytes'] += len(chunk)
                    buffers[key.data].extend(chunk[:allowance])
                    if len(chunk) > allowance:
                        status['dropped_known_bytes'] += len(chunk) - allowance
                        status['overflow'] = status['unread_or_dropped_unknown'] = True
                        latch(status, 'combined-stream-overflow', Stop('overflow'))
                        fail(label + '-combined-stream-overflow', Stop('overflow'))
                        break
                if status['overflow']:
                    break
    except BaseException as exc:
        latch(status, 'collect', exc)
        fail(label + '-collection', exc)
        if isinstance(exc, (TimeoutError, subprocess.TimeoutExpired)):
            status['timed_out'] = status['deadline_loss'] = True
    finally:
        if proc is not None:
            try:
                # No enclosing alarm/postcheck before this actual poll: retain real
                # exit even if the immediate shared-clock sample itself fails.
                code = proc.poll()
                observed_ns = observe_exit(proc, status, code)
                status['kill_eligibility_exit'] = code
                status['kill_eligibility_observed_ns'] = observed_ns
                if code is None:
                    if observed_ns < end and observed_ns < PHASE_END:
                        if status['overflow'] or status['timed_out']:
                            status['kill_attempted'] = True
                            os.kill(proc.pid, signal.SIGKILL)
                            status['kill_syscall_returned'] = True
                    else:
                        status['timed_out'] = status['deadline_loss'] = True
                        raise TimeoutError('late fresh None sample; no signal/wait extension')
                    with until(end):
                        code = proc.wait(timeout=remaining(end))
                        observe_exit(proc, status, code)
            except BaseException as exc:
                status['kill_or_reap_error'] = type(exc).__name__
                status['deadline_loss'] = True
                latch(status, 'kill-or-reap', exc)
                fail(label + '-kill-or-reap', exc)
            # No drain, second poll, signal retry, PID query, group or product signal.
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                if stream is not None:
                    try:
                        with until(end):
                            stream.close()
                    except BaseException as exc:
                        status['close_errors'].append(dict(stream=name, error=type(exc).__name__))
                        latch(status, 'pipe-close', exc)
                        fail(label + '-pipe-close', exc)
        if selector is not None:
            try:
                with until(end):
                    selector.close()
            except BaseException as exc:
                status['close_errors'].append(dict(stream='selector', error=type(exc).__name__))
                latch(status, 'selector-close', exc)
                fail(label + '-selector-close', exc)
    status['unread_or_dropped_unknown'] |= not all(status['eof'].values())
    status['retained_bytes'] = {name: len(data) for name, data in buffers.items()}
    try:
        status['finished_ns'] = clock_ns()
        status['deadline_loss'] |= status['finished_ns'] >= end or status['finished_ns'] >= PHASE_END
    except BaseException as exc:
        status['deadline_loss'] = True
        latch(status, 'final-clock', exc)
        fail(label + '-final-clock', exc)
    if not collection_complete(status):
        latch(status, 'incomplete-actual-collection', Stop('incomplete'))
        if FIRST_COMMAND_FAILURE is None:
            FIRST_COMMAND_FAILURE = dict(label=label, status=closed_json(encoded(status)))
        fail(label + '-actual-status', Stop('incomplete'))
    # All status/raw prefix bytes are retained in RAM before any later writer.
    return retained

# emit_once -- new source
def emit_once(receipt_ref, complete_data):
    # The compact record still identifies known signed exits and missing raws
    # when a full pack does not fit. It never invents a full failure pack.
    mask = lambda predicate: sum(1 << i for i, e in enumerate(OBSERVATIONS) if predicate(e['status']))
    data = encoded(dict(v='requirements-source.v1',
        state=('COLLECTED_DATA_ONLY' if complete_data else 'FAILED_UNVERIFIED'),
        r=([receipt_ref['bytes'], receipt_ref['sha256']] if receipt_ref else None),
        c=[e['status']['actual_exit'] for e in OBSERVATIONS],
        n=len(OBSERVATIONS), reaped=mask(lambda s: s['reaped']),
        eof=mask(lambda s: all(s['eof'].values())), kill=mask(lambda s: s['kill_attempted']),
        loss=mask(lambda s: s['overflow'] or s['unread_or_dropped_unknown'] or s['deadline_loss']),
        p=[{'not_attempted': 0, 'attempted': 1, 'durable': 2,
            'failed_or_partial_unverified': 3}[PUBLICATIONS.get(name, {}).get('state', 'not_attempted')]
           for name in OUTPUTS],
        raws_unpublished=PUBLICATIONS.get(OUTPUTS[1], {}).get('state') != 'durable',
        f=([FIRST_ERROR['stage'][:32], FIRST_ERROR['error'][:24]] if FIRST_ERROR else None)))
    require(len(data) <= CONSOLE_CAP, 'compact FIFO512')
    with until(REMOTE_END):
        require(stat.S_ISFIFO(os.fstat(1).st_mode) and os.get_blocking(1) is False
                and os.fpathconf(1, 'PC_PIPE_BUF') >= CONSOLE_CAP, 'same admitted nonblocking FIFO')
        remaining(REMOTE_END)
        require(os.write(1, data) == len(data), 'exact one raw console write; no retry')


# main -- new source
def main():
    global CONFIG, STAGE
    rootfd, identity, console, budget = None, None, None, None
    wrapper, projection, objectfacts = None, None, []
    before, after, capture_ref, basis_ref, receipt_ref = None, None, None, None, None
    final_nice, closing_tally = None, None
    complete_data = False
    try:
        require(sys.argv == ['-'] and sys.version_info[:3] == (3, 9, 6)
                and sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode
                and not sys.flags.optimize, 'fixed SYSTEM3.9.6 -I -S -B stdin')
        require(os.getuid() == os.geteuid() and dict(os.environ) == startup(),
                'exact reviewed startup map; no bootstrap exception')
        os.umask(0o077)
        with until(PHASE_END):
            CONFIG, identity = bind_config()  # Null template rejects BEFORE any target access.
            console = console_admission()
            rootfd = open_root()
            initial = closed_root_layout(rootfd)
            require(initial == dict(entries=260, bytes=1242294), 'exact observed initial actual tally')
            require(inventory(rootfd) == initial, 'full INITIAL actual regular-byte tally')
            for name in OUTPUTS + NO_LEDGER + OLD_REMOTE_OUTPUTS:
                absent(rootfd, name)
            STAGE = 'authenticate-old-data-and-object-filefacts'
            wrapper, projection, objectfacts = inspect_frozen(rootfd)
            require(closed_root_layout(rootfd) == initial, 'same260 before first growth')
            require(initial['bytes'] + sum(OUTPUT_CAPS) <= INITIAL_CAP
                    and initial['entries'] + 4 == 264 <= ENTRY_CAP,
                    'reserve four exact output caps; no maximum-stream-fit promise')
            budget = dict(initial)
            guard = encoded(dict(schema='openergy.requirements-source-once.v1', execution=False,
                identity=identity, evidence_root=ROOT, clock=CLOCK, remote_started_ns=REMOTE_START,
                remote_deadline_ns=REMOTE_END, phase_deadline_ns=PHASE_END,
                guard_before_any_Popen=True, observation_limit=9, sequence=sequence(),
                completed_seven_provenance=COMPLETED, parent_only=list(ROOT_PARENT_ONLY),
                own_nice_before=NICE_BEFORE, own_nice_entry=OWN_NICE))
        STAGE = 'exclusive-durable-once-guard'
        initial_put(rootfd, OUTPUTS[0], guard, budget)
        # Collection has no interleaved ROOT raw files or writers.
        STAGE = 'nine-fixed-observations'
        try:
            for ordinal in range(9):
                item = collect()
                require(item['status']['reaped'] is True
                        and type(item['status']['actual_exit']) is int, 'unknown/unreaped: STOP publication')
                require(collection_complete(item['status']), 'first failed command: stop later observations')
                with until(PHASE_END):
                    if ordinal == 0:
                        before = helper_facts(item)
                    elif ordinal == 8:
                        after = helper_facts(item)
                        require(same_data(before, after), 'before/after full source facts and hash equality')
                    else:
                        raw = bytes(item['buffers']['stdout'])
                        require(len(raw) > 0, 'missing requirements stdout body')
                        raw.decode('utf-8', errors='strict')
                        bytes(item['buffers']['stderr']).decode('utf-8', errors='strict')
        except BaseException as exc:
            fail(STAGE, exc)  # Latched before capture/basis/receipt writers.
        require(all(e['status']['reaped'] is True and type(e['status']['actual_exit']) is int
                    for e in HANDLES), 'unknown/unreaped child: no subsequent target publication')
        with until(PHASE_END):
            closing_tally = closed_root_layout(rootfd)  # Recheck both observed DATA pins AFTER collection.
            require(closing_tally == budget, 'same initial+guard actual count/bytes')
            final_nice = os.getpriority(os.PRIO_PROCESS, 0)
            require(final_nice == 15, 'actual own NI15 final sample; no continuous claim')
            capture_data = encoded(capture_value(identity, objectfacts, before, after))
            require(len(capture_data) <= OUTPUT_CAPS[1], 'capture pack overflow: fail closed; no fallback leaf')
        STAGE = 'publish-capture'
        capture_ref = initial_put(rootfd, OUTPUTS[1], capture_data, budget)
        if FIRST_ERROR is None:
            with until(PHASE_END):
                basis_data = encoded(basis_value(wrapper, projection, capture_ref))
                require(len(basis_data) <= OUTPUT_CAPS[2], 'basis overflow: known data remain partial')
            STAGE = 'publish-basis'
            basis_ref = initial_put(rootfd, OUTPUTS[2], basis_data, budget)
        complete_data = (FIRST_ERROR is None and len(OBSERVATIONS) == 9 and basis_ref is not None
                         and not PUBLICATION_ERRORS and same_data(before, after))
        STAGE = 'publish-receipt'
        with until(PHASE_END):
            receipt = dict(schema='openergy.requirements-source-receipt.v1', execution=False,
                evidence_root=ROOT, identity=identity, clock=CLOCK, remote_started_ns=REMOTE_START,
                remote_deadline_ns=REMOTE_END, phase_deadline_ns=PHASE_END,
                guard=PUBLICATIONS[OUTPUTS[0]]['ref'], capture=capture_ref, basis=basis_ref,
                first_error=FIRST_ERROR, first_command_failure=FIRST_COMMAND_FAILURE,
                publication_errors=list(PUBLICATION_ERRORS), publications=closed_json(encoded(PUBLICATIONS)),
                actual_direct_children=direct_children(), initial_tally=initial,
                charged_INITIAL_before_receipt=dict(budget), output_caps=dict(zip(OUTPUTS, OUTPUT_CAPS)),
                caps=dict(INITIAL=INITIAL_CAP, entries=ENTRY_CAP, file=FILE_CAP, total=TOTAL_CAP,
                          combined_each_child=RAW_CAP, product=16777216, cleanup=4194304, accounting=65536),
                completed_seven_provenance=COMPLETED, parent_only=list(ROOT_PARENT_ONLY),
                own_nice_before=NICE_BEFORE, own_nice_entry=OWN_NICE, own_nice_final=final_nice,
                console=console, collection_complete_data_only=complete_data,
                state=('COLLECTED_DATA_ONLY' if complete_data else 'FAILED_UNVERIFIED'),
                external_actual_capsule_exit=None, external_EOF=None, external_D4=None,
                artifact_acceptance=None, requirements_semantic_acceptance=None,
                runtime_acceptance=None, runtime='UNVERIFIED', updater='CLOSED')
            receipt_data = encoded(receipt)
            require(len(receipt_data) <= OUTPUT_CAPS[3], 'receipt overflow; no guaranteed failure pack')
        receipt_ref = initial_put(rootfd, OUTPUTS[3], receipt_data, budget)
        with until(PHASE_END):
            require(closed_root_layout(rootfd) == budget and inventory(rootfd) == budget,
                    'full final actual regular byte/count tally')
            require(budget['entries'] == (264 if complete_data else len(PUBLICATIONS) + 260),
                    'exact durable leaf count')
    except BaseException as exc:
        fail(STAGE, exc)
        complete_data = False
    # No fallback publication on exceptions. Missing/partial packs remain missing/partial.
    if rootfd is not None:
        try:
            closing, rootfd = rootfd, None
            with until(REMOTE_END):
                os.close(closing)
        except BaseException as exc:
            fail('root-close', exc)
            complete_data = False
    try:
        require(CONFIG is not None and console is not None, 'no admitted console')
        emit_once(receipt_ref, complete_data and FIRST_ERROR is None)
        remaining(REMOTE_END)
    except BaseException:
        return 1
    return 0 if complete_data and FIRST_ERROR is None else 1
