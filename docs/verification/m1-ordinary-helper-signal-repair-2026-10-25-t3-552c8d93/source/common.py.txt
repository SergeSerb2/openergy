class DeadlineError(RuntimeError):
    pass


def remaining(deadline):
    value = deadline - time.monotonic()
    if value <= 0:
        raise DeadlineError('absolute deadline expired')
    return value


@contextlib.contextmanager
def io_until(deadline):
    """Main-thread, nestable absolute deadline, including file and pipe I/O."""
    duration = remaining(deadline)
    previous_handler = signal.getsignal(signal.SIGALRM)
    previous_timer = signal.getitimer(signal.ITIMER_REAL)
    entered = time.monotonic()
    def expired(_sig, _frame):
        raise DeadlineError('I/O absolute deadline expired')
    signal.signal(signal.SIGALRM, expired)
    if previous_timer[0]:
        duration = min(duration, previous_timer[0])
    signal.setitimer(signal.ITIMER_REAL, duration)
    try:
        yield
        remaining(deadline)
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
        signal.signal(signal.SIGALRM, previous_handler)
        if previous_timer[0]:
            signal.setitimer(signal.ITIMER_REAL,
                            max(0.000001, previous_timer[0] - (time.monotonic() - entered)),
                            previous_timer[1])


def read(path, deadline, cap=LOG_CAP):
    with io_until(deadline):
        with Path(path).open('rb') as stream:
            value = stream.read(cap + 1)
        if len(value) > cap:
            raise RuntimeError('file byte cap exceeded: ' + str(path))
        return value


def load(path, deadline):
    return json.loads(read(path, deadline))


def exists(path, deadline):
    with io_until(deadline):
        return Path(path).exists()


def initialize_bytes(deadline):
    """Exclusive, before preparation. Never reconstruct a missing/corrupt ledger."""
    with io_until(deadline):
        initial = sum(p.stat().st_size for p in HERE.rglob('*') if p.is_file())
        if any(p.stat().st_size > LOG_CAP for p in HERE.rglob('*') if p.is_file()):
            raise RuntimeError('initial per-file cap exceeded')
        value = dict(charged=initial, regular=initial,
                     product_reserved=PRODUCT_RESERVE, cleanup_reserved=CLEANUP_RESERVE,
                     infrastructure_reserved=ACCOUNTING_RESERVE)
        if initial > EVIDENCE_CAP - PRODUCT_RESERVE - CLEANUP_RESERVE - ACCOUNTING_RESERVE:
            raise RuntimeError('initial source/evidence leaves no reserved capacity')
        with (HERE / 'bytes.json').open('xb') as stream:
            stream.write((json.dumps(value, sort_keys=True) + '\n').encode())
            stream.flush()
            os.fsync(stream.fileno())
        (HERE / 'bytes.lock').touch(exist_ok=False)


@contextlib.contextmanager
def byte_lock(deadline):
    with io_until(deadline):
        # r+b refuses to recreate a lost lock or ledger after initialization.
        with (HERE / 'bytes.lock').open('r+b') as lock:
            while True:
                try:
                    fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
                    break
                except BlockingIOError:
                    pause(deadline)
            yield


def charge(size, deadline, critical):
    """Called under byte_lock, BEFORE any payload growth. Charges never refunded.

    The one bounded ledger temporary and observation counters use the fixed
    64 KiB infrastructure reservation. A partial/failed ledger is fail-closed.
    """
    value = load(HERE / 'bytes.json', deadline)
    if (set(value) != {'charged', 'regular', 'product_reserved', 'cleanup_reserved',
                       'infrastructure_reserved'} or
            value['product_reserved'] != PRODUCT_RESERVE or
            value['cleanup_reserved'] != CLEANUP_RESERVE or
            value['infrastructure_reserved'] != ACCOUNTING_RESERVE or
            not 0 <= value['regular'] <= value['charged']):
        raise RuntimeError('invalid byte ledger')
    value['charged'] += size
    if not critical:
        value['regular'] += size
    terminal_hold = 0 if critical == 'terminal' else TERMINAL_RESERVE
    if (value['charged'] > EVIDENCE_CAP - PRODUCT_RESERVE - ACCOUNTING_RESERVE - terminal_hold or
            value['regular'] > EVIDENCE_CAP - PRODUCT_RESERVE - ACCOUNTING_RESERVE - CLEANUP_RESERVE):
        raise RuntimeError('reserved aggregate byte budget exhausted')
    data = (json.dumps(value, sort_keys=True) + '\n').encode()
    if len(data) > 4096:
        raise RuntimeError('byte ledger metadata cap')
    temporary = HERE / 'bytes.next'
    with temporary.open('wb') as stream:
        stream.write(data)
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(temporary, HERE / 'bytes.json')
    directory = os.open(HERE, os.O_RDONLY)
    try:
        os.fsync(directory)
    finally:
        os.close(directory)


def write_bytes(path, data, deadline, *, critical=False, keep_first=False, stream=False):
    path = Path(path)
    if not path.resolve().is_relative_to(HERE):
        raise RuntimeError('evidence path outside owned folder')
    with byte_lock(deadline):
        if not stream and path.exists():
            if keep_first:
                return  # No charge, temporary, or rewrite for an existing first receipt.
            raise FileExistsError(str(path))
        size = path.stat().st_size if path.exists() else 0
        if len(data) + (size if stream else 0) > LOG_CAP:
            raise RuntimeError('evidence per-file byte cap exceeded')
        # Double exclusive records conservatively cover temporary + final names.
        charge(len(data) * (1 if stream else 2), deadline, critical)
        if stream:
            with path.open('ab', buffering=0) as output:
                if output.write(data) != len(data):
                    raise RuntimeError('short evidence stream write')
            return
        temporary = path.with_name(path.name + '.writing-' + str(os.getpid()))
        try:
            with temporary.open('xb') as output:
                if output.write(data) != len(data):
                    raise RuntimeError('short evidence record write')
                output.flush()
            os.link(temporary, path)
        finally:
            if temporary.exists():
                temporary.unlink()


def encoded(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':')) + '\n').encode()


def save(path, value, deadline, *, critical=False):
    write_bytes(path, encoded(value), deadline, critical=critical)


def first(path, value, deadline, *, critical=True):
    write_bytes(path, encoded(value), deadline, critical=critical, keep_first=True)


def append(path, value, deadline, *, critical=False):
    write_bytes(path, encoded(value), deadline, critical=critical, stream=True)


class Evidence:
    """Publication errors fail acceptance; they never authorize/disable an action."""
    def __init__(self):
        self.count, self.errors = 0, []

    def attempt(self, operation, *args, **kwargs):
        try:
            return operation(*args, **kwargs)
        except Exception as error:
            self.count += 1
            if len(self.errors) < 32:
                self.errors.append(dict(operation=operation.__name__, error=repr(error)[:2048]))
            return None


def archive_reserved(source, destination, deadline):
    """Move a pre-reserved product log/report without a second payload copy.

    Both paths must be on one filesystem (checked before launch). No EXDEV
    copy fallback. The permanent reservation remains charged after rename.
    """
    with byte_lock(deadline):
        source, destination = Path(source), Path(destination)
        if destination.exists() or source.is_symlink() or not source.is_file():
            raise RuntimeError('invalid exclusive reserved archive')
        if source.stat().st_size > PRODUCT_FILE_CAP:
            raise RuntimeError('reserved product file cap exceeded')
        os.rename(source, destination)


def sha(path, deadline):
    digest = hashlib.sha256()
    with io_until(deadline):
        with Path(path).open('rb') as stream:
            while chunk := stream.read(262144):
                remaining(deadline)
                digest.update(chunk)
    return digest.hexdigest()


def tree(root, deadline):
    records = []
    with io_until(deadline):
        root = Path(root).resolve()
        for path in sorted(root.rglob('*')):
            remaining(deadline)
            item = dict(path=str(path.relative_to(root)), mode=oct(path.lstat().st_mode & 0o7777))
            if path.is_symlink():
                if not path.resolve().is_relative_to(root):
                    raise RuntimeError('bundle symlink escape')
                item['link'] = os.readlink(path)
            elif path.is_file():
                item.update(bytes=path.stat().st_size, sha256=sha(path, deadline))
            elif path.is_dir():
                item['directory'] = True
            else:
                raise RuntimeError('unexpected bundle node')
            records.append(item)
    return records


def command(argv, deadline, cap=4194304, *, coverage=None):
    """Read-only tools/probes only. Products always use exec_owned.py + a Popen ledger.

    The parent exclusively reaps this retained direct child. No other thread,
    waitpid caller or SIGCHLD discard may reap it; an unreaped PID cannot be reused.
    """
    first_error, recorder, process = None, None, None
    try:
        with io_until(deadline):
            try:
                argv = list(map(str, argv))
                if coverage is not None:
                    from coverage import CommandCoverage
                    recorder = CommandCoverage(*coverage, argv, deadline)
                process = subprocess.Popen(argv, cwd=REPO,
                                           stdout=subprocess.PIPE, stderr=subprocess.PIPE)
                try:
                    if recorder is None:
                        stdout, stderr = process.communicate(timeout=remaining(deadline) * 0.8)
                    else:
                        communication_end = time.monotonic() + remaining(deadline) * 0.8
                        recorder.deadline = communication_end
                        recorder.spawn(process)
                        while True:
                            recorder.sample(process)  # Before the first blocking wait; no command retry.
                            try:
                                stdout, stderr = process.communicate(timeout=min(0.02, remaining(communication_end)))
                                break
                            except subprocess.TimeoutExpired:
                                remaining(communication_end)
                except BaseException as error:
                    first_error = error  # Before cleanup or coverage sinks.
                    try:
                        # poll retains the actual signed code on this same Popen
                        # before the immediate same-process clock sample.
                        code = process.poll()
                        sampled = time.monotonic()
                        if code is None and sampled < deadline:
                            os.kill(process.pid, signal.SIGKILL)
                        if sampled >= deadline:
                            raise DeadlineError('helper eligibility poll reached absolute deadline')
                        # Syscall return proves no delivery, exit or reap. Keep
                        # the original remaining helper budget; never renew it.
                        process.communicate(timeout=remaining(deadline))
                    except BaseException as cleanup_error:
                        if len(COVERAGE_ERRORS) < 32:
                            COVERAGE_ERRORS.append(('helper cleanup: ' + repr(cleanup_error))[:2048])
                    raise
                # Latch command/cap failure before recorder.finish can fail.
                if len(stdout) + len(stderr) > cap:
                    raise RuntimeError('tool output byte cap exceeded')
                if process.returncode != 0:
                    raise RuntimeError(repr(dict(argv=list(map(str, argv)), code=process.returncode,
                                                 stdout=stdout.decode(errors='replace'),
                                                 stderr=stderr.decode(errors='replace'))))
            except BaseException as error:
                if first_error is None:
                    first_error = error
                raise
            finally:
                if recorder is not None and process is not None:
                    try:
                        recorder.deadline = deadline
                        recorder.finish(process)  # Actual signed status; None is never reaped.
                    except BaseException as recorder_error:
                        if first_error is None:
                            first_error = recorder_error
                            raise
                        if len(COVERAGE_ERRORS) < 32:
                            COVERAGE_ERRORS.append(('helper coverage finish: ' + repr(recorder_error))[:2048])
            return stdout
    except BaseException:
        # io_until restoration/clock errors cannot replace an earlier failure.
        if first_error is not None:
            raise first_error
        raise


def git_file(path, deadline):
    return command(['git', 'show', ORIGINAL + ':' + ORIGINAL_DIR + path], deadline)


def nice15():
    os.setpriority(os.PRIO_PROCESS, 0, 15)
    if os.getpriority(os.PRIO_PROCESS, 0) != 15:
        raise RuntimeError('absolute NI15 assignment failed')
    os.umask(0o077)


def pause(deadline, seconds=0.02):
    time.sleep(min(seconds, remaining(deadline)))
