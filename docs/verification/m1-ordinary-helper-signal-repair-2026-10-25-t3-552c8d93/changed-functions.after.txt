# common.command
def command(argv, deadline, cap=4194304, *, coverage=None):
    """Read-only tools/probes only. Products always use exec_owned.py + a Popen ledger.

    The parent exclusively reaps this retained direct child. No other thread,
    waitpid caller or SIGCHLD discard may reap it; an unreaped PID cannot be reused.
    """
    first_error, recorder, process = None, None, None
    try:
        with io_until(deadline):
            try:
                argv = list(map(str, argv))
                if coverage is not None:
                    from coverage import CommandCoverage
                    recorder = CommandCoverage(*coverage, argv, deadline)
                process = subprocess.Popen(argv, cwd=REPO,
                                           stdout=subprocess.PIPE, stderr=subprocess.PIPE)
                try:
                    if recorder is None:
                        stdout, stderr = process.communicate(timeout=remaining(deadline) * 0.8)
                    else:
                        communication_end = time.monotonic() + remaining(deadline) * 0.8
                        recorder.deadline = communication_end
                        recorder.spawn(process)
                        while True:
                            recorder.sample(process)  # Before the first blocking wait; no command retry.
                            try:
                                stdout, stderr = process.communicate(timeout=min(0.02, remaining(communication_end)))
                                break
                            except subprocess.TimeoutExpired:
                                remaining(communication_end)
                except BaseException as error:
                    first_error = error  # Before cleanup or coverage sinks.
                    try:
                        # poll retains the actual signed code on this same Popen
                        # before the immediate same-process clock sample.
                        code = process.poll()
                        sampled = time.monotonic()
                        if code is None and sampled < deadline:
                            os.kill(process.pid, signal.SIGKILL)
                        if sampled >= deadline:
                            raise DeadlineError('helper eligibility poll reached absolute deadline')
                        # Syscall return proves no delivery, exit or reap. Keep
                        # the original remaining helper budget; never renew it.
                        process.communicate(timeout=remaining(deadline))
                    except BaseException as cleanup_error:
                        if len(COVERAGE_ERRORS) < 32:
                            COVERAGE_ERRORS.append(('helper cleanup: ' + repr(cleanup_error))[:2048])
                    raise
                # Latch command/cap failure before recorder.finish can fail.
                if len(stdout) + len(stderr) > cap:
                    raise RuntimeError('tool output byte cap exceeded')
                if process.returncode != 0:
                    raise RuntimeError(repr(dict(argv=list(map(str, argv)), code=process.returncode,
                                                 stdout=stdout.decode(errors='replace'),
                                                 stderr=stderr.decode(errors='replace'))))
            except BaseException as error:
                if first_error is None:
                    first_error = error
                raise
            finally:
                if recorder is not None and process is not None:
                    try:
                        recorder.deadline = deadline
                        recorder.finish(process)  # Actual signed status; None is never reaped.
                    except BaseException as recorder_error:
                        if first_error is None:
                            first_error = recorder_error
                            raise
                        if len(COVERAGE_ERRORS) < 32:
                            COVERAGE_ERRORS.append(('helper coverage finish: ' + repr(recorder_error))[:2048])
            return stdout
    except BaseException:
        # io_until restoration/clock errors cannot replace an earlier failure.
        if first_error is not None:
            raise first_error
        raise


# port_diagnostic.query
def query(port, deadline, retained_handles, result):
    """One fixed direct helper, with the existing owned kill/reap timeout policy.

    The caller supplies a SUBSET of its existing port deadline. No product or
    group signals, native ABI, shell, lsof, second command, sleep, or retry.
    An exceptional unreaped handle stays in the caller's diagnostic-only map;
    it is never inserted into product intent/exit or ownership ledgers.
    """
    started = time.monotonic()
    result.update(argv=list(ARGV), filter_port=port, cwd=str(REPO),
                  locale=dict(LANG='C', LC_ALL='C'), started_monotonic=started,
                  absolute_deadline=deadline, attempted=False, pid=None, returncode=None,
                  reaped=False, timed_out=False, output_cap=OUTPUT_CAP,
                  stdout_bytes_read=0, stderr_bytes_read=0, stdout_eof=False,
                  stderr_eof=False, output_cap_reached=False, output_truncated=True,
                  kill_requested=False, kill_returned=False,
                  kill_eligibility_observed=False, kill_eligibility_exit=None,
                  kill_eligibility_monotonic=None, errors=[],
                  status='INCOMPLETE', cause='UNKNOWN', snapshot_is_atomic_with_bind=False)
    if port not in PORTS or deadline - started < MIN_QUERY_SECONDS:
        result.update(skip_reason='undeclared port or insufficient existing budget',
                      ended_monotonic=time.monotonic())
        return result
    process, stdout, stderr = None, bytearray(), bytearray()
    # Keep the same 80/20 read-versus-owned-kill/reap partition as common.command.
    read_deadline = started + (deadline - started) * 0.8
    result['read_deadline'] = read_deadline
    try:
        with io_until(read_deadline):
            result['attempted'] = True
            process = subprocess.Popen(list(ARGV), cwd=REPO,
                env=dict(os.environ, LANG='C', LC_ALL='C'), stdin=subprocess.DEVNULL,
                stdout=subprocess.PIPE, stderr=subprocess.PIPE)
            retained_handles[port] = process  # Before all later I/O and formatting.
            result.update(pid=process.pid, spawned_monotonic=time.monotonic())
            streams = {process.stdout: ('stdout', stdout), process.stderr: ('stderr', stderr)}
            for stream in streams:
                os.set_blocking(stream.fileno(), False)
            while streams:
                ready, _, _ = select.select(list(streams), [], [], remaining(read_deadline))
                for stream in ready:
                    name, output = streams[stream]
                    capacity = OUTPUT_CAP - len(stdout) - len(stderr)
                    if capacity <= 0:
                        result['output_cap_reached'] = result['output_truncated'] = True
                        raise RuntimeError('diagnostic output cap reached; no further read')
                    try:
                        chunk = os.read(stream.fileno(), min(16384, capacity))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        result[name + '_eof'] = True
                        del streams[stream]
                    else:
                        output.extend(chunk)
                        result[name + '_bytes_read'] = len(output)
                        result['last_read_monotonic'] = time.monotonic()
            process.wait(timeout=remaining(read_deadline))
    except (DeadlineError, subprocess.TimeoutExpired) as error:
        result['timed_out'] = True
        result['errors'].append(dict(stage='query', type=type(error).__name__, error=str(error)[:1024]))
    except BaseException as error:
        result['errors'].append(dict(stage='query', type=type(error).__name__, error=str(error)[:1024]))
    finally:
        if process is not None:
            try:
                with io_until(deadline):
                    # The parent exclusively reaps this retained direct child:
                    # no other thread/waitpid or SIGCHLD discard may reap it.
                    # Its PID cannot be reused while it remains unreaped.
                    code = process.poll()
                    result['kill_eligibility_exit'] = code
                    result['kill_eligibility_observed'] = True
                    if code is not None:
                        result.update(returncode=code, reaped=True)
                    sampled = time.monotonic()
                    result['kill_eligibility_monotonic'] = sampled
                    if code is None and sampled < deadline:
                        result['kill_requested'] = True
                        result['kill_requested_monotonic'] = sampled
                        os.kill(process.pid, signal.SIGKILL)
                        result['kill_returned'] = True  # Syscall return only; no delivery/reap proof.
                    if sampled >= deadline:
                        raise DeadlineError('helper eligibility poll reached absolute deadline')
                    process.wait(timeout=remaining(deadline))
                    result.update(returncode=process.returncode, reaped=True,
                                  reaped_monotonic=time.monotonic())
                    del retained_handles[port]
            except BaseException as error:
                if isinstance(error, (DeadlineError, subprocess.TimeoutExpired)):
                    result['timed_out'] = True
                result['errors'].append(dict(stage='owned-helper-reap', type=type(error).__name__,
                                             error=str(error)[:1024]))
            finally:
                if process.returncode is not None:
                    # poll/wait may have reaped just before an I/O alarm. Retain
                    # that actual result even when the surrounding scope fails.
                    result.update(returncode=process.returncode, reaped=True)
                    retained_handles.pop(port, None)
                for stream in (process.stdout, process.stderr):
                    if stream is not None:
                        try:
                            stream.close()
                        except BaseException as error:
                            result['errors'].append(dict(stage='pipe-close', type=type(error).__name__))
        result.update(stdout_bytes_read=len(stdout), stderr_bytes_read=len(stderr),
                      ended_monotonic=time.monotonic())
    # Never claim total emitted bytes when EOF was not observed. Discard all
    # unrelated TCP rows; there is no raw host stdout file or process inventory.
    result['output_truncated'] = (result['output_cap_reached'] or
                                 not (result['stdout_eof'] and result['stderr_eof']))
    result['stderr'] = bytes(stderr[:STDERR_CAP]).decode('utf-8', errors='replace')
    result['stderr_retention_truncated'] = len(stderr) > STDERR_CAP
    result['socket_rows'] = socket_rows(bytes(stdout), port)
    complete = (result['reaped'] and result['returncode'] == 0 and not result['timed_out'] and
                not result['errors'] and not result['output_truncated'] and
                not result['stderr_retention_truncated'] and result['socket_rows']['parse_complete'])
    result['status'] = 'COLLECTED_CONTEXT_ONLY' if complete else 'INCOMPLETE'
    return result
