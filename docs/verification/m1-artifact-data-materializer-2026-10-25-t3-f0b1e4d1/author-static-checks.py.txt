def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def pairs(rows):
    result = {}
    for key, value in rows:
        if key in result:
            raise ValueError('duplicate JSON key')
        result[key] = value
    return result

def strict(raw):

def literal(node):
    # Decode literal DATA nodes only. No eval, compile, import or AST execution.
    if isinstance(node, ast.Constant):
        return node.value
    if isinstance(node, (ast.List, ast.Tuple)):
        values = [literal(n) for n in node.elts]
        return tuple(values) if isinstance(node, ast.Tuple) else values
    if isinstance(node, ast.Dict):
        return dict(zip([literal(n) for n in node.keys], [literal(n) for n in node.values]))
    raise ValueError(type(node).__name__)

def assignment(tree, name):
    nodes = [n for n in tree.body if isinstance(n, ast.Assign)
             and any(isinstance(t, ast.Name) and t.id == name for t in n.targets)]
    require(len(nodes) == 1, 'one assignment: ' + name)
    return nodes[0].value

def inventory(raw, tree):
    lines = raw.splitlines(keepends=True)
    rows = {}
    def walk(node, prefix):
        for child in ast.iter_child_nodes(node):
            if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                name = prefix + child.name
                data = b''.join(lines[child.lineno - 1:child.end_lineno])
                rows[name] = dict(first_line=child.lineno, last_line=child.end_lineno,
                                  bytes=len(data), sha256=sha(data), raw=data)
                walk(child, name + '.')
            else:
                walk(child, prefix)
    walk(tree, '')
    return rows

# Width arithmetic only. These objects contain symbolic widths, not mock data,
# a bound config, a rendered future payload or runtime input.
class Width:
    def __init__(self, canonical_width, token_width=None):
        self.c = canonical_width
        self.t = canonical_width if token_width is None else token_width

def size(value, token=False):
    if isinstance(value, Width):
        return value.t if token else value.c
    if isinstance(value, dict):
        return 2 + max(0,len(value)-1) + sum(size(k,token)+1+size(v,token) for k,v in value.items())
    if isinstance(value, (tuple,list)):
        return 2 + max(0,len(value)-1) + sum(size(v,token) for v in value)
    raw = json.dumps(value, ensure_ascii=True, separators=(',',':'), allow_nan=False)
    return len(json.dumps(raw,ensure_ascii=True))-2 if token else len(raw)
