Future coordinator: trusted local macOS CPython >=3.10, isolated startup.
Transport facts only. Parent observer and payload receipts remain mandatory.
"""
import time

ENTRY_NS = time.monotonic_ns()

import hashlib
import json
import os
import re
import selectors
import stat
import subprocess
import sys

SECOND = 1_000_000_000
OPTIONS = [
    "-o", "BatchMode=yes", "-o", "ConnectTimeout=8",
    "-o", "ConnectionAttempts=1", "-o", "StrictHostKeyChecking=yes",
    "-o", "ControlMaster=no", "-o", "ControlPath=none",
    "-o", "ControlPersist=no", "-o", "ServerAliveInterval=5",
    "-o", "ServerAliveCountMax=3", "-p", "22",
]
PRIVATE_OPTIONS = ("HostName", "HostKeyAlias", "UserKnownHostsFile")
SOURCE_GUARD_SHA256 = "9e0b9a5afb47b7755d1ebaf5f50edf45cf0a7372baaa2f83b83d9255011aed4c"
REMOTE = "/usr/bin/python3 -I -S -B -"
FLAGS = (
    "--phase", "--profile", "--profile-sha256", "--payload",
    "--payload-sha256", "--run-dir", "--collector-sha256",
    "--envelope-start-ns", "--clock-sha256",
)
CAPS = {"stdout": 131072, "stderr": 65536}
COMBINED_CAP = 196608
INPUT_CAPS = {"profile": 131072, "payload": 1048576, "collector": 131072}
ENVIRONMENT = {"PATH": "/usr/bin:/bin", "LC_ALL": "C", "LANG": "C"}


class Rejected(Exception):
    pass


def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(",", ":"),
                       ensure_ascii=True, allow_nan=False) + "\n").encode("ascii")


def digest(data):
    return hashlib.sha256(data).hexdigest()


def clock_descriptor():
    info = time.get_clock_info("monotonic")
    if (sys.platform != "darwin" or sys.implementation.name != "cpython"
            or sys.version_info[:2] < (3, 10)
            or not info.monotonic or info.adjustable
            or info.implementation != "mach_absolute_time()"):
        raise Rejected("unqualified_coordinator")
    if not (sys.flags.isolated and sys.flags.no_site
            and sys.flags.dont_write_bytecode and not sys.flags.optimize):
        raise Rejected("startup_flags")
    return {
        "schema": "openergy.local-macos-clock.v1", "clock": "time.monotonic_ns",
        "implementation": info.implementation, "monotonic": info.monotonic,
        "adjustable": info.adjustable, "resolution_seconds": info.resolution,
        "unit": "integer_nanoseconds", "origin": "macOS Mach absolute clock",
        "scope": "same running local Mac only; CPython >=3.10",
        "cross_host_comparison": False,
    }


def arguments(tokens):
    if len(tokens) != 2 * len(FLAGS):
        raise Rejected("closed_cli")
    result = {}
    for index in range(0, len(tokens), 2):
        key, value = tokens[index:index + 2]
        if key not in FLAGS or key in result or not value:
            raise Rejected("closed_cli")
        result[key] = value
    if set(result) != set(FLAGS):
        raise Rejected("closed_cli")
    if result["--phase"] not in ("metadata-source", "metadata-copy"):
        raise Rejected("phase")
    for key in ("--profile-sha256", "--payload-sha256", "--collector-sha256",
                "--clock-sha256"):
        if re.fullmatch(r"[0-9a-f]{64}", result[key]) is None:
            raise Rejected("digest_shape")
    if re.fullmatch(r"[0-9]{1,20}", result["--envelope-start-ns"]) is None:
        raise Rejected("clock_start_shape")
    return result


def directory(path):
    """Open every absolute directory component without following links."""
    if (not path.startswith("/") or "\x00" in path
            or (path != "/" and any(x in ("", ".", "..")
                                     for x in path[1:].split("/")))):
        raise Rejected("absolute_path")
    fd = os.open("/", os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
    try:
        for part in path[1:].split("/") if path != "/" else ():
            next_fd = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW
                              | os.O_NONBLOCK, dir_fd=fd)
            os.close(fd)
            fd = next_fd
        if os.fstat(fd).st_uid != os.getuid():
            raise Rejected("directory_owner")
        return fd
    except BaseException:
        os.close(fd)
        raise


def leaf(path):
    parent, name = os.path.split(path)
    if not name or name in (".", ".."):
        raise Rejected("leaf_path")
    return directory(parent), name


def identity(value):
    return (value.st_dev, value.st_ino, value.st_mode, value.st_uid,
            value.st_nlink, value.st_size, value.st_mtime_ns, value.st_ctime_ns)


def read_bound(path, expected, limit, before_ns):
    parent, name = leaf(path)
    fd = None
    try:
        before_ns()
        fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK,
                     dir_fd=parent)
        initial = os.fstat(fd)
        if (not stat.S_ISREG(initial.st_mode) or initial.st_uid != os.getuid()
                or stat.S_IMODE(initial.st_mode) not in (0o400, 0o600)
                or initial.st_nlink != 1 or initial.st_size > limit):
            raise Rejected("input_type_owner_mode_size")
        data = bytearray()
        while True:
            before_ns()
            chunk = os.read(fd, min(65536, limit - len(data) + 1))
            before_ns()
            if not chunk:
                break
            if len(data) + len(chunk) > limit:
                raise Rejected("input_read_cap")
            data.extend(chunk)
        linked = os.stat(name, dir_fd=parent, follow_symlinks=False)
        if (identity(initial) != identity(os.fstat(fd))
                or identity(initial) != identity(linked)
                or len(data) != initial.st_size or digest(data) != expected):
            raise Rejected("input_binding_changed")
        return bytes(data)
    finally:
        if fd is not None:
            os.close(fd)
        os.close(parent)


def unique_object(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise Rejected("duplicate_json_key")
        result[key] = value
    return result


def profile_vector(data):
    profile = json.loads(data.decode("utf-8"), object_pairs_hook=unique_object)
    if (not isinstance(profile, dict)
            or set(profile) != {"schema", "source_guard", "argv"}
            or profile["schema"] != "openergy.m1-bound-ssh-profile.v1"):
        raise Rejected("profile_shape")
    reference = profile["source_guard"]
    if (not isinstance(reference, dict) or set(reference) != {"path", "sha256"}
            or not isinstance(reference["path"], str)
            or not reference["path"].startswith("/")
            or "\x00" in reference["path"]
            or any(part in ("", ".", "..")
                   for part in reference["path"][1:].split("/"))
            or reference["sha256"] != SOURCE_GUARD_SHA256):
        raise Rejected("source_guard_reference")
    argv = profile["argv"]
    if (not isinstance(argv, list) or len(argv) != 29
            or not all(isinstance(token, str) for token in argv)
            or argv[0] != "/usr/bin/ssh"
            or argv[1:21] != OPTIONS or argv[21:27:2] != ["-o"] * 3
            or argv[-1] != REMOTE
            or re.fullmatch(r"[A-Za-z0-9_][A-Za-z0-9_.@:-]{0,254}",
                            argv[-2]) is None):
        raise Rejected("closed_ssh_vector")
    for index, name in enumerate(PRIVATE_OPTIONS):
        token = argv[22 + 2 * index]
        prefix = name + "="
        if not token.startswith(prefix):
            raise Rejected("private_option_slot")
        value = token[len(prefix):]
        if name == "UserKnownHostsFile":
            if (re.fullmatch(r"/[A-Za-z0-9_./:@+-]+", value) is None
                    or any(part in ("", ".", "..")
                           for part in value[1:].split("/"))):
                raise Rejected("known_hosts_path_shape")
        elif re.fullmatch(r"[A-Za-z0-9_\[\]:][A-Za-z0-9_.:\[\]-]{0,254}",
                          value) is None:
            raise Rejected("host_option_shape")
    # Syntax is not provenance. Parent must compare the complete argv with the
    # exact saved 1563-byte guard before binding the whole profile SHA-256.
    # Do not open the private guard or infer effective inherited SSH config.
    return argv


class Collector:
    def __init__(self, args, clock):
        self.args = args
        self.start = int(args["--envelope-start-ns"])
        self.work = self.start + 130 * SECOND
        self.settle_end = self.start + 132 * SECOND
        self.absolute = self.start + 135 * SECOND
        self.clock = clock
        self.first_failure = None
        self.publication_failures = []
        self.publication_failure_count = 0
        self.process = None
        self.actual_exit = None
        self.exit_observed_ns = None
        self.kill = {"attempted": False, "returned": False, "observed_ns": None,
                     "policy": "exact retained local SSH only; cap or deadline"}
        self.root = None
        self.outputs = {}
        self.inputs_verified = {name: False for name in INPUT_CAPS}
        self.input_bytes = {name: None for name in INPUT_CAPS}
        self.streams = {
            name: {"observed_bytes": 0, "written_bytes": 0, "eof": False,
                   "overflow": False, "read_drop": False,
                   "unknown_unread_bytes": None, "fsynced": False,
                   "prefix_sha256": None}
            for name in CAPS
        }
        self.timeout = False
        self.spawn_started_ns = None
        self.spawn_returned_ns = None
        self.guard_durable = False
        self.settle_stop_ns = None
        self.final_stop_ns = None
        self.hashers = {name: hashlib.sha256() for name in CAPS}

    def fail(self, code, stage):
        if self.first_failure is None:
            self.first_failure = {"code": code, "stage": stage,
                                  "observed_ns": time.monotonic_ns()}

    def check(self, end, stage):
        if time.monotonic_ns() >= end:
            self.timeout = True
            self.fail("deadline", stage)
            raise Rejected("deadline")

    def admission_clock(self):
        self.check(self.work, "admission")

    def publication_failure(self, stage, error):
        self.fail("publication_failure", stage)
        self.publication_failure_count += 1
        if len(self.publication_failures) < 16:
            self.publication_failures.append({"stage": stage,
                "exception_type": type(error).__name__,
                "observed_ns": time.monotonic_ns()})

    def create(self, name, mode=0o600):
        return os.open(name, os.O_RDWR | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW
                       | os.O_NONBLOCK, mode, dir_fd=self.root)

    def write(self, fd, data, end, stage, hasher=None, stream=None):
        offset = 0
        while offset < len(data):
            self.check(end, stage)
            amount = os.write(fd, memoryview(data)[offset:])
            if amount <= 0:
                raise Rejected("short_write")
            if hasher is not None:
                hasher.update(memoryview(data)[offset:offset + amount])
                stream["written_bytes"] += amount
            offset += amount
            self.check(end, stage)

    def put(self, name, data, cap, end):
        if len(data) > cap:
            raise Rejected("publication_cap")
        self.check(end, name)
        fd = self.create(name)
        try:
            self.write(fd, data, end, name)
            os.fsync(fd)
            self.check(end, name)
        finally:
            os.close(fd)
        os.fsync(self.root)
        self.check(end, name)

    def bindings(self):
        return {
            "collector": {"path": os.path.abspath(__file__),
                          "sha256": self.args["--collector-sha256"],
                          "verified": self.inputs_verified["collector"],
                          "bytes": self.input_bytes["collector"]},
            "profile": {"path": self.args["--profile"],
                        "sha256": self.args["--profile-sha256"],
                        "verified": self.inputs_verified["profile"],
                        "bytes": self.input_bytes["profile"]},
            "payload": {"path": self.args["--payload"],
                        "sha256": self.args["--payload-sha256"],
                        "verified": self.inputs_verified["payload"],
                        "bytes": self.input_bytes["payload"]},
        }

    def deadlines(self):
        return {"started_ns": self.start, "entry_ns": ENTRY_NS,
                "work_until_ns": self.work, "settle_until_ns": self.settle_end,
                "absolute_until_ns": self.absolute,
                "work_seconds": 130, "settle_seconds": 2,
                "final_seconds": 3, "total_seconds": 135}

    def observe_exit(self):
        if self.process is not None and self.actual_exit is None:
            code = self.process.poll()
            observed = time.monotonic_ns()  # Immediately after actual poll/wait.
            if code is not None:
                self.actual_exit = code
                self.exit_observed_ns = observed
                if code != 0:
                    self.fail("ssh_exit_nonzero", "poll")

    def capture(self):
        selector = selectors.DefaultSelector()
        pipes = {"stdout": self.process.stdout, "stderr": self.process.stderr}
        try:
            for name, pipe in pipes.items():
                os.set_blocking(pipe.fileno(), False)
                selector.register(pipe, selectors.EVENT_READ, name)
            while self.first_failure is None:
                self.check(self.work, "capture")
                self.observe_exit()
                self.check(self.work, "after_poll")
                if self.first_failure is not None:
                    break
                if not selector.get_map() and self.actual_exit is not None:
                    break
                delay = max(0.0, min(0.05,
                    (self.work - time.monotonic_ns()) / SECOND))
                ready = selector.select(delay)
                self.check(self.work, "capture")
                for key, unused_events in ready:
                    self.check(self.work, "before_read")
                    name = key.data
                    stream = self.streams[name]
                    total = sum(item["observed_bytes"] for item in self.streams.values())
                    remaining = min(CAPS[name] - stream["observed_bytes"],
                                    COMBINED_CAP - total)
                    try:
                        chunk = os.read(key.fileobj.fileno(), min(16384, remaining + 1))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        stream["eof"] = True
                        stream["unknown_unread_bytes"] = 0
                        selector.unregister(key.fileobj)
                        self.check(self.work, "eof_read_complete")
                        continue
                    stream["observed_bytes"] += len(chunk)
                    prefix = chunk[:remaining]
                    if len(chunk) > remaining:
                        stream["overflow"] = True
                        stream["read_drop"] = True
                        self.fail("output_cap", name)  # BEFORE the prefix write.
                    self.check(self.work, "read_complete")
                    try:
                        self.write(self.outputs[name], prefix, self.work, name,
                                   self.hashers[name], stream)
                    except BaseException as error:
                        stream["read_drop"] = True
                        self.publication_failure(name, error)
                        raise
                    if self.first_failure is not None:
                        break
        finally:
            selector.close()
            for name, pipe in pipes.items():
                if not self.streams[name]["eof"]:
                    self.streams[name]["read_drop"] = True
                pipe.close()  # No subsequent drain, including after KILL.

    def settle(self):
        self.settle_stop_ns = min(self.settle_end, time.monotonic_ns() + 2 * SECOND)
        self.observe_exit()
        if time.monotonic_ns() >= self.settle_stop_ns:
            self.timeout = True
            self.fail("settle_deadline", "settle")
        if self.process is None or self.actual_exit is not None:
            return
        if self.first_failure is None:
            self.fail("unreaped_transport", "settle")
        # Only this retained local SSH handle. poll() rechecks before kill().
        eligible = (self.timeout or any(item["overflow"] for item in self.streams.values())
                    or time.monotonic_ns() >= self.work)
        if (eligible and time.monotonic_ns() < self.settle_stop_ns
                and self.process.poll() is None):
            self.kill["attempted"] = True
            self.kill["observed_ns"] = time.monotonic_ns()
            try:
                self.process.kill()
                self.kill["returned"] = True
            except OSError:
                pass
        self.observe_exit()
        remaining = (self.settle_stop_ns - time.monotonic_ns()) / SECOND
        if self.actual_exit is None and remaining > 0:
            try:
                code = self.process.wait(timeout=remaining)
                observed = time.monotonic_ns()  # BEFORE any receipt writes.
                self.actual_exit = code
                self.exit_observed_ns = observed
                if code != 0:
                    self.fail("ssh_exit_nonzero", "wait")
            except subprocess.TimeoutExpired:
                self.timeout = True
                self.fail("ssh_reap_timeout", "settle")
        if time.monotonic_ns() >= self.settle_stop_ns:
            self.timeout = True
            self.fail("settle_deadline", "settle")

    def run(self):
        payload_fd = None
        publication_stage = None
        try:
            if self.start > ENTRY_NS:
                raise Rejected("future_clock_start")
            if digest(canonical(self.clock)) != self.args["--clock-sha256"]:
                raise Rejected("clock_binding")
            self.admission_clock()
            parent, name = leaf(self.args["--run-dir"])
            try:
                os.mkdir(name, 0o700, dir_fd=parent)  # New root; never reset.
                self.root = os.open(name, os.O_RDONLY | os.O_DIRECTORY
                                    | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=parent)
                info = os.fstat(self.root)
                if info.st_uid != os.getuid() or stat.S_IMODE(info.st_mode) != 0o700:
                    raise Rejected("run_directory_owner_mode")
            finally:
                os.close(parent)
            for label, path in (("collector", os.path.abspath(__file__)),
                                ("profile", self.args["--profile"]),
                                ("payload", self.args["--payload"])):
                data = read_bound(path, self.args["--" + label + "-sha256"],
                                  INPUT_CAPS[label], self.admission_clock)
                self.inputs_verified[label] = True
                self.input_bytes[label] = len(data)
                if label == "profile":
                    argv = profile_vector(data)
                elif label == "payload":
                    payload = data
            guard = {"schema": "openergy.metadata-transport-once.v1",
                     "phase": self.args["--phase"], "bindings": self.bindings(),
                     "clock": self.clock, "deadlines": self.deadlines(),
                     "python": {"resolved_executable": os.path.realpath(sys.executable),
                                "version": sys.version,
                                "implementation": sys.implementation.name},
                     "caps": {"streams": CAPS, "combined": COMBINED_CAP,
                              "inputs": INPUT_CAPS},
                     "retries": 0, "remote_state": "UNVERIFIED",
                     "remote_child_exit": None}
            publication_stage = "once_guard"
            self.put("once.guard.json", canonical(guard), 16384, self.work)
            self.guard_durable = True
            # Snapshot the already hash-checked bounded bytes. Retain this FD as
            # stdin so later edits of the original pathname cannot change input.
            payload_fd = self.create("payload.stdin", 0o400)
            publication_stage = "payload_snapshot"
            self.write(payload_fd, payload, self.work, "payload_snapshot")
            os.fsync(payload_fd)
            os.fsync(self.root)
            os.lseek(payload_fd, 0, os.SEEK_SET)  # Retain the exclusive snapshot FD.
            for name in CAPS:
                publication_stage = name + "_create"
                self.outputs[name] = self.create(name + ".prefix.bin")
            publication_stage = None
            self.check(self.work, "before_spawn")
            self.spawn_started_ns = time.monotonic_ns()
            self.process = subprocess.Popen(
                argv, executable="/usr/bin/ssh", stdin=payload_fd,
                stdout=subprocess.PIPE, stderr=subprocess.PIPE, shell=False,
                close_fds=True, cwd="/", env=ENVIRONMENT, bufsize=0,
            )
            self.spawn_returned_ns = time.monotonic_ns()
            self.check(self.work, "spawn_returned")
            self.capture()
        except BaseException as error:
            self.fail(str(error) if isinstance(error, Rejected) else
                      type(error).__name__, "work")
            if publication_stage is not None:
                self.publication_failure(publication_stage, error)
        finally:
            # This outer finally also handles a spawn that returns after work130.
            if self.process is not None:
                for name in CAPS:
                    pipe = getattr(self.process, name)
                    if pipe is not None and not pipe.closed:
                        self.streams[name]["read_drop"] = True
                        pipe.close()
            try:
                self.settle()
            except BaseException as error:
                self.fail(type(error).__name__, "settle")
            if payload_fd is not None:
                os.close(payload_fd)
        if self.root is None:
            return 1  # No safe new directory exists in which to publish.
        self.final_stop_ns = min(self.absolute, time.monotonic_ns() + 3 * SECOND)
        try:
            self.check(self.final_stop_ns, "final_start")
            for name, fd in self.outputs.items():
                try:
                    self.check(self.final_stop_ns, name + "_fsync")
                    os.fsync(fd)
                    self.check(self.final_stop_ns, name + "_fsync")
                    self.streams[name]["fsynced"] = True
                except BaseException as error:
                    self.publication_failure(name + "_fsync", error)
                finally:
                    os.close(fd)
                    self.streams[name]["prefix_sha256"] = self.hashers[name].hexdigest()
            complete = (self.first_failure is None and self.actual_exit == 0
                        and self.exit_observed_ns is not None
                        and self.exit_observed_ns < self.work
                        and all(item["eof"] and not item["read_drop"]
                                and not item["overflow"] and item["fsynced"]
                                for item in self.streams.values()))
            if not complete and self.first_failure is None:
                self.fail("incomplete_transport", "final")
            for item in self.streams.values():
                item["read_dropped_bytes"] = item["observed_bytes"] - item["written_bytes"]
            receipt = {
                "schema": "openergy.metadata-transport-receipt.v1",
                "status": "TRANSPORT_COMPLETE" if complete else "FAILED_UNVERIFIED",
                "phase": self.args["--phase"], "bindings": self.bindings(),
                "clock": self.clock, "deadlines": self.deadlines(),
                "python": {"resolved_executable": os.path.realpath(sys.executable),
                           "version": sys.version,
                           "implementation": sys.implementation.name},
                "guard_durable": self.guard_durable,
                "spawn_started_ns": self.spawn_started_ns,
                "spawn_returned_ns": self.spawn_returned_ns,
                "settle_stop_ns": self.settle_stop_ns,
                "final_stop_ns": self.final_stop_ns,
                "actual_local_ssh_pid": self.process.pid if self.process else None,
                "actual_local_ssh_exit": self.actual_exit,
                "actual_exit_observed_ns": self.exit_observed_ns,
                "handle_retained": self.process is not None,
                "kill": self.kill, "timeout": self.timeout,
                "streams": self.streams, "first_observed_failure": self.first_failure,
                "secondary_publication_failures": self.publication_failures,
                "publication_failure_count": self.publication_failure_count,
                "caps": {"streams": CAPS, "combined": COMBINED_CAP,
                         "inputs": INPUT_CAPS, "guard": 16384, "receipt": 32768},
                "retries": 0, "completion_passes": 0,
                "remote_child_exit": None, "remote_current_state": "UNVERIFIED",
                "payload_receipt_assessed": False, "artifact_accepted": False,
                "r9_reserved_publisher": False,
                "own_cli_exit_observed": False,
                "independent_parent_observation_required": True,
                "receipt_prewrite_ns": time.monotonic_ns(),
            }
            self.put("receipt.json", canonical(receipt), 32768, self.final_stop_ns)
            self.check(self.final_stop_ns, "after_receipt")
            return 0 if complete and self.first_failure is None else 1
        except BaseException as error:
            self.publication_failure("final_publication", error)
            # One bounded exclusive side record; no receipt replacement/retry.
            try:
                self.put("publication-failure.json", canonical({
                    "first_observed_failure": self.first_failure,
                    "secondary_publication_failures": self.publication_failures,
                    "publication_failure_count": self.publication_failure_count,
                    "receipt_may_be_partial_or_stale": True,
                }), 16384, self.final_stop_ns)
            except BaseException:
                pass  # Parent observes nonzero/missing facts; never claim durable.
            return 1
        finally:
            os.close(self.root)


def main():
    try:
        os.umask(0o077)
        args = arguments(sys.argv[1:])
        clock = clock_descriptor()
        return Collector(args, clock).run()
    except BaseException:
        return 1  # No raw argv, profile values, payload, or exception text on console.


if __name__ == "__main__":
    raise SystemExit(main())
