def windows_command_units(argv):
    # Complete Windows argv serialization: executable, quotes, spaces, UTF-16 and NUL.
    if any(not isinstance(a,str) or '\0' in a for a in argv):raise ValueError('invalid Windows argv')
    units=len(subprocess.list2cmdline(argv).encode('utf-16le'))//2+1
    if units>32767:raise ValueError('complete Windows command line exceeds 32767 UTF-16 units including NUL')
    return units

def transport_payload(script):
    # Raw DEFLATE, fixed parameters, exact UTF-8 length/hash; no remote script file.
    data=script.encode('utf-8')
    if '\0' in script or not 1<=len(data)<=65536:raise ValueError('transport source size/encoding refused')
    compressor=zlib.compressobj(9,zlib.DEFLATED,-15)
    literal=base64.b64encode(compressor.compress(data)+compressor.flush()).decode('ascii')
    if len(literal)>10400:raise ValueError('compressed transport literal exceeds 10400 ASCII characters')
    wrapper=r'''$ErrorActionPreference='Stop';(Get-Process -Id $PID).PriorityClass='BelowNormal'
$transportStream=$null;$transportInflater=$null;$transportOutput=$null;$transportHash=$null;$transportFirstError=$null
try{

 $transportStream=[IO.MemoryStream]::new($transportBytes,$false)
 $transportInflater=[IO.Compression.DeflateStream]::new($transportStream,[IO.Compression.CompressionMode]::Decompress)
 $transportOutput=[IO.MemoryStream]::new();$transportHash=[Security.Cryptography.SHA256]::Create()

  if($transportOutput.Length+$transportCount -gt __BYTES__){throw 'transport expansion length exceeded'}
  $transportOutput.Write($transportBuffer,0,$transportCount)
 }

 $transportText=[Text.UTF8Encoding]::new($false,$true).GetString($transportPlain)

    return wrapper,base64.b64encode(wrapper.encode('utf-16le')).decode('ascii')

def transport_argv(alias,encoded,binding):
    native=['powershell.exe','-NoLogo','-NoProfile','-NonInteractive','-EncodedCommand',encoded]
    units=windows_command_units(native)
    native_line=subprocess.list2cmdline(native);launch_units=[]
    for layer in binding['literal_command_layers']:
        line=layer['prefix']+native_line+layer['suffix']
        if '\0' in line:raise ValueError('NUL in bound Windows launcher')
        count=len(line.encode('utf-16le'))//2+1
        if count>layer['limit_including_nul']:raise ValueError('complete bound Windows launcher command exceeds its limit')
        launch_units.append(count)
    return ['/usr/bin/ssh',*OPTIONS,alias,*native],units,launch_units

    def audit_retained_root(self):
        # No process is killed and no tree is removed after incomplete collection.
        state64=base64.b64encode(json.dumps(self.state).encode()).decode()
        script=(HERE/'safety.ps1').read_text()+r'''
$ErrorActionPreference='Stop';(Get-Process -Id $PID).PriorityClass='BelowNormal'
$deadline=[Diagnostics.Stopwatch]::GetTimestamp()+30L*[Diagnostics.Stopwatch]::Frequency
$state=[Text.Encoding]::UTF8.GetString([Convert]::FromBase64String('__STATE__'))|ConvertFrom-Json
$record=[ordered]@{root_retained=$null;cleanup_acceptance='UNVERIFIED';historical_preservation='UNVERIFIED';processes_killed=0;root_removed=$null}
try{
 $owned=Fresh-Owned $state $deadline
 $record.root_retained=$true;$record.root_removed=$false
 $items=@(Fresh-Tree $owned $deadline)
 $processes=@(Get-CimInstance Win32_Process -OperationTimeoutSec 1|Where-Object {$_.ProcessId -ne $PID -and (($_.ExecutablePath -and $_.ExecutablePath.StartsWith($owned+'\',[StringComparison]::OrdinalIgnoreCase)) -or ($_.CommandLine -and $_.CommandLine.Contains($owned)))}|Select-Object ProcessId,ParentProcessId,CreationDate,ExecutablePath,CommandLine)
 Fresh-Before $deadline
 $record.remaining_process_references=$processes;$record.items=$items.Count;$record.ownership_audit_completed=$true
}catch{$record.ownership_audit_completed=$false;$record.error=($_|Out-String)}
$record|ConvertTo-Json -Depth 12 -Compress

'''
        rec,out=self.remote('retained-root-audit',script.replace('__STATE__',state64),45)
        self.result['retained_root_audit_exit']=rec['exit_code']
        save(self.private/'retained-root-audit.json',json.loads(out.decode('utf-8-sig')))

    def export(self):
        # Raw records stay private. No automatic raw-to-Git export.
        current=local_prior();save(self.private/'local-prior-after.json',current)
        self.result['local_prior_evidence_equal']=current==self.prior
        self.result['passed']=self.result['passed'] and current==self.prior
        self.result['controller_exit']=0 if self.result['passed'] else 1
        save(self.private/'result.json',self.result)
        files=[]
        for p in sorted(self.private.rglob('*')):
            if p.is_file():
                data=p.read_bytes();files.append({'path':str(p.relative_to(self.private)),
                    'bytes':len(data),'sha256':sha(data)})
        save(self.private/'data-receipts.json',{'files':files,'raw_records_remain_private':True})

    def run(self):
        first_error=None
        try:
            self.prepare_inputs();self.bind_source();self.setup()
            rec,_=self.stage('qualify',600);self.result['qualification_exit']=rec['exit_code']
            if rec['exit_code']==0:
                self.result['compiler_stage_attempts']+=1
                rec,_=self.stage('build',660);self.result['build_exit']=rec['exit_code']
                if rec['exit_code']==0:
                    self.result['runtime_stage_attempts']+=1
                    rec,_=self.stage('runtime',150+30);self.result['runtime_exit']=rec['exit_code']
        except BaseException as error:
            first_error=error;self.result['first_flow_failure']=traceback.format_exc()
        finally:
            # A failed build or its receipt must still reach the one postflight path.
            if self.root is not None:
                try:self.collect()
                except BaseException as error:
                    if first_error is None:first_error=error
                    else:self.result['postflight_collection_failure']=traceback.format_exc()
                    self.result['fresh_root_cleanup']='UNVERIFIED_AFTER_COLLECTION_OR_REMOVAL_FAILURE'
                    self.result['fresh_selected_preservation']='UNVERIFIED'
                    try:self.audit_retained_root()
                    except BaseException:self.result['retained_root_audit_failure']=traceback.format_exc()
        if first_error is not None:raise first_error
        self.result['passed']=all(self.result.get(x)==0 for x in ('qualification_exit','build_exit','runtime_exit','postflight_exit','cleanup_exit')) and self.result['cleanup']['owned_root_removed']
