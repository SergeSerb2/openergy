 public static int RemainingMs(long deadline) {
  long left=deadline-Stopwatch.GetTimestamp();
  if(left<=0)throw new TimeoutException("absolute controller deadline exceeded");
  return (int)Math.Max(1,Math.Min(int.MaxValue,Math.Ceiling(1000.0*left/Stopwatch.Frequency)));
 }

 public sealed class Child : IDisposable {
  public Process Process; public IntPtr NativeProcess, NativeThread; public long BeforeCreateTicks, AfterCreateTicks, ResumeTicks; public bool InJob, Closed; public int NativePid; public bool RecoveryKilled,RecoverySignalled; public uint RecoveryCode;
  public void Resume() { long before=Stopwatch.GetTimestamp();uint previous=ResumeThread(NativeThread);if(previous==uint.MaxValue)throw Error("ResumeThread");ResumeTicks=before;if(previous!=1)throw new InvalidOperationException("ResumeThread expected one suspension");if(!CloseHandle(NativeThread))throw Error("CloseThread");NativeThread=IntPtr.Zero; }
  // Only for a child that was registered while suspended and never resumed.
  public void RecoverUnresumed(long deadline) {
   if(ResumeTicks!=0||NativeProcess==IntPtr.Zero)throw new InvalidOperationException("not an owned unresumed child");
   uint initial=WaitForSingleObject(NativeProcess,0);if(initial==uint.MaxValue)throw Error("owned unresumed wait");
   if(initial!=0) {
    if(!TerminateProcess(NativeProcess,1))throw Error("terminate owned unresumed child");
    RecoveryKilled=true;
   }
   uint wait=WaitForSingleObject(NativeProcess,(uint)Math.Min(10000,RemainingMs(deadline)));
   RecoverySignalled=wait==0;
   if(!RecoverySignalled)throw new TimeoutException("owned unresumed recovery did not signal");
   if(!GetExitCodeProcess(NativeProcess,out RecoveryCode))throw Error("unresumed recovery exit code");
  }
  public void Dispose() {
   if(Closed)return;
   if(NativeThread!=IntPtr.Zero){if(!CloseHandle(NativeThread))throw Error("CloseThread");NativeThread=IntPtr.Zero;}
   if(NativeProcess!=IntPtr.Zero){if(!CloseHandle(NativeProcess))throw Error("CloseProcess");NativeProcess=IntPtr.Zero;}
   if(Process!=null)Process.Dispose();Closed=true;
  }
 }
 public static Child Start(string exe,string[] args,string cwd,string stdout,string stderr,System.Collections.IList owners) {
  bool attributesReady=false;
  var sa=new SA{nLength=Marshal.SizeOf<SA>(),bInheritHandle=1};
  IntPtr input=IntPtr.Zero,output=IntPtr.Zero,error=IntPtr.Zero,attrs=IntPtr.Zero,values=IntPtr.Zero;
  try {
   input=CreateFileW("NUL",0x80000000,3,ref sa,3,0,IntPtr.Zero);
   output=CreateFileW(stdout,0x40000000,3,ref sa,1,0,IntPtr.Zero);
   error=CreateFileW(stderr,0x40000000,3,ref sa,1,0,IntPtr.Zero);
   if(input==new IntPtr(-1)||output==new IntPtr(-1)||error==new IntPtr(-1))throw Error("redirect creation");
   IntPtr size=IntPtr.Zero;InitializeProcThreadAttributeList(IntPtr.Zero,1,0,ref size);
   attrs=Marshal.AllocHGlobal(size);if(!InitializeProcThreadAttributeList(attrs,1,0,ref size))throw Error("attribute initialization");attributesReady=true;

   if(!UpdateProcThreadAttribute(attrs,0,new IntPtr(0x20002),values,new IntPtr(3*IntPtr.Size),IntPtr.Zero,IntPtr.Zero))throw Error("explicit handle list");
   var si=new SIX{info=new SI{cb=Marshal.SizeOf<SIX>(),dwFlags=0x100,hStdInput=input,hStdOutput=output,hStdError=error},attributes=attrs};
   var command=new StringBuilder(Arg(exe));foreach(var a in args)command.Append(" ").Append(Arg(a));
   var child=new Child{BeforeCreateTicks=Stopwatch.GetTimestamp()};PI pi;
   // Suspended + BelowNormal at creation + no console window + explicit inherited handles.
   if(!CreateProcessW(exe,command,IntPtr.Zero,IntPtr.Zero,true,0x00000004|0x00004000|0x08000000|0x00080000,IntPtr.Zero,cwd,ref si,out pi))throw Error("CreateProcess");
   child.AfterCreateTicks=Stopwatch.GetTimestamp();child.NativeProcess=pi.process;child.NativeThread=pi.thread;child.NativePid=pi.pid;
   owners.Add(child); // Register exact native handles before any fallible managed identity work.
   child.Process=Process.GetProcessById(pi.pid);var retained=child.Process.SafeHandle;
   if(!IsProcessInJob(pi.process,IntPtr.Zero,out child.InJob))throw Error("IsProcessInJob");
   return child;
  } finally {
   foreach(var h in new[]{input,output,error})if(h!=IntPtr.Zero&&h!=new IntPtr(-1))CloseHandle(h);
   if(attrs!=IntPtr.Zero){if(attributesReady)DeleteProcThreadAttributeList(attrs);Marshal.FreeHGlobal(attrs);}if(values!=IntPtr.Zero)Marshal.FreeHGlobal(values);
  }
 }

 public sealed class LogMarkerReader : IDisposable {
  public const int MaxBytes=2*1024*1024, ChunkBytes=4096;
  public long Offset; public int MissingOpens; public bool Found;
  readonly string path,marker; string tail=""; FileStream stream;
  public LogMarkerReader(string path,string marker){this.path=path;this.marker=marker;}
  public async Task<bool> PollAsync(long deadline) {
   Before(deadline);
   if(stream==null) {
    try { stream=new FileStream(path,FileMode.Open,FileAccess.Read,FileShare.ReadWrite|FileShare.Delete,1,FileOptions.Asynchronous|FileOptions.SequentialScan); }
    catch(FileNotFoundException) { MissingOpens++;Before(deadline);return false; }
   }
   Before(deadline);
   long length=stream.Length;
   if(!stream.CanSeek||length>MaxBytes||length<Offset||stream.Position!=Offset)throw new IOException("readiness log file/offset bound failed");
   int count=(int)Math.Min(ChunkBytes,length-Offset);
   if(count==0){Before(deadline);return false;}
   var bytes=new byte[count];int read;
   using(var cancel=new CancellationTokenSource(RemainingMs(deadline))) {
    read=await stream.ReadAsync(bytes.AsMemory(0,count),cancel.Token);
   }
   Before(deadline);
   if(read<=0||read>count)throw new IOException("readiness log changed during bounded read");
   Offset+=read;
   if(Offset>MaxBytes||stream.Length>MaxBytes||stream.Length<Offset)throw new IOException("readiness log size bound failed");
   string text=tail+Encoding.UTF8.GetString(bytes,0,read);
   Found=text.Contains(marker,StringComparison.Ordinal);
   tail=text.Substring(Math.Max(0,text.Length-(marker.Length-1)));
   Before(deadline);return Found;
  }
  public void Dispose(){if(stream!=null)stream.Dispose();}
 }
 public sealed class Exchange : IDisposable {
  public NamedPipeServerStream Listener; public Task Work; public string Request,Response; public uint ClientPid,ServerPid;
  public long RequestReadTicks,ForwardAfterTicks,ResponseReadTicks,ClientWriteTicks;
  long forwardBeforeTicks; public long ForwardBeforeTicks { get { return Interlocked.Read(ref forwardBeforeTicks); } }
  readonly CancellationTokenSource cancel; readonly CancellationToken token; readonly long deadline; bool disposed;
  public Exchange(string relay,string daemon,long deadline) {
   this.deadline=deadline;cancel=new CancellationTokenSource(RemainingMs(deadline));token=cancel.Token;
   Listener=new NamedPipeServerStream(relay,PipeDirection.InOut,1,PipeTransmissionMode.Byte,PipeOptions.Asynchronous|PipeOptions.CurrentUserOnly);
   Work=Run(daemon);
  }
  async Task<byte[]> ReadFrame(Stream s) {
   var header=new byte[4];await s.ReadExactlyAsync(header,token);
   int length=(header[0]<<24)|(header[1]<<16)|(header[2]<<8)|header[3];if(length<0||length>16*1024*1024)throw new IOException("invalid frame size");
   var frame=new byte[length+4];header.CopyTo(frame,0);await s.ReadExactlyAsync(frame.AsMemory(4),token);return frame;
  }
  async Task Run(string daemon) {
   await Listener.WaitForConnectionAsync(token);
   if(!GetNamedPipeClientProcessId(Listener.SafePipeHandle,out ClientPid))throw Error("relay client identity");
   using(var target=new NamedPipeClientStream(".",daemon,PipeDirection.InOut,PipeOptions.Asynchronous)) {
    await target.ConnectAsync(RemainingMs(deadline),token);Before(deadline);
    if(!GetNamedPipeServerProcessId(target.SafePipeHandle,out ServerPid))throw Error("daemon pipe identity");
    var request=await ReadFrame(Listener);RequestReadTicks=Stopwatch.GetTimestamp();Request=Encoding.UTF8.GetString(request,4,request.Length-4);
    Interlocked.Exchange(ref forwardBeforeTicks,Stopwatch.GetTimestamp());await target.WriteAsync(request,token);await target.FlushAsync(token);ForwardAfterTicks=Stopwatch.GetTimestamp();
    var response=await ReadFrame(target);ResponseReadTicks=Stopwatch.GetTimestamp();Response=Encoding.UTF8.GetString(response,4,response.Length-4);
    await Listener.WriteAsync(response,token);await Listener.FlushAsync(token);ClientWriteTicks=Stopwatch.GetTimestamp();Before(deadline);
   }
  }
  public void Dispose(){if(disposed)return;disposed=true;cancel.Cancel();Listener.Dispose();cancel.Dispose();}
 }
