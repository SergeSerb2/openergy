# INERT proposed filesystem boundary. No invocation was made during preparation.
function Fresh-Before($deadline){if([Diagnostics.Stopwatch]::GetTimestamp() -ge $deadline){throw 'fresh filesystem deadline reached'}}
function Fresh-Path($path){
 if(-not ($path -is [string]) -or $path -notmatch '^[A-Za-z]:\\' -or $path -match '[\x00-\x1f\x7f''"`$]' -or $path.Substring(2).Contains(':')){throw 'local drive path required; UNC/device/stream/shell path refused'}
 $full=[IO.Path]::GetFullPath($path).TrimEnd('\')
 if($full.Length -le 3 -or $full -cne $path.TrimEnd('\')){throw 'normalized non-root path required'}
 foreach($part in $full.Substring(3).Split('\')){
  if(-not $part -or $part -in @('.','..') -or $part.IndexOfAny([char[]]'<>|?*[]') -ge 0 -or $part.EndsWith('.') -or $part.EndsWith(' ') -or $part -match '^(?i:CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9])(?:\.|$)'){throw 'unsafe Windows path component'}
 }
 return $full
}
function Fresh-NoReparse($path,$deadline){
 $full=Fresh-Path $path;$cursor=$full.Substring(0,3)
 $parts=@('')+@($full.Substring(3).Split('\'))
 foreach($part in $parts){
  Fresh-Before $deadline
  if($part){$cursor=Join-Path $cursor $part}
  $item=Get-Item -LiteralPath $cursor -Force -ErrorAction Stop
  if($item.Attributes -band [IO.FileAttributes]::ReparsePoint){throw 'reparse ancestor refused'}
 }
 return $full
}
function Fresh-Tree($path,$deadline){
 $null=Fresh-NoReparse $path $deadline
 $pending=[Collections.Generic.Stack[string]]::new();$pending.Push($path)
 $items=[Collections.Generic.List[object]]::new()
 while($pending.Count){
  Fresh-Before $deadline
  $item=Get-Item -LiteralPath $pending.Pop() -Force
  if($item.Attributes -band [IO.FileAttributes]::ReparsePoint){throw 'reparse tree entry refused'}
  $items.Add($item)
  if($items.Count -gt 200000){throw 'fresh tree entry ceiling reached'}
  if($item.PSIsContainer){foreach($child in @(Get-ChildItem -LiteralPath $item.FullName -Force)){
   Fresh-Before $deadline
   if($child.Attributes -band [IO.FileAttributes]::ReparsePoint){throw 'reparse child refused before traversal'}
   $pending.Push($child.FullName)
  }}
 }
 return $items.ToArray()
}
function Fresh-Owned($state,$deadline){
 $parent=Fresh-NoReparse $state.original_temp $deadline
 $owned=Fresh-NoReparse $state.root $deadline
 if($state.owned_root_leaf -notmatch '^oe-stop-fresh-[0-9a-f]{32}$' -or (Split-Path $owned -Leaf) -cne $state.owned_root_leaf -or (Split-Path $owned -Parent) -ine $parent){throw 'fresh root containment mismatch'}
 $item=Get-Item -LiteralPath $owned -Force
 if($item.CreationTimeUtc.ToFileTimeUtc() -ne [long]$state.root_created_filetime){throw 'fresh directory generation changed'}
 $sid=[Security.Principal.WindowsIdentity]::GetCurrent().User
 $acl=Get-Acl -LiteralPath $owned;$rules=@($acl.Access)
 if(-not $acl.AreAccessRulesProtected -or $acl.Owner -ne $sid.Translate([Security.Principal.NTAccount]).Value -or $rules.Count -ne 1 -or $rules[0].IdentityReference.Translate([Security.Principal.SecurityIdentifier]).Value -ne $sid.Value -or $rules[0].AccessControlType -ne 'Allow' -or $rules[0].FileSystemRights -ne [Security.AccessControl.FileSystemRights]::FullControl){throw 'fresh root owner/ACL changed'}
 $markerPath=Join-Path $owned 'fresh-owner.json'
 $null=Fresh-NoReparse $markerPath $deadline
 $marker=Get-Content -Raw -LiteralPath $markerPath|ConvertFrom-Json
 if($marker.schema -ne 'openergy.fresh-root-owner.v1' -or $marker.nonce -cne $state.ownership_nonce -or [long]$marker.root_created_filetime -ne [long]$state.root_created_filetime -or $marker.sid -cne $sid.Value){throw 'fresh ownership marker mismatch'}
 return $owned
}
function Fresh-Remove($state,$deadline){
 $owned=Fresh-Owned $state $deadline
 # Observation is a refusal gate, never kill authority. Unknown processes are untouched.
 $processes=@(Get-CimInstance Win32_Process -OperationTimeoutSec 1|Where-Object {$_.ProcessId -ne $PID -and (($_.ExecutablePath -and $_.ExecutablePath.StartsWith($owned+'\',[StringComparison]::OrdinalIgnoreCase)) -or ($_.CommandLine -and $_.CommandLine.Contains($owned)))})
 Fresh-Before $deadline
 if($processes.Count){throw 'process references fresh root; retain root without killing'}
 $items=@(Fresh-Tree $owned $deadline)
 $marker=Join-Path $owned 'fresh-owner.json'
 foreach($item in @($items|Where-Object {$_.FullName -ine $owned -and $_.FullName -ine $marker}|Sort-Object { $_.FullName.Length } -Descending)){
  Fresh-Before $deadline
  $null=Fresh-Owned $state $deadline
  $null=Fresh-NoReparse $item.FullName $deadline
  # No recursive deletion. A changed/nonempty directory fails closed.
  Remove-Item -LiteralPath $item.FullName -Force -ErrorAction Stop
 }
 $null=Fresh-Owned $state $deadline;Fresh-Before $deadline
 Remove-Item -LiteralPath $marker -Force -ErrorAction Stop
 Fresh-Before $deadline
 if((Get-Item -LiteralPath $owned -Force).CreationTimeUtc.ToFileTimeUtc() -ne [long]$state.root_created_filetime){throw 'fresh root generation changed at final removal'}
 Remove-Item -LiteralPath $owned -Force -ErrorAction Stop
 Fresh-Before $deadline
 if(Test-Path -LiteralPath $owned){throw 'fresh root removal unverified'}
 [ordered]@{owned_root_removed=$true;removed_only_fresh_owned_root=$true;process_references_before_removal=0;recursive_delete=$false;generation_and_marker_verified=$true;normal_exit_acceptance=$false}
}
