267: fn client(
268:     mut stream: UnixStream,
269:     tx: SyncSender<IoCommand>,
270:     waker: Arc<mio::Waker>,
271:     bus: Arc<EventBus>,
272:     stop: Arc<AtomicBool>,
273: ) {
274:     #[cfg(unix)]
275:     {
276:         if stream.set_nonblocking(false).is_err() {
277:             return;
278:         }
279:         let _ = stream.set_read_timeout(Some(Duration::from_millis(50)));
280:         let _ = stream.set_write_timeout(Some(Duration::from_secs(1)));
281:     }
282:     let queue = bus.client();
283:     let mut buf = Vec::new();
284:     let mut scratch = [0u8; 8192];
285:     // Empty first requests and incomplete frames must not hold all 64 client
286:     // slots forever. Completed requests may idle indefinitely for subscriptions.
287:     // The deadline is absolute: additional partial bytes do not extend it.
288:     let mut request_deadline = Some(Instant::now() + REQUEST_READ_TIMEOUT);
289:     while !stop.load(Ordering::Relaxed) {
290:         if request_deadline.is_some_and(|deadline| Instant::now() >= deadline) {
291:             return;
292:         }
293:         match stream.read(&mut scratch) {
294:             Ok(0) => return,
295:             Ok(n) => {
296:                 request_deadline.get_or_insert_with(|| Instant::now() + REQUEST_READ_TIMEOUT);
297:                 buf.extend_from_slice(&scratch[..n]);
298:             }
299:             Err(e)
300:                 if matches!(
301:                     e.kind(),
302:                     std::io::ErrorKind::WouldBlock | std::io::ErrorKind::TimedOut | std::io::ErrorKind::Interrupted
303:                 ) => {}
304:             Err(_) => return,
305:         }
306:         while let Some(h) = buf.get(..4) {
307:             let Ok(header) = <[u8; 4]>::try_from(h) else { return };
308:             let n = u32::from_be_bytes(header) as usize;
309:             if n > MAX_FRAME_LEN {
310:                 return;
311:             }
312:             if buf.len() < n + 4 {
313:                 break;
314:             }
315:             let request = serde_json::from_slice::<Request>(&buf[4..n + 4]);
316:             buf.drain(..n + 4);
317:             let Ok(req) = request else { return };
318:             let (reply, answer) = mpsc::channel::<Response>();
319:             if tx
320:                 .try_send(IoCommand::Ipc {
321:                     req,
322:                     reply,
323:                     events: queue.clone(),
324:                 })
325:                 .is_err()
326:             {
327:                 return;
328:             }
329:             let _ = waker.wake();
330:             let response = {
331:                 let started = std::time::Instant::now();
332:                 loop {
333:                     let result = answer.recv_timeout(Duration::from_millis(50));
334:                     if !matches!(result, Err(mpsc::RecvTimeoutError::Timeout))
335:                         || stop.load(Ordering::Relaxed)
336:                         || started.elapsed() >= Duration::from_secs(10)
337:                     {
338:                         break result;
339:                     }
340:                 }
341:             };
342:             let Ok(response) = response else {
343:                 return;
344:             };
345:             let Ok(value) = serde_json::to_value(&response) else {
346:                 return;
347:             };
348:             if let Err(error) = write_frame(&mut stream, &value) {
349:                 if error.kind() != std::io::ErrorKind::InvalidData {
350:                     return;
351:                 }
352:                 let error = Response::err(
353:                     response.id,
354:                     IpcError::new(
355:                         "response_too_large",
356:                         "IPC response exceeds the frame limit; request a smaller result or a summary",
357:                     ),
358:                 );
359:                 let Ok(value) = serde_json::to_value(error) else { return };
360:                 if write_frame(&mut stream, &value).is_err() {
361:                     return;
362:                 }
363:             }
364:             // Start the next frame's read budget after the preceding response;
365:             // time spent handling that request is not charged to its successor.
366:             request_deadline = (!buf.is_empty()).then(|| Instant::now() + REQUEST_READ_TIMEOUT);
367:         }
368:         for ev in queue.drain() {
369:             let Ok(value) = serde_json::to_value(ev) else { return };
370:             if write_frame(&mut stream, &value).is_err() {
371:                 return;
372:             }
373:         }
374:     }
375: }
