"""Review-authored data reader. Never extract, import, or execute supplied code.

Invoke with /usr/bin/python3 -I -S -B. JSON goes to stdout. All inputs are
opened read-only. This is an inspection of real retained data, not a test.
"""
import ast
import collections
import gzip
import hashlib
import io
import json
import os
from pathlib import Path, PurePosixPath
import re
import stat
import struct
import tarfile
import unicodedata

ROOT = Path(__file__).resolve().parents[3]
INPUT = ROOT / 'docs/verification/m1-python-prerequisite-data-2026-10-24-t3-27c54a9f'
PRIVATE = Path('/Users/serge/.local/state/openergy-verification/m1-python-prerequisite-2026-10-24-t3-27c54a9f')
ARCHIVE = 'cpython-3.13.9+20251014-aarch64-apple-darwin-install_only.tar.gz'
EXPECTED = '931db8f735e18700d4eab9ee39dbbd0b4c114d7d039dd2707b2d932ded039698'


def need(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def read_data(path, limit):
    fd = os.open(str(path), os.O_RDONLY | os.O_NOFOLLOW)
    with os.fdopen(fd, 'rb') as stream:
        before = os.fstat(stream.fileno())
        need(stat.S_ISREG(before.st_mode), 'nonregular input: ' + str(path))
        need(before.st_size <= limit, 'input bound: ' + str(path))
        data = stream.read(limit + 1)
        after = os.fstat(stream.fileno())
        need(len(data) == before.st_size <= limit, 'input size changed')
        need((before.st_dev, before.st_ino, before.st_size, before.st_mtime_ns, before.st_ctime_ns)
             == (after.st_dev, after.st_ino, after.st_size, after.st_mtime_ns, after.st_ctime_ns),
             'input changed while read')
    return data, {'bytes': len(data), 'sha256': sha(data),
                  'mode': format(stat.S_IMODE(before.st_mode), '04o'),
                  'uid': before.st_uid, 'nlink': before.st_nlink}


source = {}
for name in ('README.md', 'receipt.json', 'safe-extraction-recipe.py.txt'):
    data, source[name] = read_data(INPUT / name, 1024 * 1024)
receipt = json.loads((INPUT / 'receipt.json').read_bytes())
need(source['receipt.json']['sha256'] == 'ea5aaf52ae45e21dc9b8bcb9b18fcfcf1a2045f459701c3213cbc0065fbdb270', 'receipt binding')
need(source['safe-extraction-recipe.py.txt']['sha256'] == 'bab38f8ed1bf4ab1d15843fc8ad3872c553e6665faf5eb97e432261f51f9b027', 'recipe binding')
recipe_tree = ast.parse((INPUT / 'safe-extraction-recipe.py.txt').read_text(), feature_version=(3, 9))

# Inspect every private path ancestor as file metadata; no process or OS query.
ancestor_metadata = []
for path in reversed((PRIVATE,) + tuple(PRIVATE.parents)):
    st = path.lstat()
    need(stat.S_ISDIR(st.st_mode), 'symlink or nondirectory ancestor: ' + str(path))
    ancestor_metadata.append({'path': str(path), 'mode': format(stat.S_IMODE(st.st_mode), '04o'), 'uid': st.st_uid})
need(ancestor_metadata[-1]['mode'] == '0700', 'private leaf mode')
private_uid = ancestor_metadata[-1]['uid']
private_bindings = {}
private_bytes = {}
for name, expected in receipt['private_file_bindings'].items():
    data, observed = read_data(PRIVATE / name, expected['bytes'])
    need(all(observed[key] == expected[key] for key in ('bytes', 'sha256', 'mode')), 'private binding: ' + name)
    need(observed['uid'] == private_uid and observed['nlink'] == 1, 'private owner/alias: ' + name)
    private_bindings[name] = observed
    private_bytes[name] = data
data, observed = read_data(PRIVATE / 'receipt.json', source['receipt.json']['bytes'])
need(observed['sha256'] == source['receipt.json']['sha256'] and observed['mode'] == '0600', 'private receipt differs')
need(observed['uid'] == private_uid and observed['nlink'] == 1, 'private receipt owner/alias')
private_bindings['receipt.json'] = observed
private_inventory = sorted(p.name for p in PRIVATE.iterdir())

# Primary retained publisher metadata, rather than receipt assertions alone.
release = json.loads(private_bytes['release-20251014.json'])
need(release['id'] == 254523032 and release['tag_name'] == '20251014', 'release identity')
need(release['immutable'] is True and release['draft'] is False and release['prerelease'] is False, 'release state')
selected_assets = {}
for name in (ARCHIVE, 'SHA256SUMS'):
    matches = [asset for asset in release['assets'] if asset['name'] == name]
    need(len(matches) == 1, 'asset identity ambiguous')
    asset = matches[0]
    selected_assets[name] = {key: asset[key] for key in ('id', 'name', 'size', 'digest', 'state', 'url', 'browser_download_url')}
    need(asset['size'] == private_bindings[name]['bytes'], 'asset size')
    need(asset['digest'] == 'sha256:' + private_bindings[name]['sha256'], 'asset digest')
need(selected_assets[ARCHIVE]['id'] == 304305223 and selected_assets['SHA256SUMS']['id'] == 304306156, 'asset IDs')
checksum_rows = [line for line in private_bytes['SHA256SUMS'].decode('ascii').splitlines()
                 if line.split() and line.split()[-1] == ARCHIVE]
need(checksum_rows == [EXPECTED + '  ' + ARCHIVE], 'exact checksum entry')
need(sha(private_bytes[ARCHIVE]) == EXPECTED, 'archive digest')

# Read one bounded decompressed stream into memory, with no member output files.
chunks = []
expanded = 0
with gzip.GzipFile(fileobj=io.BytesIO(private_bytes[ARCHIVE]), mode='rb') as stream:
    while True:
        chunk = stream.read(min(1024 * 1024, 1024 ** 3 + 1 - expanded))
        if not chunk:
            break
        expanded += len(chunk)
        need(expanded <= 1024 ** 3, 'expanded bound')
        chunks.append(chunk)
raw = b''.join(chunks)

# Independent physical header walk; this exact archive has no GNU/PAX headers.
physical = []
offset = 0
while raw[offset:offset + 512] != b'\0' * 512:
    header = raw[offset:offset + 512]
    need(len(header) == 512 and len(physical) < 12000, 'physical header bound')
    stored = int(header[148:156].strip(b'\0 '), 8)
    need(sum(header[:148]) + 8 * 32 + sum(header[156:]) == stored, 'header checksum')
    size = int(header[124:136].strip(b'\0 ') or b'0', 8)
    need(header[156:157] in (b'0', b'\0', b'2'), 'unexpected physical type')
    payload_offset = offset + 512
    end = payload_offset + size
    padded_end = (end + 511) // 512 * 512
    need(end <= len(raw) and not any(raw[end:padded_end]), 'payload padding/bound')
    physical.append((offset, payload_offset, size))
    offset = padded_end
need(raw[offset:offset + 1024] == b'\0' * 1024 and not any(raw[offset:]), 'tar terminator/tail')

entries = []
payloads = {}
aliases = {}
ancestors = set()
max_path = max_component = max_file = 0
with tarfile.open(fileobj=io.BytesIO(raw), mode='r:') as archive:
    for item in archive:
        i = len(entries)
        need(i < len(physical) and (item.offset, item.offset_data, item.size) == physical[i], 'reader/header disagreement')
        name = item.name
        parts = name.split('/')
        need(parts[0] == 'python' and all(p and p not in ('.', '..') for p in parts), 'path root/components')
        need(not any(c in name for c in '\\:') and not any(unicodedata.category(c).startswith('C') for c in name), 'unsafe path character')
        need(len(name.encode('utf-8')) <= 4096 and not item.mode & ~0o777, 'path/mode bound')
        max_path = max(max_path, len(name.encode('utf-8')))
        max_component = max(max_component, *(len(p.encode('utf-8')) for p in parts))
        for j in range(1, len(parts) + 1):
            prefix = '/'.join(parts[:j])
            key = unicodedata.normalize('NFD', unicodedata.normalize('NFD', prefix).casefold())
            need(key not in aliases or aliases[key] == prefix, 'path alias')
            aliases[key] = prefix
            if j < len(parts):
                ancestors.add(prefix)
        need(item.isfile() or item.issym(), 'unexpected logical type')
        kind = 'file' if item.isfile() else 'symlink'
        entry = {'name': name, 'kind': kind, 'size': item.size,
                 'source_mode': oct(item.mode), 'payload_offset': item.offset_data}
        if kind == 'file':
            need(not item.linkname, 'regular linkname')
            payload = raw[item.offset_data:item.offset_data + item.size]
            entry['sha256'] = sha(payload)
            payloads[name] = payload
            max_file = max(max_file, item.size)
        else:
            need(item.size == 0, 'link payload')
            # Stronger observation for this archive: every target is a sibling basename.
            need(item.linkname and '/' not in item.linkname and item.linkname not in ('.', '..'), 'nonsibling link')
            need(not any(c in item.linkname for c in '\\:') and not any(unicodedata.category(c).startswith('C') for c in item.linkname), 'unsafe link name')
            entry['linkname'] = item.linkname
            entry['target'] = str(PurePosixPath(name).parent / item.linkname)
        entries.append(entry)
need(len(entries) == len(physical) == 1844, 'member totals')
names = {entry['name']: entry for entry in entries}
need(len(names) == len(entries), 'duplicate names')
need(not ancestors.intersection(names), 'file/link used as directory')
links = [entry for entry in entries if entry['kind'] == 'symlink']
need(len(links) == 8 and len(payloads) == 1836, 'types')
for entry in links:
    need(entry['target'] in payloads, 'link target not direct regular file')
manifest = sha(b''.join((json.dumps(entry, sort_keys=True, ensure_ascii=True) + '\n').encode('ascii') for entry in entries))
need(manifest == receipt['static_tar_inspection']['manifest_sha256_before_resolved_target_fields'], 'full manifest binding')
for entry in receipt['selected_member_bindings']:
    need(entry == names[entry['name']], 'selected member binding')
for entry in receipt['symlinks']:
    need(entry['resolved_target'] == entry['target'], 'retained link is not direct')
    need({k: v for k, v in entry.items() if k != 'resolved_target'} == names[entry['name']], 'symlink binding')

# Parse archive text as syntax/literals only, never through import/eval/exec.
patch = payloads['python/include/python3.13/patchlevel.h'].decode('utf-8')
need(re.search(r'^#define\s+PY_VERSION\s+"3\.13\.9"\s*$', patch, re.M), 'declared version')
need(re.search(r'^#define\s+PY_RELEASE_LEVEL\s+PY_RELEASE_LEVEL_FINAL\s*$', patch, re.M), 'release level')
need(re.search(r'^#define\s+PY_RELEASE_SERIAL\s+0\s*$', patch, re.M), 'release serial')
config_tree = ast.parse(payloads['python/lib/python3.13/_sysconfigdata__darwin_darwin.py'].decode('utf-8'))
assignments = [node.value for node in config_tree.body if isinstance(node, ast.Assign)
               and any(isinstance(target, ast.Name) and target.id == 'build_time_vars' for target in node.targets)]
need(len(assignments) == 1, 'build_time_vars assignment')
build_vars = ast.literal_eval(assignments[0])
selected_config = {key: build_vars[key] for key in receipt['version_and_platform']['sysconfig_literals']}
need(selected_config == receipt['version_and_platform']['sysconfig_literals'], 'sysconfig bindings')

# Independent Mach-O load-command reader. Keep LC_ID_DYLIB distinct from loads.
machos = {}
other_magic = {bytes.fromhex(value) for value in ('feedface', 'cefaedfe', 'feedfacf', 'cafebabe', 'bebafeca', 'cafebabf', 'bfbafeca')}
for name, payload in payloads.items():
    need(payload[:4] not in other_magic, 'other Mach-O format')
    if payload[:4] != b'\xcf\xfa\xed\xfe':
        continue
    magic, cpu, subtype, kind, count, size, flags, reserved = struct.unpack_from('<8I', payload)
    need(cpu == 0x0100000c and count <= 4096 and size + 32 <= len(payload), 'Mach-O header')
    cursor = 32
    commands = []
    for i in range(count):
        need(cursor + 8 <= 32 + size, 'Mach-O command header')
        command, length = struct.unpack_from('<II', payload, cursor)
        need(length >= 8 and cursor + length <= 32 + size, 'Mach-O command length')
        record = {'command': hex(command)}
        if command == 0x32:
            platform, minimum, sdk, tools = struct.unpack_from('<4I', payload, cursor + 8)
            need(24 + tools * 8 <= length, 'Mach-O build tools')
            version = lambda n: '{}.{}.{}'.format(n >> 16, (n >> 8) & 255, n & 255)
            record.update(platform=platform, min_os=version(minimum), sdk=version(sdk))
        elif command in (0xc, 0xd, 0x80000018, 0x8000001f, 0x80000023, 0x8000001c):
            string_offset = struct.unpack_from('<I', payload, cursor + 8)[0]
            need(12 <= string_offset < length, 'Mach-O string offset')
            text = payload[cursor + string_offset:cursor + length]
            need(b'\0' in text, 'Mach-O unterminated string')
            record['value'] = text.split(b'\0', 1)[0].decode('utf-8')
        if len(record) > 1:
            commands.append(record)
        cursor += length
    need(cursor == size + 32, 'Mach-O command total')
    machos[name] = {'cpu_type': hex(cpu), 'cpu_subtype': subtype, 'file_type': kind, 'load_commands': commands}
need(set(machos) == set(receipt['macho_metadata']) and len(machos) == 8, 'Mach-O inventory')
for name, meta in machos.items():
    retained = receipt['macho_metadata'][name]
    need(all(meta[key] == retained[key] for key in ('cpu_type', 'cpu_subtype', 'file_type')), 'Mach-O identity')
    need([{'command': 'LC_BUILD_VERSION', **{k: v for k, v in c.items() if k != 'command'}} for c in meta['load_commands'] if c['command'] == '0x32'] == retained['build_versions'], 'Mach-O versions')
    need([c['value'] for c in meta['load_commands'] if c['command'] in ('0xc', '0xd', '0x80000018', '0x8000001f', '0x80000023')] == retained['dylibs'], 'Mach-O dylibs')
    need([c['value'] for c in meta['load_commands'] if c['command'] == '0x8000001c'] == retained['rpaths'], 'Mach-O rpaths')

retained_scan = json.loads(private_bytes['static-inspection.json'])
need(retained_scan['members'] == [dict(e, resolved_target=e['target']) if e['kind'] == 'symlink' else e for e in entries], 'all retained member records')
shared_functions = ('require', 'sha256_file', 'clean_path', 'alias_key', 'safe_link_target', 'macho_metadata', 'scan')
scanner_tree = ast.parse(private_bytes['static-inspection.py.txt'].decode('utf-8'))
get_function = lambda tree, name: next(node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == name)
shared_match = {name: ast.dump(get_function(recipe_tree, name)) == ast.dump(get_function(scanner_tree, name)) for name in shared_functions}
need(all(shared_match.values()), 'shared scanner function source differs')

release_source, release_binding = read_data(ROOT / 'tools/release/release.py', 1024 * 1024)
need('file_digest' in payloads['python/lib/python3.13/hashlib.py'].decode('utf-8'), 'file_digest declaration absent')
total_payload = sum(entry['size'] for entry in entries)
need(total_payload == 46504040 and expanded == 47918592, 'expanded totals')
print(json.dumps({
    'schema': 'openergy.m1-python-extractor-independent-data-review.v1',
    'source_bindings': source,
    'recipe_python39_grammar_parse': True,
    'private_ancestor_file_metadata': ancestor_metadata,
    'private_inventory': private_inventory,
    'private_file_bindings': private_bindings,
    'release': {key: release[key] for key in ('id', 'tag_name', 'published_at', 'immutable', 'draft', 'prerelease', 'html_url', 'url', 'target_commitish')},
    'selected_assets': selected_assets,
    'matching_checksum_line': checksum_rows[0],
    'independent_archive_inspection': {
        'physical_headers': len(physical), 'members': len(entries),
        'types': dict(collections.Counter(e['kind'] for e in entries)),
        'payload_bytes': total_payload, 'decompressed_bytes': expanded,
        'manifest_sha256_before_resolved_target_fields': manifest,
        'implied_directories': len(ancestors), 'max_path_utf8_bytes': max_path,
        'max_component_utf8_bytes': max_component, 'largest_file_bytes': max_file,
        'symlinks': links, 'selected_members': receipt['selected_member_bindings'],
        'all_member_records_match_retained_scan': True,
        'all_headers_payloads_padding_and_tar_tail_read': True,
        'all_links_direct_internal_regular_targets': True,
        'no_file_or_symlink_parent': True, 'no_path_alias': True,
        'machos': machos, 'sysconfig_literals': selected_config,
        'declared_version': '3.13.9 final serial0',
        'stdlib_hashlib_file_digest_declaration_only': True,
    },
    'shared_scanner_function_ast_matches': shared_match,
    'release_py_binding': release_binding,
    'scope': {'archive_members_extracted': 0, 'supplied_code_executed_or_imported': False,
              'tests_or_mocks': False, 'remote_connection': False,
              'network_fetch': False, 'process_port_or_os_query': False,
              'executable_modes_changed': False, 'native_qualification': 'UNVERIFIED'},
}, indent=2, sort_keys=True))
