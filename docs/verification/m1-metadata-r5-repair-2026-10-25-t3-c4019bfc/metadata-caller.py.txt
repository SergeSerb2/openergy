class Stop(RuntimeError):

def require(condition, binding):
    if not condition:
        raise Stop(binding)


def clock_ns():
    """The only NEW absolute clock. No monotonic fallback or inferred offset."""
    require(sys.platform == 'darwin' and hasattr(time, 'clock_gettime_ns')
            and getattr(time, 'CLOCK_MONOTONIC', None) == CLOCK['clock_id'], 'explicit shared clock unavailable')
    value = time.clock_gettime_ns(time.CLOCK_MONOTONIC)
    require(type(value) is int and value >= 0, 'shared integer clock')
    return value


def remaining(deadline):
    value = deadline - clock_ns()
    if value <= 0:
        raise TimeoutError('absolute shared-clock deadline')
    return math.nextafter(value / NS, -math.inf)


@contextlib.contextmanager
def until(deadline):
    """Shared absolute ns deadline; nested alarm and postcheck. Parent owns outer."""
    duration = remaining(deadline)
    previous_handler = signal.getsignal(signal.SIGALRM)
    previous_timer = signal.getitimer(signal.ITIMER_REAL)
    entered = clock_ns()
    def expired(_signum, _frame):
        raise TimeoutError('absolute shared-clock deadline')
    signal.signal(signal.SIGALRM, expired)
    signal.setitimer(signal.ITIMER_REAL, min(duration, previous_timer[0]) if previous_timer[0] else duration)
    try:
        yield
        remaining(deadline)
    finally:
        signal.setitimer(signal.ITIMER_REAL, 0)
        signal.signal(signal.SIGALRM, previous_handler)
        if previous_timer[0]:
            signal.setitimer(signal.ITIMER_REAL, max(0.000001, previous_timer[0] - (clock_ns() - entered) / NS), previous_timer[1])


def common_deadline(deadline):
    """Conservative remaining duration, consumed only in unchanged common's domain."""
    # Sample common's local clock FIRST. Time between the samples is discarded.
    # Shorten by 1ms and round down; neither shared absolute ns nor an offset is
    # passed to common. The enclosing shared alarm + postcheck stay authoritative.
    local_before = time.monotonic()
    shared_after = clock_ns()
    budget = deadline - shared_after - 1000000
    require(budget > 0 and math.isfinite(local_before), 'common remaining budget')
    duration = math.nextafter(budget / NS, -math.inf)
    local_end = math.nextafter(local_before + duration, -math.inf)
    require(math.isfinite(local_end) and local_end > local_before, 'common local deadline')
    remaining(deadline)
    return local_end


def canonical(path):
    path = Path(path)
    require(path.is_absolute() and '..' not in path.parts and path.resolve(strict=True) == path, str(path))
    return path


def read(path, deadline, cap=FILE_CAP):
    with until(deadline):
        path = canonical(path)
        info = path.lstat()
        require(stat.S_ISREG(info.st_mode) and info.st_size <= cap, str(path))
        with path.open('rb') as stream:
            data = stream.read(cap + 1)
        require(len(data) == info.st_size and len(data) <= cap, str(path))
        return data


def sha(data):
    return hashlib.sha256(data).hexdigest()


def bound(path, expected, deadline, cap=FILE_CAP):
    data = read(path, deadline, cap)
    require(sha(data) == expected, str(path) + '#sha256')
    return data


def encoded(value):
    data = (json.dumps(value, sort_keys=True, separators=(',', ':'), ensure_ascii=True) + '\n').encode('utf-8')
    require(len(data) <= FILE_CAP, 'encoded file cap')
    return data


def sync_file(path, deadline):
    with until(deadline):
        fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
        try:
            os.fsync(fd)
        finally:
            os.close(fd)
        fd = os.open(path.parent, os.O_RDONLY)
        try:
            os.fsync(fd)
        finally:
            os.close(fd)


class Writer:
    def __init__(self, root, mode, deadline, common):
        self.root, self.mode, self.deadline, self.common = root, mode, deadline, common
        self.initial = 0
        self.first_failed_comparison = None
        self.first_failed_command = None
        self.active_command = None
        self.collected_commands = []
        self.publication_errors = []
        if mode == 'source':
            # One complete sum BEFORE the guard or any source write. No ledger.
            with until(deadline):
                require(not (root / 'bytes.json').exists() and not (root / 'bytes.lock').exists(), 'source must precede ledger')
                for parent, directories, files in os.walk(root, followlinks=False):
                    for name in directories + files:
                        p = Path(parent) / name
                        st = p.lstat()
                        require(st.st_uid == os.getuid() and not stat.S_ISLNK(st.st_mode), str(p))
                        if stat.S_ISDIR(st.st_mode):
                            require(stat.S_IMODE(st.st_mode) == 0o700, str(p))
                        else:
                            require(stat.S_ISREG(st.st_mode) and stat.S_IMODE(st.st_mode) == 0o600 and st.st_size <= FILE_CAP, str(p))
                            self.initial += st.st_size
                require(self.initial <= INITIAL_CAP, 'initial regular reservation')
        else:
            # Never initialize, rebuild or repair the existing ledger.
            read(root / 'bytes.json', deadline)
            require((root / 'bytes.lock').is_file() and not (root / 'bytes.lock').is_symlink(), 'existing byte lock')

    def _put(self, name, data):
        require(Path(name).name == name and len(data) <= FILE_CAP, 'fixed output filename/file cap')
        path = self.root / name
        with until(self.deadline):
            if self.mode == 'source':
                require(self.initial + len(data) <= INITIAL_CAP, 'initial aggregate reservation')
                # Exclusive0600 initial evidence. Prepare alone charges it later.
                self.initial += len(data)
                fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600)
                write_error = None
                try:
                    # One raw write. No Python buffer, short-write retry or flush.
                    require(os.write(fd, data) == len(data), 'short initial write')
                    os.fsync(fd)
                except BaseException as exc:
                    write_error = exc
                    raise
                finally:
                    # One close, including after an alarm/write/fsync failure.
                    # A raw fd owns no deferred payload. Never retry close.
                    try:
                        os.close(fd)
                    except Exception as exc:
                        if write_error is None:
                            raise
                        # Preserve the earlier failure and retain close separately.
                        self.publication_errors.append(dict(name=name, error='source close after earlier failure: '
                                                            + type(exc).__name__ + ': ' + str(exc)[:1024]))
            else:
                # Existing common charges regular capacity BEFORE temporary/final growth.
                self.common.write_bytes(path, data, common_deadline(self.deadline), critical=False)
            sync_file(path, self.deadline)
        return dict(path=str(path), bytes=len(data), sha256=sha(data))

    def put(self, name, data):
        try:
            return self._put(name, data)
        except Exception as exc:
            self.publication_errors.append(dict(name=name, error=type(exc).__name__ + ': ' + str(exc)[:1024]))
            raise

    def save(self, name, value):
        # Account actual UTF-8 JSON bytes, including worst-case escapes, before writes.
        try:
            data = encoded(value)
        except Exception as exc:
            self.publication_errors.append(dict(name=name, error=type(exc).__name__ + ': ' + str(exc)[:1024]))
            raise
        return self.put(name, data)

def collect(label, argv, root, phase_deadline):
    """Direct child only. Bounded nonblocking reads; no communicate or PID inference."""
    started = clock_ns()
    deadline = min(started + 10 * NS, phase_deadline - NS)  # Leave publication time inside phase.
    require(deadline - started > 750000000, label + ': no command/reap allowance')
    work_end = deadline - 500000000  # Kill/reap share the command's original ceiling.
    buffers = {'stdout': bytearray(), 'stderr': bytearray()}
    eof = {'stdout': False, 'stderr': False}
    observed, dropped, drop_unknown = 0, 0, False
    overflow, timed_out, deadline_loss = False, False, False
    proc, error, kill_error, reaped = None, None, None, False
    selector = selectors.DefaultSelector()
    kill_attempted, kill_delivered = False, False
    try:
        with until(work_end):
            proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                                    stderr=subprocess.PIPE, cwd=root, env=native_environment(root),
                                    shell=False, close_fds=True)
            HANDLES.append(dict(label=label, argv=list(argv), handle=proc, collector_status=None))
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                os.set_blocking(stream.fileno(), False)
                selector.register(stream, selectors.EVENT_READ, name)
            while True:
                status = proc.poll()  # Actual waitpid status from the retained handle.
                if status is not None and all(eof.values()):
                    reaped = True
                    break
                if clock_ns() >= work_end:
                    timed_out = deadline_loss = True
                    break
                if not selector.get_map():
                    # EOF without exit: block only within the remaining work interval.
                    try:
                        proc.wait(timeout=remaining(work_end))
                    except subprocess.TimeoutExpired:
                        timed_out = deadline_loss = True
                    continue
                for key, _mask in selector.select(min(0.05, remaining(work_end))):
                    # Read at most remaining capacity + one overflow byte. Never capture then cap.
                    allowance = STREAM_CAP - sum(len(x) for x in buffers.values())
                    try:
                        chunk = os.read(key.fd, min(65536, allowance + 1))
                    except BlockingIOError:
                        continue
                    if not chunk:
                        eof[key.data] = True
                        selector.unregister(key.fileobj)
                        continue
                    observed += len(chunk)
                    buffers[key.data].extend(chunk[:allowance])
                    if len(chunk) > allowance:
                        dropped += len(chunk) - allowance
                        overflow, drop_unknown = True, True
                        break
                if overflow:
                    break
    except Exception as exc:
        error = type(exc).__name__ + ': ' + str(exc)[:1024]
        if isinstance(exc, (TimeoutError, subprocess.TimeoutExpired)):
            timed_out = deadline_loss = True
    finally:
        if proc is not None:
            try:
                with until(deadline):
                    if proc.poll() is None:
                        kill_attempted = True
                        proc.kill()  # Only this retained, unreaped direct child.
                        kill_delivered = True
                    proc.wait(timeout=remaining(deadline))
                    reaped = True
            except Exception as exc:
                kill_error = type(exc).__name__ + ': ' + str(exc)[:1024]
                deadline_loss = True
            # No drain pass after failure; unread bytes remain explicitly unknown.
            if not all(eof.values()):
                drop_unknown = True
            if clock_ns() > deadline:
                deadline_loss = True
            for name in ('stdout', 'stderr'):
                stream = getattr(proc, name)
                if stream is not None:
                    try:
                        stream.close()
                    except Exception as exc:
                        if error is None:
                            error = type(exc).__name__ + ': ' + str(exc)[:1024]
        try:
            selector.close()
        except Exception as exc:
            if error is None:
                error = type(exc).__name__ + ': ' + str(exc)[:1024]
    finished = clock_ns()
    if finished > deadline:
        deadline_loss = True
    status = proc.returncode if proc is not None else None
    record = dict(label=label, argv=argv, pid=proc.pid if proc is not None else None,
                  handle_retained=proc is not None, actual_exit=status, reaped=reaped,
                  stdout_eof=eof['stdout'], stderr_eof=eof['stderr'],
                  observed_stream_bytes=observed, dropped_observed_bytes=dropped,
                  unread_or_dropped_unknown=drop_unknown, overflow=overflow,
                  timed_out=timed_out, deadline_loss=deadline_loss,
                  kill_attempted=kill_attempted, kill_delivered=kill_delivered,
                  error=error, kill_or_reap_error=kill_error, clock=CLOCK, started_ns=started,
                  command_deadline_ns=deadline, finished_ns=finished)
    if proc is not None:
        HANDLES[-1]['collector_status'] = record
    return record, bytes(buffers['stdout']), bytes(buffers['stderr'])


def complete(record):
    return (type(record['actual_exit']) is int and record['actual_exit'] == 0 and record['reaped'] is True
            and record['stdout_eof'] is True and record['stderr_eof'] is True and record['overflow'] is False
            and record['timed_out'] is False and record['deadline_loss'] is False
            and record['unread_or_dropped_unknown'] is False and record['dropped_observed_bytes'] == 0
            and record['error'] is None and record['kill_or_reap_error'] is None
            and record['kill_attempted'] is False
            and record['clock'] == CLOCK and type(record['started_ns']) is int
            and type(record['finished_ns']) is int and type(record['command_deadline_ns']) is int
            and record['started_ns'] <= record['finished_ns'] <= record['command_deadline_ns']
            and record['command_deadline_ns'] <= record['started_ns'] + 10 * NS)


def normalize(raw, bundle, label):
    value = raw.decode('utf-8', errors='strict').replace(bundle, BUNDLE)
    return sorted(value.splitlines()) if label.endswith('-verify') else value


def perform(writer, label, argv, bundle, basis=None, fresh=None, helper_mode=None, exact_stdout=None):
    record, stdout, stderr = collect(label, argv, writer.root, writer.deadline)
    # Retain the active label and ALL bounded collector facts BEFORE any writer,
    # fsync or comparator. This is a command failure, not a fabricated comparator.
    item = dict(label=label, collector_status=dict(record), stdout=None, stderr=None,
                os_record=None, comparison=None, publication={}, comparison_state='not_started')
    writer.active_command = item
    writer.collected_commands.append(item)
    if not complete(record) and writer.first_failed_command is None:
        writer.first_failed_command = dict(label=label, collector_status=dict(record))
    prefix = 'metadata-' + writer.mode + '-' + label
    for key, suffix in (('stdout', '.stdout'), ('stderr', '.stderr'), ('os_record', '-os.json')):
        item['publication'][key] = dict(path=str(writer.root / (prefix + suffix)), state='not_attempted', ref=None)
    for key, data in (('stdout', stdout), ('stderr', stderr), ('os_record', None)):
        publication = item['publication'][key]
        publication['state'] = 'attempted'
        try:
            ref = writer.save(prefix + '-os.json', record) if key == 'os_record' else writer.put(prefix + '.' + key, data)
        except Exception:
            # May be absent, partial, linked but not fsynced, or not durable.
            # Keep the attempted path separate from a verified durable reference.
            publication['state'] = 'failed_or_partial_unverified'
            raise
        publication.update(state='durable', ref=ref)
        item[key] = ref
        if key != 'os_record':
            record[key] = ref
    raw_record = item['os_record']
    # Payload comparison starts ONLY AFTER all raw/status publication is durable.
    item['comparison_state'] = 'evaluating'
    comparisons = {}
    comparison_error = None
    try:
        if complete(record):
            if basis is not None:
                comparisons['saved_basis'] = all(normalize(raw, bundle, label) == normalize(basis[name].encode('utf-8'), BUNDLE, label) for name, raw in (('stdout', stdout), ('stderr', stderr)))
            if fresh is not None:
                comparisons['fresh_source'] = all(normalize(raw, bundle, label) == normalize(fresh[name], BUNDLE, label) for name, raw in (('stdout', stdout), ('stderr', stderr)))
            if exact_stdout is not None:
                comparisons['fixed_python_metadata'] = stdout == exact_stdout.encode('utf-8') and stderr == b''
            if helper_mode is not None:
                facts = json.loads(stdout.decode('utf-8'))
                comparisons['file_observation'] = (facts['mode'] == helper_mode and facts['prefix_sha256'] == PREFIX_HASH and facts['prefix_entries'] == 2006 and facts['prefix_regular_bytes'] == 46504040 and facts['prefix_root_mode'] == '0o700' and facts['modules'] == 10)
                if helper_mode == 'copy':
                    comparisons['copy_file_observation'] = facts['source_files'] == 702 and facts['bundle_records'] == 167 and facts['sdk_headers'] == 10 and facts['copied_bundle'] == bundle
    except Exception as exc:
        comparison_error = type(exc).__name__ + ': ' + str(exc)[:1024]
    match = complete(record) and comparison_error is None and bool(comparisons) and all(comparisons.values())
    # The first failed comparator is exclusive and follows durable raw output/status.
    comparison = dict(label=label, os_record=raw_record, observations_complete=complete(record),
                      comparisons=comparisons, comparison_error=comparison_error, match=match)
    if not match and writer.first_failed_comparison is None:
        writer.first_failed_comparison = comparison
    item['comparison_state'] = 'assigned'
    ref = writer.save(prefix + '-comparison.json', comparison)
    item['comparison'] = ref
    item['comparison_state'] = 'durable'
    if not match:
        writer.save('metadata-' + writer.mode + '-first-failure.json', comparison)
        raise Stop(label + ': first failed status/comparison; see retained raw records')
    return dict(label=label, os_record=raw_record, comparison=ref, stdout=record['stdout'], stderr=record['stderr'])


def fresh_source(root, deadline, vectors):
    receipt = json.loads(read(root / 'metadata-source-result.json', deadline))
    require(receipt['schema'] == 'openergy.metadata-observations.v2' and receipt['clock'] == CLOCK and receipt['mode'] == 'source'
            and receipt['expected_basis_sha256'] == EXPECTED_HASH and receipt['completed_native_commands'] == 50
            and receipt['first_error'] is None and receipt['first_failed_command'] is None
            and receipt['first_failed_comparison'] is None and receipt['publication_errors'] == []
            and receipt['observation_sequence_complete'] is True,
            'fresh source result')
    items = receipt['native_records']
    require(len(items) == 50, 'fresh complete50')
    records = []
    for vector, item in zip(vectors, items):
        require(item['label'] == vector['label'], 'fresh source order')
        os_record = json.loads(parent_reference(root, item['os_record'], deadline))
        comparison = json.loads(parent_reference(root, item['comparison'], deadline))
        require(complete(os_record) and os_record['label'] == vector['label'] and os_record['argv'] == expand(vector, BUNDLE)
                and comparison['label'] == vector['label'] and comparison['match'] is True
                and comparison['os_record'] == item['os_record'] and comparison['comparisons'] == {'saved_basis': True}, 'fresh source actual status/comparator')
        require(os_record['stdout'] == item['stdout'] and os_record['stderr'] == item['stderr'], 'fresh source raw binding')
        streams = {name: parent_reference(root, item[name], deadline) for name in ('stdout', 'stderr')}
        require(sum(len(data) for data in streams.values()) <= STREAM_CAP, 'fresh source combined stream cap')
        records.append(streams)
    return records


def direct_children():
    # Do not poll or wait again here. Only actual statuses already observed on
    # retained handles; null remains null. Every direct handle appears once.
    return [dict(label=e['label'], argv=e['argv'], pid=e['handle'].pid,
                 actual_exit=e['handle'].returncode, handle_retained=True,
                 collector_status=e['collector_status']) for e in HANDLES]


def emit_summary(result, deadline):
    with until(deadline):
        data = encoded(result)
        require(len(data) <= STREAM_CAP, 'bounded outer fallback summary')
        # Leave stdout nonblocking. No BufferedWriter ever receives this payload.
        # One raw attempt only: EAGAIN, partial, alarm or other error ends it.
        os.set_blocking(1, False)
        remaining(deadline)
        require(os.write(1, data) == len(data), 'short outer summary')


def main():
    require(len(sys.argv) == 5 and sys.argv[1] in ('source', 'copy'), 'fixed CLI')
    mode, root = sys.argv[1], canonical(sys.argv[2])
    require(root == Path(__file__).resolve(strict=True).parent and Path.cwd() == root, 'materialized caller cwd')
    require(root.parent == Path('/Users/m1-dev/.local/state/openergy-verification') and root.name.startswith('m1-ordinary-reviewed-flow-20261025-'), 'fresh evidence namespace')
    info = root.stat()
    require(info.st_uid == os.getuid() and stat.S_IMODE(info.st_mode) == 0o700 and info.st_dev == Path('/private/tmp').stat().st_dev, 'evidence owner/mode/filesystem')
    require(sys.argv[3].isascii() and sys.argv[3].isdigit() and len(sys.argv[3]) <= 20, 'integer shared-clock CLI')
    started = int(sys.argv[3])
    require(str(started) == sys.argv[3] and started <= clock_ns(), 'parent explicit shared-clock start')
    deadline = started + 120 * NS
    os.umask(0o077)
    writer = None
    result = dict(schema='openergy.metadata-observations.v2', clock=CLOCK, mode=mode, expected_basis_sha256=EXPECTED_HASH,
                  started_ns=started, phase_deadline_ns=deadline, native_records=[], auxiliary_records=[],
                  completed_native_commands=0, observation_sequence_complete=False, first_error=None,
                  artifact_gate_authority=False, runtime='UNVERIFIED', actual_direct_children=[],
                  first_failed_command=None, first_failed_comparison=None, collected_commands=[],
                  active_command=None, publication_errors=[])
    try:
        with until(deadline):
            normalize_startup(root, mode)
            vectors, expected = pin_inputs(root, deadline)
            if mode == 'source':
                parent_bytes = bound(root / 'metadata-source-parent.json', sys.argv[4], deadline, STREAM_CAP)
            else:
                # Parent captures entry authentication in memory. No unmetered
                # post-prepare file or second private writer entry is needed.
                parent_bytes = sys.stdin.buffer.read(STREAM_CAP + 1)
                require(len(parent_bytes) <= STREAM_CAP and sha(parent_bytes) == sys.argv[4], 'copy bounded stdin parent binding')
            parent = json.loads(parent_bytes)
            require(set(parent) == {'schema', 'mode', 'clock', 'phase_started_ns', 'phase_deadline_ns', 'source_file_qualification', 'entry_authentication'}, 'parent record shape')
            require(parent['schema'] == 'openergy.metadata-parent-boundary.v2' and parent['mode'] == mode
                    and parent['clock'] == CLOCK and type(parent['phase_started_ns']) is int
                    and type(parent['phase_deadline_ns']) is int and parent['phase_started_ns'] == started
                    and parent['phase_deadline_ns'] == deadline, 'parent phase binding')
            common = None
            if mode == 'source':
                # Parent owns exact /usr/bin/python3 launch. A system launcher
                # can report its toolchain executable in sys.executable.
                require(sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode and not sys.flags.optimize, 'source system -I -S -B flags')
                require(parent['entry_authentication'] is None, 'source authenticates entry-template itself')
                inspect_completed_source(root, parent['source_file_qualification'], started, deadline)
            else:
                require(parent['source_file_qualification'] is None, 'copy parent boundary')
                auth = parent['entry_authentication']
                parent_streams = inspect_parent_helper(root, auth, started, deadline)
                # This check binds data AFTER entry. It cannot replace the parent's pre-entry gate.
                common = import_common(root, deadline)
            writer = Writer(root, mode, deadline, common)
            writer.save('metadata-' + mode + '-once.json', dict(mode=mode, clock=CLOCK, started_ns=started, phase_deadline_ns=deadline, parent_record_sha256=sys.argv[4], execution_attempts=1))
            if mode == 'copy':
                # ALL new post-prepare evidence uses regular pre-write reservations.
                writer.put('metadata-copy-parent.json', parent_bytes)
                for name, data in parent_streams.items():
                    writer.put('metadata-copy-parent-entry.' + name, data)
            bundle = BUNDLE if mode == 'source' else copy_bundle(root, deadline)
            helper = [SYSTEM, '-I', '-S', '-B', str(root / 'file-qualification.recipe.py.txt'), str(root)]
            if mode == 'source':
                result['auxiliary_records'].append(perform(writer, 'entry-template', helper + ['entry-template'], bundle, helper_mode='entry-template'))
                # Exact first private-Python entry only AFTER the trusted helper completed.
                result['auxiliary_records'].append(perform(writer, 'python-version', [PYTHON, '-I', '-S', '-B', '--version'], bundle, exact_stdout='Python 3.13.9\n'))
                result['auxiliary_records'].append(perform(writer, 'python-architecture', ['/usr/bin/lipo', '-archs', PYTHON], bundle, exact_stdout='arm64\n'))
                fresh = [None] * 50
            else:
                fresh = fresh_source(root, deadline, vectors)
                result['auxiliary_records'].append(perform(writer, 'copy-files-before', helper + ['copy'], bundle, helper_mode='copy'))
            for vector, basis, source in zip(vectors, expected, fresh):
                result['native_records'].append(perform(writer, vector['label'], expand(vector, bundle), bundle, basis=basis, fresh=source))
                result['completed_native_commands'] += 1
            if mode == 'copy':
                result['auxiliary_records'].append(perform(writer, 'copy-files-after', helper + ['copy'], bundle, helper_mode='copy'))
            result['observation_sequence_complete'] = True
    except Exception as exc:
        result['first_error'] = type(exc).__name__ + ': ' + str(exc)[:2048]
        result['observation_sequence_complete'] = False
    if writer is not None:
        result.update(collected_commands=writer.collected_commands, active_command=writer.active_command,
                      first_failed_command=writer.first_failed_command, first_failed_comparison=writer.first_failed_comparison,
                      publication_errors=writer.publication_errors)
        if writer.first_failed_command is not None:
            result['stop_detail'] = result['first_error']
            result['first_error'] = writer.first_failed_command['label'] + ': first failed COMMAND status'
        elif writer.first_failed_comparison is not None:
            result['stop_detail'] = result['first_error']
            result['first_error'] = writer.first_failed_comparison['label'] + ': first failed status/comparison'
    # This is a pre-publication observation, never a claim about caller actual exit.
    # Parent owns actual outer exit/EOF and must include it within the SAME phase.
    try:
        with until(deadline):
            result['observations_finished_ns'] = clock_ns()
            result['actual_direct_children'] = direct_children()
            require(writer is not None, 'no admitted writer')
            writer.save('metadata-' + mode + '-result.json', result)
    except Exception as exc:
        result['observation_sequence_complete'] = False
        result['result_publication_error'] = type(exc).__name__ + ': ' + str(exc)[:1024]
        if result['first_error'] is None:
            result['first_error'] = result['result_publication_error']
        result['actual_direct_children'] = direct_children()
    # One stdout summary attempt, no retry or late writer/extra allowance. If
    # time is gone, the parent keeps its actual handle/partial streams/exit and
    # missing facts stay missing. A partial summary is never issued a second time.
    try:
        emit_summary(result, deadline)
        remaining(deadline)
    except Exception:
        return 1
    return 0 if result['observation_sequence_complete'] else 1


if __name__ == '__main__':
    # Bypass finalization on every return or uncaught pre-admission exception.
    # No shutdown flush, traceback writer, handle destructor or payload retry.
    try:
        exit_status = main()
    except BaseException:
        exit_status = 1
