function Build-Failure($errorRecord){
 $text=if($errorRecord -is [Management.Automation.ErrorRecord]){($errorRecord|Out-String)}else{[string]$errorRecord}
 if($null -eq $script:buildFirstError){$script:buildFirstError=$text}
 $script:buildErrors.Add($text)
}
function Build-Before($deadline){if([Diagnostics.Stopwatch]::GetTimestamp() -ge $deadline){throw 'absolute build lifecycle deadline reached'}}
function Build-Ms($deadline,$cap){
 Build-Before $deadline
 $left=[Math]::Floor(1000.0*($deadline-[Diagnostics.Stopwatch]::GetTimestamp())/[Diagnostics.Stopwatch]::Frequency)
 if($left -lt 1){throw 'no build lifecycle wait budget remains'}
 [int][Math]::Min($cap,$left)
}
function Build-Identity($process,$deadline){
 if((Build-Ms $deadline 1000) -lt 1000){throw 'no bounded CIM observation budget remains'}
 $cim=Get-CimInstance Win32_Process -Filter ('ProcessId='+$process.Id) -OperationTimeoutSec 1
 Build-Before $deadline
 if(-not $cim -or -not $cim.CreationDate -or -not $cim.ExecutablePath -or -not $cim.CommandLine){throw 'compiler creation identity unavailable'}
 if(-not (Same-Time $process.StartTime $cim.CreationDate) -or $process.Path -ine $cim.ExecutablePath){throw 'compiler retained handle identity mismatch'}
 [ordered]@{pid=$process.Id;start_utc=$process.StartTime.ToUniversalTime().ToString('o');creation_utc=$cim.CreationDate.ToUniversalTime().ToString('o');parent_pid=$cim.ParentProcessId;path=$process.Path;command_line=$cim.CommandLine;session=$process.SessionId;priority=$process.PriorityClass.ToString();retained_handle=$true;recorded_utc=[DateTime]::UtcNow.ToString('o')}
}
function Build-Owner($process,$isRoot){
 $record=[ordered]@{pid=$null;root=$isRoot;identity=$null;owned_generation_verified=$isRoot;retained_handle=$false;kill_called=$false;kill_ticks=$null;retained_handle_signalled=$false;exit_code=$null;handle_closed=$false;normal_exit_acceptance=$false;process_disposed=$false;error=$null}
 $entry=@{process=$process;handle=$null;started=$false;verified=$isRoot;record=$record}
 $script:buildOwners.Add($entry)
 return $entry
}
function Observe-Compilers($deadline){
 if((Build-Ms $deadline 1000) -lt 1000){throw 'no bounded compiler poll budget remains'}
 $all=@(Get-CimInstance Win32_Process -Filter "Name='cargo.exe' OR Name='rustc.exe' OR Name='cargo-clippy.exe' OR Name='clippy-driver.exe' OR Name='link.exe' OR Name='cl.exe' OR Name LIKE 'build-script%'" -OperationTimeoutSec 1)
 Build-Before $deadline
 for($pass=0;$pass -lt 4;$pass++){foreach($p in $all){
  Build-Before $deadline
  # A reused PID is never a new owned generation merely because its number is known.
  if($script:buildKnown.ContainsKey([int]$p.ProcessId)){continue}
  if(-not $script:buildKnown.ContainsKey([int]$p.ParentProcessId)){continue}
  $parent=$script:buildKnown[[int]$p.ParentProcessId]
  if($parent.process.WaitForExit(0) -or $p.CreationDate.ToUniversalTime() -lt ([DateTime]$parent.record.identity.start_utc).ToUniversalTime()){continue}
  try{$live=[Diagnostics.Process]::GetProcessById([int]$p.ProcessId)}catch [ArgumentException]{
   $races.Add([ordered]@{pid=$p.ProcessId;message=$_.Exception.Message;recorded_utc=[DateTime]::UtcNow.ToString('o');ownership_not_inferred=$true});continue
  }
  $entry=Build-Owner $live $false;$entry.started=$true
  try{
   $entry.handle=$live.SafeHandle;$entry.record.retained_handle=$true;$entry.record.pid=$live.Id
   $id=Build-Identity $live $deadline
   if(-not (Same-Time ([DateTime]$id.creation_utc) $p.CreationDate) -or $id.parent_pid -ne $p.ParentProcessId -or $id.path -ine $p.ExecutablePath -or $id.command_line -cne $p.CommandLine -or $parent.process.WaitForExit(0)){throw 'compiler parent/generation changed; no ownership inferred'}
   $entry.record.identity=$id;$entry.verified=$true;$entry.record.owned_generation_verified=$true
   $script:buildKnown[[int]$p.ProcessId]=$entry
   $samples.Add([ordered]@{recorded_utc=$id.recorded_utc;name=$p.Name;pid=$id.pid;parent_pid=$id.parent_pid;creation_utc=$id.creation_utc;session=$id.session;priority=$id.priority;path=$id.path;command_line=$id.command_line})
   if($id.priority -ne 'BelowNormal' -or $id.session -ne 0){throw 'compiler descendant priority/session mismatch'}
  }catch{
   if($entry.verified -or -not $entry.handle -or -not $live.WaitForExit(0)){throw}
   $entry.record.error=($_|Out-String)
   $races.Add([ordered]@{pid=$p.ProcessId;message=$entry.record.error;recorded_utc=[DateTime]::UtcNow.ToString('o');retained_handle_already_signalled=$true;ownership_not_inferred=$true})
  }
 }}
}
