def sha(p):
    with p.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()

@contextmanager
def exchange_socket(s):
    try:
        with ACTION_LOCK:
            budget()
            with IO_LOCK:
                SOCKETS.add(s)
        yield s
    finally:
        with IO_LOCK:
            SOCKETS.discard(s)
        s.close()


def stop_io():
    with IO_LOCK:
        sockets, readers = list(SOCKETS), list(READERS)
    for s in sockets:
        try:
            s.shutdown(socket.SHUT_RDWR)
        except OSError:
            pass  # close also interrupts connect/recv/send; no request is retried
        s.close()
    for p in readers:
        # These are direct read-only command children, never product or shared processes.
        if p.poll() is None:
            p.kill()

def inventory(root):
    result = {}
    total = 0
    for p in sorted(root.rglob('*')):
        require(len(result) < 8192, 'inventory entry limit')
        s = p.lstat()
        require(s.st_uid == UID, 'foreign owner')
        n = p.relative_to(root).as_posix()
        if p.is_symlink():
            require(p.resolve(strict=True).is_relative_to(root.resolve()), 'escaping link')
            result[n] = dict(symlink=os.readlink(p))
        elif stat.S_ISREG(s.st_mode):
            require(s.st_nlink == 1, 'hard link')
            total += s.st_size
            require(total <= 256 * 1024 * 1024, 'inventory byte limit')
            result[n] = dict(bytes=s.st_size, sha256=sha(p), mode=oct(stat.S_IMODE(s.st_mode)))
        elif stat.S_ISDIR(s.st_mode):
            result[n] = dict(directory=True, mode=oct(stat.S_IMODE(s.st_mode)))
        else:
            raise RuntimeError('special file in immutable inventory')
    return result

class BSD(C.Structure):
    _fields_ = [(n, C.c_uint32) for n in ['flags', 'status', 'xstatus', 'pid', 'ppid', 'uid',
                  'gid', 'ruid', 'rgid', 'svuid', 'svgid', 'reserved']] + [
                  ('comm', C.c_char * 16), ('name', C.c_char * 32)] + [
                  (n, C.c_uint32) for n in ['nfiles', 'pgid', 'pjobc', 'tdev', 'tpgid']] + [
                  ('nice', C.c_int32), ('start_sec', C.c_uint64), ('start_usec', C.c_uint64)]


LIBPROC = None


def raw_identity(pid):
    info = BSD()
    C.set_errno(0)
    n = LIBPROC.proc_pidinfo(pid, 3, 0, C.byref(info), C.sizeof(info))
    if n == 0 and C.get_errno() in (errno.ESRCH, errno.ENOENT):
        return None
    require(n == C.sizeof(info), 'independent process identity unavailable')
    buf = C.create_string_buffer(4096)
    n = LIBPROC.proc_pidpath(pid, buf, len(buf))
    require(n > 0, 'process executable path unavailable; no exit inference')
    return dict(pid=pid, ppid=info.ppid, uid=info.uid, start_sec=info.start_sec,
                start_usec=info.start_usec, nice=info.nice, executable=os.fsdecode(buf.value), status=info.status)

def same(now, prior):

def key(p):

    return tuple(p[n] for n in ['pid', 'uid', 'start_sec', 'start_usec'])


class IdentityLedger:
    def __init__(self):
        self.lock = threading.RLock()
        self.rows, self.parents = {}, {}
        self.initial_pid = self.old_key = None
        self.last_sample = None

    def bind_initial(self, pid):
        with self.lock:
            require(self.initial_pid is None, 'no repeated launch')
            self.initial_pid = pid
            self.validate()

    def add(self, p):
        require(p['uid'] == UID, 'fixture UID mismatch')
        k = key(p)
        previous = self.rows.get(k)
        if previous:
            require(previous['executable'] == p['executable'] or
                    (previous['executable'] == str(ENTRY) and p['executable'] == str(IMAGE)), 'unexpected exec')
        else:
            # Preserve the first overflow row, then fail closed without an unbounded in-memory ledger.
            save('identity-first-observed', p)
            require(len(self.rows) < 32, 'total process identity bound; retain overflow evidence')
        self.rows[k] = dict(p)
        if p['executable'] in [str(ENTRY), str(IMAGE), str(DAEMON)]:
            require(p['nice'] == 15, 'observed target/daemon must be absolute nice15')

    def observe(self, rows):
        with self.lock:
            # Record every supplied identity before checking role limits.
            for p in rows:
                self.add(p)
            for p in rows:
                if p['executable'] != str(DAEMON):
                    continue
                child_key = key(p)
                parent = raw_identity(p['ppid'])
                if parent and parent['uid'] == UID and parent['executable'] in [str(ENTRY), str(IMAGE)]:
                    self.add(parent)
                    parent_key = key(parent)
                    require(child_key not in self.parents or self.parents[child_key] == parent_key, 'daemon parent changed')
                    self.parents[child_key] = parent_key
                elif child_key not in self.parents:
                    prior = [k for k, row in self.rows.items() if row['pid'] == p['ppid'] and
                             row['executable'] in [str(ENTRY), str(IMAGE)]]
                    require(parent is None and len(prior) == 1, 'daemon has no independently matched parent')
                    self.parents[child_key] = prior[0]
                # An already associated child may be reparented when its app exits.
            self.validate()

    def validate(self):
        apps = {k: p for k, p in self.rows.items() if p['executable'] in [str(ENTRY), str(IMAGE)]}
        daemons = {k: p for k, p in self.rows.items() if p['executable'] == str(DAEMON)}
        if self.initial_pid is not None:
            old = [k for k, p in apps.items() if p['pid'] == self.initial_pid]
            require(len(old) <= 1, 'initial app PID reused')
            if old:
                require(self.old_key is None or self.old_key == old[0], 'old parent identity changed')
                self.old_key = old[0]
        if self.old_key is not None:
            old_children = [k for k, parent in self.parents.items() if parent == self.old_key]
            require(len(old_children) <= 1, 'old-parent replacement daemon; candidate allowance cannot be used')
            require(all((p['start_sec'], p['start_usec']) >= self.old_key[2:] for p in apps.values()), 'preexisting app appeared')
        require(len(apps) <= 2 and len(daemons) <= 2, 'unexpected app/daemon restart identity')
        for parent in set(self.parents.values()):
            require(sum(value == parent for value in self.parents.values()) <= 1, 'replacement daemon for matched parent')

    def match_ready(self, build, app, child):
        self.observe([app, child])
        with self.lock:
            require(self.old_key is not None and self.parents[key(child)] == key(app), 'independent parent binding')
            require((key(app) == self.old_key) == (build == '1'), 'old/new readiness role mismatch')
            save('ledger-parent-role', dict(build=build, app=app, daemon=child))

    def owned(self):
        with self.lock:
            return [dict(p) for p in self.rows.values() if p['executable'] in EXECUTABLE_HASHES]

    def old_children(self):
        with self.lock:
            return [dict(self.rows[k]) for k, parent in self.parents.items() if parent == self.old_key]

    def old_members(self):
        with self.lock:
            return ([dict(self.rows[self.old_key])] if self.old_key in self.rows else []) + self.old_children()

def environment(pid):
    # KERN_PROCARGS2 only for independently owned target/daemon PIDs. Never host-wide.
    p = identity(pid)
    require(p and p['uid'] == UID and p['executable'] in [str(IMAGE), str(DAEMON)], 'environment PID gate')
    libc = C.CDLL('/usr/lib/libSystem.B.dylib', use_errno=True)
    mib = (C.c_int * 3)(1, 49, pid)
    data = C.create_string_buffer(256 * 1024)
    size = C.c_size_t(len(data))
    require(libc.sysctl(mib, 3, data, C.byref(size), None, 0) == 0, 'owned environment unavailable')
    raw = data.raw[:size.value]
    argc = struct.unpack_from('i', raw)[0]
    require(0 < argc < 32, 'argument count')
    offset = raw.index(b'\0', 4) + 1
    while offset < len(raw) and raw[offset] == 0:
        offset += 1
    args = []
    for _ in range(argc):
        end = raw.index(b'\0', offset)
        args.append(os.fsdecode(raw[offset:end])); offset = end + 1
    values = {}
    for value in raw[offset:].split(b'\0'):
        if b'=' in value:
            k, v = value.split(b'=', 1)
            values[os.fsdecode(k)] = os.fsdecode(v)
    return dict(arguments=args, values=values)


def ro(label, args, allowed=(0,), filtered=False):
    # Read-only metadata/OS evidence commands, each once, bounded output and time.
    deadline = time.monotonic() + 5
    with ACTION_LOCK:
        budget()
        p = subprocess.Popen(list(map(str, args)), stdout=subprocess.PIPE, stderr=subprocess.PIPE, env=ENV)
        with IO_LOCK:
            READERS.add(p)
    try:
        stdout, stderr = p.communicate(timeout=remaining(deadline))
        budget()
    except BaseException as error:
        if p.poll() is None:
            p.kill()
        try:
            p.communicate(timeout=.5)  # bounded drain/reap; no reader thread or unlimited wait
        except subprocess.TimeoutExpired:
            pass  # hard timer ends the controller; retain state, never claim reader cleanup
        save(label + '-first-error', dict(error=str(error), argv=list(map(str, args))))
        raise
    finally:
        with IO_LOCK:
            if p.poll() is not None:
                READERS.discard(p)
    require(len(stdout) + len(stderr) <= 32 * 1024 * 1024, 'post-capture OS evidence byte limit')
    out = stdout.decode(errors='replace')
    if filtered:
        out = '\n'.join(line for line in out.splitlines() if IDENTIFIER in line or str(ROOT) in line)
    save(label, dict(argv=list(map(str, args)), exit=p.returncode, stdout=out, stderr=stderr.decode(errors='replace')))
    require(p.returncode in allowed, 'OS command failed: ' + label)
    return p.returncode, out

def exact_read(s, count, deadline):
    data = b''
    while len(data) < count:
        s.settimeout(remaining(deadline))
        chunk = s.recv(count - len(data))
        require(chunk, 'IPC EOF')
        data += chunk
    return data


def ipc():
    result = {}
    # Two requests per launch. No subscriptions or mutation methods. At most four total.
    with exchange_socket(socket.socket(socket.AF_UNIX)) as s:
        # Each exchange has one monotonic deadline. The first includes connect.
        deadline = min(time.monotonic() + 2, WORK_DEADLINE, WHOLE_DEADLINE)
        s.settimeout(remaining(deadline))
        s.connect(str(SOCKET))
        for request_id, method in [(1, 'status.get'), (2, 'config.get')]:
            if request_id == 2:
                deadline = min(time.monotonic() + 2, WORK_DEADLINE, WHOLE_DEADLINE)
            data = json.dumps(dict(id=request_id, method=method, params=None), separators=(',', ':')).encode()
            require(len(data) <= 256, 'IPC request bound')
            s.settimeout(remaining(deadline))
            s.sendall(struct.pack('!I', len(data)) + data)
            length = struct.unpack('!I', exact_read(s, 4, deadline))[0]
            require(0 < length <= 65536, 'IPC response bound')
            value = json.loads(exact_read(s, length, deadline))
            remaining(deadline)
            require(value.get('id') == request_id and 'error' not in value and 'result' in value, 'IPC response')
            result[method] = value['result']
    return result

class Feed(socketserver.BaseRequestHandler):
    def handle(self):
        with exchange_socket(self.request):
            self.exchange()

    def exchange(self):
        budget()
        server = self.server
        connection_deadline = min(time.monotonic() + 2, WORK_DEADLINE, WHOLE_DEADLINE)
        raw = b''
        while b'\r\n\r\n' not in raw:
            self.request.settimeout(remaining(connection_deadline))
            part = self.request.recv(1024)
            require(part and len(raw) + len(part) <= 8192, 'HTTP header bound')
            raw += part
        head, body = raw.split(b'\r\n\r\n', 1)
        require(not body, 'HTTP request body refused')
        lines = head.split(b'\r\n')
        method, path, version = lines[0].decode('ascii').split(' ')
        require(method in ['GET', 'HEAD'] and version in ['HTTP/1.0', 'HTTP/1.1'], 'HTTP protocol')
        require(path in ['/appcast.xml', '/openergy-fixture.zip'], 'HTTP route refused')
        require(not any(line.lower().startswith((b'range:', b'transfer-encoding:', b'content-length:')) for line in lines[1:]), 'HTTP body/range refused')
        key = method + ' ' + path
        require(key not in server.requests and len(server.requests) < 4, 'duplicate/request budget')
        server.requests.add(key)
        p = ROOT / 'public' / path[1:]
        data = p.read_bytes()
        response = (f'HTTP/1.1 200 OK\r\nContent-Length: {len(data)}\r\n'
                    'Content-Type: application/octet-stream\r\nConnection: close\r\n\r\n').encode()
        self.request.settimeout(remaining(connection_deadline))
        self.request.sendall(response)
        if method == 'GET':
            self.request.settimeout(remaining(connection_deadline))
            self.request.sendall(data)
            remaining(connection_deadline)
            server.sent[path] = len(data)
        save('HTTP-complete', dict(method=method, path=path, length=len(data), content_sha256=sha(p),
             peer=self.client_address[0], sent_body=method == 'GET'))


class Server(socketserver.TCPServer):
    allow_reuse_address = False
    request_queue_size = 2
    def handle_error(self, request, client_address):
        self.failure = True
        fault('HTTP', sys.exc_info()[1])


def serve_until_cancelled(server):
    try:
        server.timeout = .1
        while not CANCEL.is_set():
            server.handle_request()
    except BaseException as error:
        fault('HTTP-thread', error)
    finally:
        server.server_close()


def terminal_stop(source, message):
    # Independent of the main thread and all Python evidence/ledger locks.
    # One fixed terminal record, then process exit kills all controller threads.
    try:
        data = json.dumps(dict(source=source, error=message, at=time.monotonic(),
                               retained_state=True, runtime_accepted=False)).encode() + b'\n'
        fd = os.open(RAW / 'terminal-deadline-or-quiescence.json', os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
        try:
            os.write(fd, data)
        finally:
            os.close(fd)
    finally:
        os._exit(1)


def hard_deadline():

    except BaseException as error:
        terminal_stop('hard-timer', str(error))


def recover(plan, caches, server_thread, observer):

        if not CLEANUP.wait(max(0, WORK_DEADLINE - time.monotonic())):
            fault('work-deadline', '180s control cutoff; enter owned recovery within the 240s whole deadline')
        CANCEL.set()
        if not ACTION_LOCK.acquire(timeout=1):
            terminal_stop('control-quiescence', 'active launch/quit could not quiesce; retain state; no late cleanup')
        ACTION_LOCK.release()  # CANCEL prevents any later control mutation under this gate.
        cleanup_start = time.monotonic()
        stop_io()
        server_thread.join(timeout=remaining(time.monotonic() + 2.5))
        require(not server_thread.is_alive(), 'HTTP thread not stopped; retain caches')
        # No normal quit here: before validated-new, only delayed recovery is authorized.
        term_seen = set()
        while time.monotonic() - cleanup_start < 15:
            budget()
            for p in PRODUCTS:
                p.poll()  # reap owned direct children before independent ESRCH checks
            for p in LEDGER.owned():
                if gone(p):
                    continue
                if time.monotonic() - cleanup_start >= 10 and key(p) not in term_seen:
                    signal_owned(p, signal.SIGTERM)
                    term_seen.add(key(p))
            time.sleep(.05)
        for p in LEDGER.owned():
            if not gone(p):
                require(time.monotonic() - cleanup_start >= 15, 'forcekill deadline')
                signal_owned(p, signal.SIGKILL)
        os_evidence('postflight')
        for job in JOBS:
            code, output = ro('cleanup-job-state', ['/bin/launchctl', 'print', f'gui/{UID}/{job}'], allowed=(0, 113))
            if code == 0:
                require(time.monotonic() - cleanup_start >= 15, 'job bootout deadline')
                programs = re.findall(r'^\s*program = (.+)$', output, re.M)
                require(len(programs) == 1, 'job program not independently identified')
                program = Path(programs[0])
                require(program.is_file() and not program.is_symlink() and program.lstat().st_uid == UID and
                        program.resolve(strict=True) == program and
                        (program.is_relative_to(ROOT) or any(program.is_relative_to(p) for p in caches)) and
                        sha(program) in plan['allowed_helper_sha256'], 'job program ownership/hash mismatch')
                ro('cleanup-exact-job-bootout', ['/bin/launchctl', 'bootout', f'gui/{UID}/{job}'])
                ro('cleanup-exact-job-absent', ['/bin/launchctl', 'print', f'gui/{UID}/{job}'], allowed=(113,))
        # Always observe the full quiet interval, including when the first sample is empty.
        quiet_deadline = time.monotonic() + 5
        while time.monotonic() < quiet_deadline:
            budget()
            for p in PRODUCTS:
                p.poll()
            residual = fixture_processes()
            save('quiet-process-sample', residual)
            require(not residual, 'owned helper/process remnants; retain caches and report')
            time.sleep(.05)
        require(all(gone(p) for p in LEDGER.owned()), 'independent final process exits missing')
        old_children = LEDGER.old_children()
        require(all(gone(p) for p in old_children), 'independent exit missing for an observed old child')
        save('independent-all-old-child-exits', dict(children=old_children, each_ESRCH=True))
        predicate = 'eventMessage CONTAINS "' + IDENTIFIER + '" OR eventMessage CONTAINS "' + str(ROOT) + '"'
        ro('fixture-LS-Gatekeeper-XProtect-log', ['/usr/bin/log', 'show', '--last', '4m', '--style', 'ndjson',
            '--predicate', predicate], filtered=True)
        require(CONTROL_DONE.is_set(), 'control path still blocked; retain caches and stop controller')
        with IO_LOCK:
            require(all(p.poll() is not None for p in READERS), 'OS reader still live; retain caches')
        require(not any(e['source'] in ['observer', 'ledger'] for e in ERRORS), 'observer incomplete; retain caches')
        for p in caches:
            budget()
            require(not fixture_processes(), 'late fixture process; retain cache')
            if p.exists() or p.is_symlink():
                s = p.lstat()
                require(stat.S_ISDIR(s.st_mode) and s.st_uid == UID and p.resolve() == p, 'cache cleanup ownership')
                before = dict(device=s.st_dev, inode=s.st_ino, inventory=inventory(p))
                save('owned-cache-before-removal', dict(path=str(p), **before))
                current = p.lstat()
                require((current.st_dev, current.st_ino) == (s.st_dev, s.st_ino), 'cache identity changed')
                budget()
                import shutil
                shutil.rmtree(p)
                require(not p.exists(), 'cache removal incomplete')
        require(not SOCKET.exists(), 'socket remains; never unlink an unproven endpoint')
    except BaseException as error:
        cleanup_failure = str(error)
        fault('recovery', error)
    finally:
        CANCEL.set()
        stop_io()
        # Preserve each old identity's independent exit outcome, even on recovery failure.
        for p in LEDGER.old_members():
            try:
                exited = gone(p)
                save('old-identity-exit-outcome', dict(identity=p, ESRCH=exited))

        if observer.is_alive() or server_thread.is_alive():
            terminal_stop('background-quiescence', 'observer/HTTP thread remains; retain state; no late mutation')
        save('result', dict(errors=ERRORS, cleanup_failure=cleanup_failure, runtime_accepted=False,
             independent_result_review_required=True, total_elapsed=time.monotonic() - START,
             signals=SIGNALS, validated_new=VALIDATED_NEW, quit_sent=QUIT_SENT,

             persistent_LS_assessment_state='unknown; no full restoration promise', normal_checks_enabled=True,

             proof_limit='50ms sampled identities only; no complete kernel process/exec coverage. Sampled logs and post-capture limits are not strict peak-resource bounds. Missing independent LS/NSWorkspace/exec evidence remains UNVERIFIED.'))
        if not CONTROL_DONE.is_set():
            terminal_stop('blocked-control', 'recovery finished with control still blocked; retained state; stop all threads')
        budget()
        RECOVERY_DONE.set()


def recover_guarded(*args):
    try:
        recover(*args)
    except BaseException as error:

        terminal_stop('recovery-thread', str(error))
