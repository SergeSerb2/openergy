def sha(p):
    with p.open('rb') as f:
        return hashlib.file_digest(f, 'sha256').hexdigest()


def save(name, value):

def inventory(root):
    result = {}
    total = 0
    for p in sorted(root.rglob('*')):
        require(len(result) < 8192, 'inventory entry limit')
        s = p.lstat()
        require(s.st_uid == UID, 'foreign owner')
        n = p.relative_to(root).as_posix()
        if p.is_symlink():
            require(p.resolve(strict=True).is_relative_to(root.resolve()), 'escaping link')
            result[n] = dict(symlink=os.readlink(p))
        elif stat.S_ISREG(s.st_mode):
            require(s.st_nlink == 1, 'hard link')
            total += s.st_size
            require(total <= 256 * 1024 * 1024, 'inventory byte limit')
            result[n] = dict(bytes=s.st_size, sha256=sha(p), mode=oct(stat.S_IMODE(s.st_mode)))
        elif stat.S_ISDIR(s.st_mode):
            result[n] = dict(directory=True, mode=oct(stat.S_IMODE(s.st_mode)))
        else:
            raise RuntimeError('special file in immutable inventory')
    return result


# Layout from the locally inspected macOS SDK sys/proc_info.h, not a ps time string.
class BSD(C.Structure):
    _fields_ = [(n, C.c_uint32) for n in ['flags', 'status', 'xstatus', 'pid', 'ppid', 'uid',
                  'gid', 'ruid', 'rgid', 'svuid', 'svgid', 'reserved']] + [
                  ('comm', C.c_char * 16), ('name', C.c_char * 32)] + [
                  (n, C.c_uint32) for n in ['nfiles', 'pgid', 'pjobc', 'tdev', 'tpgid']] + [
                  ('nice', C.c_int32), ('start_sec', C.c_uint64), ('start_usec', C.c_uint64)]


LIBPROC = None


def identity(pid):
    info = BSD()
    C.set_errno(0)
    n = LIBPROC.proc_pidinfo(pid, 3, 0, C.byref(info), C.sizeof(info))
    if n == 0 and C.get_errno() in (errno.ESRCH, errno.ENOENT):
        return None
    require(n == C.sizeof(info), 'independent process identity unavailable')
    buf = C.create_string_buffer(4096)
    n = LIBPROC.proc_pidpath(pid, buf, len(buf))
    require(n > 0, 'process executable path unavailable; no exit inference')
    return dict(pid=pid, ppid=info.ppid, uid=info.uid, start_sec=info.start_sec,
                start_usec=info.start_usec, nice=info.nice, executable=os.fsdecode(buf.value), status=info.status)


def same(now, prior):
    return now is not None and all(now[k] == prior[k] for k in ['pid', 'uid', 'start_sec', 'start_usec', 'executable'])

def fixture_processes():

    require(0 < n < C.sizeof(buf), 'process list unavailable or truncated')
    result = []

def environment(pid):
    # KERN_PROCARGS2 only for independently owned target/daemon PIDs. Never host-wide.
    p = identity(pid)
    require(p and p['uid'] == UID and p['executable'] in [str(IMAGE), str(DAEMON)], 'environment PID gate')
    libc = C.CDLL('/usr/lib/libSystem.B.dylib', use_errno=True)
    mib = (C.c_int * 3)(1, 49, pid)
    data = C.create_string_buffer(256 * 1024)
    size = C.c_size_t(len(data))
    require(libc.sysctl(mib, 3, data, C.byref(size), None, 0) == 0, 'owned environment unavailable')
    raw = data.raw[:size.value]
    argc = struct.unpack_from('i', raw)[0]
    require(0 < argc < 32, 'argument count')
    offset = raw.index(b'\0', 4) + 1
    while offset < len(raw) and raw[offset] == 0:
        offset += 1
    args = []
    for _ in range(argc):
        end = raw.index(b'\0', offset)
        args.append(os.fsdecode(raw[offset:end])); offset = end + 1
    values = {}
    for value in raw[offset:].split(b'\0'):
        if b'=' in value:
            k, v = value.split(b'=', 1)
            values[os.fsdecode(k)] = os.fsdecode(v)
    return dict(arguments=args, values=values)

def exact_read(s, count):
    data = b''
    while len(data) < count:
        chunk = s.recv(count - len(data))
        require(chunk, 'IPC EOF')
        data += chunk
    return data


def ipc():
    result = {}
    # Two requests per launch. No subscriptions or mutation methods. At most four total.
    with socket.socket(socket.AF_UNIX) as s:
        s.settimeout(2)
        s.connect(str(SOCKET))
        for request_id, method in [(1, 'status.get'), (2, 'config.get')]:
            data = json.dumps(dict(id=request_id, method=method, params=None), separators=(',', ':')).encode()
            require(len(data) <= 256, 'IPC request bound')
            s.sendall(struct.pack('!I', len(data)) + data)
            length = struct.unpack('!I', exact_read(s, 4))[0]
            require(0 < length <= 65536, 'IPC response bound')
            value = json.loads(exact_read(s, length))
            require(value.get('id') == request_id and 'error' not in value and 'result' in value, 'IPC response')
            result[method] = value['result']
    return result

def normal_quit(prior):
    # Native process-addressed normal termination API; no desktop/input/AppleScript.
    budget()
    require(same(identity(prior['pid']), prior) and prior['uid'] == UID and prior['executable'] == str(IMAGE), 'quit ownership')
    C.CDLL('/System/Library/Frameworks/AppKit.framework/AppKit')
    objc = C.CDLL('/usr/lib/libobjc.A.dylib')
    objc.objc_getClass.restype = C.c_void_p
    objc.sel_registerName.restype = C.c_void_p
    klass = objc.objc_getClass(b'NSRunningApplication')
    sel = objc.sel_registerName(b'runningApplicationWithProcessIdentifier:')
    app = C.CFUNCTYPE(C.c_void_p, C.c_void_p, C.c_void_p, C.c_int)(('objc_msgSend', objc))(klass, sel, prior['pid'])
    require(app, 'native running application lookup')
    terminate = objc.sel_registerName(b'terminate')
    accepted = C.CFUNCTYPE(C.c_bool, C.c_void_p, C.c_void_p)(('objc_msgSend', objc))(app, terminate)
    save('normal-quit', dict(identity=prior, API='NSRunningApplication.terminate', accepted=accepted))
    require(accepted, 'normal quit refused')

class Feed(socketserver.BaseRequestHandler):
    def handle(self):
        budget()
        server = self.server
        connection_deadline = time.monotonic() + 2
        self.request.settimeout(2)
        raw = b''
        while b'\r\n\r\n' not in raw:
            remaining = connection_deadline - time.monotonic()
            require(remaining > 0, 'HTTP absolute connection deadline')
            self.request.settimeout(remaining)
            part = self.request.recv(1024)
            require(part and len(raw) + len(part) <= 8192, 'HTTP header bound')
            raw += part
        head, body = raw.split(b'\r\n\r\n', 1)
        require(not body, 'HTTP request body refused')
        lines = head.split(b'\r\n')
        method, path, version = lines[0].decode('ascii').split(' ')
        require(method in ['GET', 'HEAD'] and version in ['HTTP/1.0', 'HTTP/1.1'], 'HTTP protocol')
        require(path in ['/appcast.xml', '/openergy-fixture.zip'], 'HTTP route refused')
        require(not any(line.lower().startswith((b'range:', b'transfer-encoding:', b'content-length:')) for line in lines[1:]), 'HTTP body/range refused')
        key = method + ' ' + path
        require(key not in server.requests and len(server.requests) < 4, 'duplicate/request budget')
        server.requests.add(key)
        p = ROOT / 'public' / path[1:]
        data = p.read_bytes()
        response = (f'HTTP/1.1 200 OK\r\nContent-Length: {len(data)}\r\n'
                    'Content-Type: application/octet-stream\r\nConnection: close\r\n\r\n').encode()
        remaining = connection_deadline - time.monotonic()
        require(remaining > 0, 'HTTP absolute connection deadline')
        self.request.settimeout(remaining)
        self.request.sendall(response)
        if method == 'GET':
            self.request.sendall(data)
            server.sent[path] = len(data)
        save('HTTP-complete', dict(method=method, path=path, length=len(data), content_sha256=sha(p),
             peer=self.client_address[0], sent_body=method == 'GET'))


class Server(socketserver.TCPServer):
    allow_reuse_address = False
    request_queue_size = 2
    def handle_error(self, request, client_address):
        self.failure = True
        save('HTTP-failure', dict(peer=client_address[0], error=str(sys.exc_info()[1])))

                    stdout=(RAW / 'cli.stdout').open('xb'), stderr=(RAW / 'cli.stderr').open('xb'))
        save('external-cli', dict(argv=plan['external_cli_command'], pid=cli.pid, environment=ENV))
        cli_identity = identity(cli.pid)
        require(cli_identity and cli_identity['nice'] == 15, 'CLI process identity')
        track(owned, cli_identity)
        update_deadline = time.monotonic() + 90
        new_receipt = None
        old_app_gone = old_child_gone = False
        while time.monotonic() < update_deadline:
            budget()
            launched.poll()  # reap direct child before independently checking ESRCH
            processes = fixture_processes()
            save('process-sample', processes)
            for p in processes:
                all_seen.add((p['pid'], p['start_sec'], p['start_usec'], p['executable']))
                if p['executable'] in [str(ENTRY), str(IMAGE), str(DAEMON)]:
                    require(p['start_sec'] >= old_app['start_sec'], 'preexisting process appeared')
                    track(owned, p)
            require(len(all_seen) <= 32, 'total process identity bound')
            require(len({p[0] for p in all_seen if p[3] in [str(ENTRY), str(IMAGE)]}) <= 2 and
                    len({p[0] for p in all_seen if p[3] == str(DAEMON)}) <= 2, 'unexpected restart/third target')
            for name, limit in [('cli.stdout', 8 * 1024 * 1024), ('cli.stderr', 8 * 1024 * 1024),
                                ('initial.stdout', 1024 * 1024), ('initial.stderr', 1024 * 1024)]:
                require((RAW / name).stat().st_size <= limit, 'product log byte bound')
            app_pids = {p['pid'] for p in processes if p['executable'] in [str(IMAGE), str(ENTRY)]}
            daemon_pids = {p['pid'] for p in processes if p['executable'] == str(DAEMON)}
            require(len(app_pids) <= 2 and len(daemon_pids) <= 2, 'unexpected app/daemon spawning')
            old_app_gone |= gone(old_app)
            old_child_gone |= gone(old_child)
            require(not server.failure, 'HTTP bounded server failure')
            if new_receipt is None:
                new_receipt = read_receipt('2', 'ready', seen)
            if new_receipt and cli.poll() is not None and old_app_gone and old_child_gone:
                break
            time.sleep(.05)
        require(new_receipt and cli.poll() == 0 and old_app_gone and old_child_gone, 'handoff/independent exit deadline')
        save('independent-old-exits', dict(app=old_app, daemon=old_child, both_ESRCH=True,
             limit='1.5s supervisor SIGKILL fallback exists; no signal outcome inferred from receipt'))
        require(inventory(APP) == inventories['candidate'], 'exact replacement tree')
        require(server.sent == {name: row['bytes'] for name, row in plan['http_files'].items()}, 'complete exact download')
        new_app, new_child = ready('2', new_receipt)
        track(owned, new_app); track(owned, new_child)
        require(new_receipt['launchID'] != old_receipt['launchID'] and new_app['pid'] != old_app['pid'] and
                new_child['pid'] != old_child['pid'], 'new process/lifecycle identities')
        require(sha(ROOT / 'state/config.toml') == plan['private_file_hashes'][str(ROOT / 'state/config.toml')], 'config persistence')
        ro('replacement-seal', ['/usr/bin/codesign', '--verify', '--all-architectures', '--deep', '--strict', APP])
        os_evidence('new-ready')
        # ONLY controller-requested normal quit. Sparkle requested the old quit.
        quit_sent = True
        normal_quit(new_app)
        end = time.monotonic() + 10
        while time.monotonic() < end and not (gone(new_app) and gone(new_child)):
            budget()
            time.sleep(.05)
        require(gone(new_app) and gone(new_child), 'final app/daemon exit deadline')
        require(read_receipt('1', 'exit', seen) and read_receipt('2', 'exit', seen), 'termination receipts absent')
        require(not SOCKET.exists(), 'owned socket remains')
    except BaseException as error:
        failed = str(error)
        save('first-runtime-failure', dict(error=failed, retry=False))
    finally:
        # No scope mutation or new launch. Best effort bounded recovery, never target strangers.
        cleanup_start = time.monotonic()
        if not quit_sent:
            alive_apps = [p for p in owned if p['executable'] == str(IMAGE) and same(identity(p['pid']), p)]
            require(len(alive_apps) <= 1, 'ambiguous cleanup app; leave and report')
            if alive_apps:
                quit_sent = True
                normal_quit(alive_apps[0])
        while time.monotonic() - cleanup_start < 10:
            if launched:
                launched.poll()
            if cli:
                cli.poll()
            if all(gone(p) for p in owned):
                break
            time.sleep(.1)
        for p in owned:
            if not gone(p):
                signal_owned(p, signal.SIGTERM)  # never before 10s graceful deadline
        while time.monotonic() - cleanup_start < 15 and any(not gone(p) for p in owned):
            if launched:
                launched.poll()
            if cli:
                cli.poll()
            time.sleep(.1)
        for p in owned:
            if not gone(p):
                require(time.monotonic() - cleanup_start >= 15, 'forcekill deadline')
                signal_owned(p, signal.SIGKILL)
        server.shutdown(); server.server_close(); thread.join(timeout=2)
        # Jobs/helpers normally leave themselves. Never signal shared gktool/lsregister.
        # A remaining exact job can be removed only after its frozen program hash is bound.
        os_evidence('postflight')
        for job in JOBS:
            code, output = ro('cleanup-job-state', ['/bin/launchctl', 'print', f'gui/{UID}/{job}'], allowed=(0, 113))
            if code == 0:
                while time.monotonic() - cleanup_start < 15:
                    budget()
                    time.sleep(.1)
                programs = re.findall(r'^\s*program = (.+)$', output, re.M)
                require(len(programs) == 1, 'job program not independently identified')
                program = Path(programs[0])
                require(program.is_file() and not program.is_symlink() and program.lstat().st_uid == UID and
                        program.resolve(strict=True) == program and
                        (program.is_relative_to(ROOT) or any(program.is_relative_to(p) for p in caches)) and
                        sha(program) in plan['allowed_helper_sha256'], 'job program ownership/hash mismatch')
                ro('cleanup-exact-job-bootout', ['/bin/launchctl', 'bootout', f'gui/{UID}/{job}'])
                ro('cleanup-exact-job-absent', ['/bin/launchctl', 'print', f'gui/{UID}/{job}'], allowed=(113,))
        quiet_deadline = time.monotonic() + 5
        residual = fixture_processes()
        while residual and time.monotonic() < quiet_deadline:
            time.sleep(.1)
            residual = fixture_processes()
        save('postflight-processes', residual)
        require(not residual, 'owned helper/process remnants; retain caches and report')
        predicate = 'eventMessage CONTAINS "' + IDENTIFIER + '" OR eventMessage CONTAINS "' + str(ROOT) + '"'
        ro('fixture-LS-Gatekeeper-XProtect-log', ['/usr/bin/log', 'show', '--last', '4m', '--style', 'ndjson',
            '--predicate', predicate], filtered=True)
        for p in caches:
            if p.exists() or p.is_symlink():
                s = p.lstat()
                require(stat.S_ISDIR(s.st_mode) and s.st_uid == UID and p.resolve() == p, 'cache cleanup ownership')
                before = dict(device=s.st_dev, inode=s.st_ino, inventory=inventory(p))
                save('owned-cache-before-removal', dict(path=str(p), **before))
                current = p.lstat()
                require((current.st_dev, current.st_ino) == (s.st_dev, s.st_ino), 'cache identity changed')
                import shutil
                shutil.rmtree(p)
                require(not p.exists(), 'cache removal incomplete')
        # Keep all artifacts, private preferences and runtime evidence; never unregister shared IDs.
        require(not SOCKET.exists(), 'socket remains; never unlink an unproven endpoint')
        save('result', dict(first_failure=failed, runtime_accepted=False,
             independent_result_review_required=True, total_elapsed=time.monotonic() - START,
             signals=SIGNALS, persistent_LS_assessment_state='unknown; no full restoration promise',
             normal_checks_enabled=True, controller_launches=1, controller_updates=1,
             explicit_quit_at_most_one=True,
             proof_limit='Initial invocation + same PID/path/environment and post-update OS records. Sampling can miss entry/helper events; absent independent LS/NSWorkspace/exec evidence stays UNVERIFIED.'))
    require(time.monotonic() - START <= 240, 'whole-operation deadline exceeded')
    require(failed is None, 'runtime failed; no retry')


if __name__ == '__main__':
    main()
