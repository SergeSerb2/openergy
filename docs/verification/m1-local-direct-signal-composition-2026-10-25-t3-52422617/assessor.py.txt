def require(value):
    if value is not True:

def canonical(value):
    return (json.dumps(value, sort_keys=True, separators=(',', ':'),
                       ensure_ascii=True, allow_nan=False) + '\n').encode('ascii')


def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def unique(pairs):
    result = {}
    for key, value in pairs:
        require(key not in result)
        result[key] = value
    return result


def finite_float(text):
    value = float(text)
    require(math.isfinite(value))
    return value


def nonfinite(text):

def load(raw):
    return json.loads(raw.decode('ascii'), object_pairs_hook=unique,
                      parse_float=finite_float, parse_constant=nonfinite)


def keys(value, names):
    require(type(value) is dict and set(value) == set(names))


def equal(actual, expected):
    # Python's True == 1 must never admit a number or a flag.
    require(type(actual) is type(expected))
    if type(expected) is dict:
        keys(actual, expected)
        for key in expected:
            equal(actual[key], expected[key])
    elif type(expected) is list:
        require(len(actual) == len(expected))
        for left, right in zip(actual, expected):
            equal(left, right)
    else:
        require(actual == expected)


def integer(value, minimum=0, maximum=9223372036854775807):

def hexhash(value, length=64):
    require(type(value) is str and len(value) == length

    try:
        before()

        before()
        file_state(fd, parent, name, initial, cap, mode)
        parts, seen = [], 0
        while True:
            file_state(fd, parent, name, initial, cap, mode)
            before()
            chunk = os.read(fd, min(16384, cap + 1 - seen))
            before()
            file_state(fd, parent, name, initial, cap, mode)
            seen += len(chunk)

def publish(value):
    raw = canonical(value)
    require(len(raw) <= 16384)
    retained_inputs()
    before()
    fd = os.open(OUTPUT, os.O_RDWR | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW
                 | os.O_NONBLOCK, 0o600, dir_fd=ROOT)
    try:
        before()
        initial = os.fstat(fd)
        before()
        file_state(fd, ROOT, OUTPUT, initial, 16384, 0o600)
        require(initial.st_size == 0)
        before()
        require(os.write(fd, raw) == len(raw))
        before()
        os.fsync(fd)
        before()
        written = os.fstat(fd)
        before()
        file_state(fd, ROOT, OUTPUT, written, 16384, 0o600)
        require(written.st_size == len(raw)
                and (written.st_dev, written.st_ino) == (initial.st_dev, initial.st_ino))
    finally:
        close(fd)
    os.fsync(ROOT)
    before()
    linked = os.stat(OUTPUT, dir_fd=ROOT, follow_symlinks=False)
    before()
    require(stamp(linked) == stamp(written))
    retained_inputs()
    return {'bytes': len(raw), 'sha256': digest(raw)}
