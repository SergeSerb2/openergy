def require(condition, message):
    if not condition:

def sha256_file(f):
    f.seek(0)
    h = hashlib.sha256()
    while True:
        data = f.read(1024 * 1024)
        if not data:
            break
        h.update(data)
    f.seek(0)
    return h.hexdigest()


def clean_path(name):
    require(isinstance(name, str) and name, 'empty name')
    require(len(name.encode('utf-8')) <= 4096, 'path exceeds 4096 bytes')
    require(not name.startswith('/') and '\\' not in name and ':' not in name,
            'absolute, backslash, or colon path')
    require(all(unicodedata.category(c)[0] != 'C' for c in name), 'control path')
    if name.endswith('/'):
        name = name[:-1]
    parts = name.split('/')
    require(all(p not in ('', '.', '..') for p in parts), 'empty/traversal path')
    require(parts[0] == 'python', 'path outside python root')
    return name


def alias_key(path):
    return unicodedata.normalize('NFD', unicodedata.normalize('NFD', path).casefold())


def safe_link_target(name, target, hard=False):
    require(target and not target.startswith('/') and '\\' not in target and ':' not in target,
            'unsafe link target')
    require(len(target.encode('utf-8')) <= 4096, 'link exceeds 4096 bytes')
    require(all(unicodedata.category(c)[0] != 'C' for c in target), 'control link')
    parts = ([] if hard else name.split('/')[:-1])
    for part in target.split('/'):
        if part in ('', '.'):
            continue
        if part == '..':
            require(len(parts) > 1, 'link traverses python root')
            parts.pop()
        else:
            parts.append(part)
    result = '/'.join(parts)
    require(result == 'python' or result.startswith('python/'), 'link outside python root')
    return result


def macho_metadata(data):
    if data[:4] != b'\xcf\xfa\xed\xfe':
        require(data[:4] not in (b'\xfe\xed\xfa\xce', b'\xce\xfa\xed\xfe',
                                 b'\xfe\xed\xfa\xcf', b'\xca\xfe\xba\xbe',
                                 b'\xbe\xba\xfe\xca', b'\xca\xfe\xba\xbf',
                                 b'\xbf\xba\xfe\xca'), 'unsupported Mach-O format')
        return None
    require(len(data) >= 32, 'short Mach-O header')
    magic, cpu, subtype, kind, ncmds, cmdbytes, flags, reserved = struct.unpack_from('<8I', data)
    require(ncmds <= 4096 and 32 + cmdbytes <= len(data), 'unbounded Mach-O commands')
    result = {'cpu_type': hex(cpu), 'cpu_subtype': subtype, 'file_type': kind,
              'architecture': 'arm64' if cpu == 0x0100000c else 'OTHER',
              'build_versions': [], 'dylibs': [], 'rpaths': []}
    def version(v):
        return '{}.{}.{}'.format(v >> 16, (v >> 8) & 255, v & 255)
    offset = 32
    for _ in range(ncmds):
        require(offset + 8 <= 32 + cmdbytes, 'short load command')
        cmd, size = struct.unpack_from('<II', data, offset)
        require(size >= 8 and offset + size <= 32 + cmdbytes, 'load command overflow')
        if cmd == 0x32:
            require(size >= 24, 'short LC_BUILD_VERSION')
            platform, minimum, sdk, tools = struct.unpack_from('<IIII', data, offset + 8)
            require(24 + 8 * tools <= size, 'build tools overflow')
            result['build_versions'].append({'command': 'LC_BUILD_VERSION', 'platform': platform,
                                             'min_os': version(minimum), 'sdk': version(sdk)})
        elif cmd == 0x24:
            require(size >= 16, 'short LC_VERSION_MIN_MACOSX')
            minimum, sdk = struct.unpack_from('<II', data, offset + 8)
            result['build_versions'].append({'command': 'LC_VERSION_MIN_MACOSX', 'platform': 1,
                                             'min_os': version(minimum), 'sdk': version(sdk)})
        elif cmd in (0xc, 0xd, 0x80000018, 0x8000001f, 0x80000023, 0x8000001c):
            require(size >= 12, 'short dylib/rpath')
            string_offset = struct.unpack_from('<I', data, offset + 8)[0]
            require(12 <= string_offset < size, 'invalid dylib/rpath string')
            value = data[offset + string_offset:offset + size].split(b'\0', 1)[0].decode('utf-8', 'strict')
            result['rpaths' if cmd == 0x8000001c else 'dylibs'].append(value)
        offset += size
    require(offset == 32 + cmdbytes, 'Mach-O load command total mismatch')
    return result


def scan(archive_file):

    require(stat.S_ISREG(before.st_mode), 'archive must be regular')
    require(before.st_uid == os.geteuid(), 'archive must be owned by current user')

    require(stat.S_IMODE(before.st_mode) == 0o600, 'archive must be mode0600')

    require(sha256_file(archive_file) == EXPECTED_SHA256, 'archive SHA256 mismatch')

    physical_headers = 0
    raw_bytes = 0
    file_bytes = 0

    spellings = {}
    captured = {}
    machos = {}
    manifest_hash = hashlib.sha256()

    with gzip.GzipFile(fileobj=archive_file, mode='rb') as gz:
        def read_exact(count):

            require(0 <= count <= MAX_EXPANDED - raw_bytes, 'expanded stream exceeds 1GiB')
            result = bytearray()
            while len(result) < count:
                piece = gz.read(min(count - len(result), 1024 * 1024))
                require(piece, 'truncated tar stream')
                result.extend(piece)
                raw_bytes += len(piece)
            return bytes(result)
        while True:
            header = read_exact(512)
            if header == b'\0' * 512:
                require(long_name is None and long_link is None, 'dangling GNU longname metadata')
                require(read_exact(512) == b'\0' * 512, 'missing second tar end block')
                while True:
                    chunk = gz.read(min(1024 * 1024, MAX_EXPANDED - raw_bytes + 1))
                    if not chunk:
                        break
                    raw_bytes += len(chunk)
                    require(raw_bytes <= MAX_EXPANDED and not any(chunk), 'nonzero tail or expanded bound')
                break
            physical_headers += 1
            require(physical_headers <= MAX_MEMBERS, 'too many physical tar headers')
            m = tarfile.TarInfo.frombuf(header, 'utf-8', 'strict')
            require(0 <= m.size <= MAX_EXPANDED, 'invalid member size')
            require(not m.mode & ~0o777, 'special permission bits')
            require(m.type in (tarfile.REGTYPE, tarfile.AREGTYPE, tarfile.DIRTYPE,
                               tarfile.SYMTYPE, tarfile.LNKTYPE,

                value = value[:-1].decode('utf-8', 'strict')

                    long_name = value
                else:
                    require(long_link is None, 'duplicate GNU longlink metadata')
                    long_link = value

                continue

                     'payload_offset': raw_bytes}

                data = read_exact(min(remaining, 1024 * 1024))
                digest.update(data)
                if len(prefix_data) < 1024 * 1024:
                    prefix_data.extend(data[:1024 * 1024 - len(prefix_data)])
                remaining -= len(data)
            require(not any(read_exact((-m.size) % 512)), 'nonzero member padding')
            if kind == 'file':
                entry['sha256'] = digest.hexdigest()

                    machos[name] = meta
                if name.endswith('/patchlevel.h') or name.endswith('/pyvenv.cfg') or name.endswith('/_sysconfigdata__darwin_darwin.py') or name.endswith('/python3.13-config'):

def open_directory_chain(path):
    require(path.startswith('/') and path != '/', 'private directory must be absolute')
    parts = path.split('/')[1:]
    require(all(p not in ('', '.', '..') for p in parts), 'noncanonical private directory')
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY)
    try:
        for part in parts:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = child
        return fd
    except BaseException:
        os.close(fd)
        raise


def member_parent_fd(root_fd, name):
    parts = clean_path(name).split('/')
    fd = os.dup(root_fd)
    try:
        for part in parts[:-1]:
            child = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = child
        return fd, parts[-1]
    except BaseException:
        os.close(fd)
        raise
