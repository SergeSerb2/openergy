"""One real empty transport CLI invocation. No remote/profile/payload input."""
import os
import sys
import time
import json
import hashlib
import subprocess

os.setpriority(os.PRIO_PROCESS, 0, 15)
assert os.getpriority(os.PRIO_PROCESS, 0) == 15
assert sys.platform == 'darwin' and sys.version_info[:2] == (3, 15)
assert sys.flags.isolated and sys.flags.no_site and sys.flags.dont_write_bytecode
assert not sys.flags.optimize
collector = '/Users/serge/.local/state/openergy-verification/m1-source-metadata-launch-prepared-20261025-7ae23486940a/tools/transport-r4.py'
fd = os.open(collector, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
try:
    raw = os.read(fd, 131073)
    assert len(raw) <= 131072 and os.read(fd, 1) == b''
    assert hashlib.sha256(raw).hexdigest() == 'd312814e74c6f42e2a86d56648450dc145531df6292ea8c79ccf1768f34b14e4'
finally:
    os.close(fd)
argv = ['/opt/homebrew/bin/python3.15', '-I', '-S', '-B', collector]
started = time.monotonic_ns()
deadline = started + 15000000000
proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.DEVNULL,
                        stderr=subprocess.DEVNULL, shell=False, close_fds=True)
spawn_returned = time.monotonic_ns()
actual_exit = None
timed_out = False
error = None
try:
    budget = (deadline - time.monotonic_ns()) / 1000000000
    if budget <= 0:
        raise TimeoutError('observer deadline after spawn')
    actual_exit = proc.wait(timeout=budget)
    observed = time.monotonic_ns()
except BaseException as exc:
    observed = time.monotonic_ns()
    timed_out = isinstance(exc, (TimeoutError, subprocess.TimeoutExpired))
    error = type(exc).__name__
result = dict(schema='openergy.real-transport-empty-cli-observation.v1',
              candidate_sha256=hashlib.sha256(raw).hexdigest(),
              candidate_source_review='a7e22725cbbf07b5e6ac7e7aa7e7fb0d3dff93b4',
              argv=argv, started_ns=started, deadline_ns=deadline,
              spawn_returned_ns=spawn_returned, actual_candidate_exit=actual_exit,
              post_actual_wait_observed_ns=observed, timeout=timed_out, error=error,
              strictly_before_observer_deadline=observed < deadline,
              expected_failure_observed=actual_exit == 1 and observed < deadline and error is None,
              console='DEVNULL; console bytes were not collected',
              parent_own_nice=15, candidate_nice_measurement=None,
              expected_early_branch='arguments rejects missing flags before clock_descriptor or Collector construction',
              positive_flow_D4_proven=False, remote_metadata_accepted=False,
              runtime='UNVERIFIED')
data = (json.dumps(result, sort_keys=True, separators=(',', ':'), ensure_ascii=True) + '\n').encode('ascii')
assert len(data) <= 4096
assert os.write(1, data) == len(data)
os._exit(0 if result['expected_failure_observed'] else 1)
