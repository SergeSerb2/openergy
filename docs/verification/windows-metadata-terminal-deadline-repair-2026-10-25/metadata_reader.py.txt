"""Inert proposal; execute only in a separately assigned future parent task.

POSIX local parent, Python >=3.10, standard library. No discovery or retry.
Raw output and receipts are private. This module has no import-time I/O.
"""
import argparse
import base64
import hashlib
import json
import math
import ntpath
import os
from pathlib import Path, PureWindowsPath
import selectors
import stat
import subprocess
import time

CHUNK = 4096
INPUT_CAP = 131072
STREAM_CEILING = 1048576
RECEIPT_CAP = 65536
PATHS = (
    r'C:\WINDOWS\System32\WindowsPowerShell\v1.0\powershell.exe',
    r'c:\program files\powershell\7\pwsh.exe',
    r'C:\WINDOWS\System32\OpenSSH\sshd.exe',
    r'c:\program files\powershell\7\System.Diagnostics.Process.dll',
    r'c:\program files\powershell\7\System.Private.CoreLib.dll',
    r'c:\program files\powershell\7\pwsh.runtimeconfig.json',
)
SSH_OPTIONS = ['-o', 'BatchMode=yes', '-o', 'ConnectTimeout=8', '-o',
               'StrictHostKeyChecking=yes', '-o', 'UpdateHostKeys=no', '-o',
               'ControlMaster=no', '-o', 'ServerAliveInterval=10', '-o',
               'ServerAliveCountMax=2', '-o', 'ForwardAgent=no']


def require(ok, label):
    if not ok:
        raise ValueError(label)


def digest(data):
    return hashlib.sha256(data).hexdigest()


def hex64(value):
    return (isinstance(value, str) and len(value) == 64
            and all(c in '0123456789abcdef' for c in value))


def read_bound(path, cap):
    # Bounded even if a regular input grows after fstat. Never read a FIFO/device.
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK)
    try:
        info = os.fstat(fd)
        require(stat.S_ISREG(info.st_mode) and info.st_size <= cap, 'input size/type')
        data = bytearray()
        while len(data) <= cap:
            part = os.read(fd, min(CHUNK, cap + 1 - len(data)))
            if not part:
                break
            data.extend(part)
        require(len(data) <= cap, 'input growth cap')
        return bytes(data)
    finally:
        os.close(fd)


def strict_json(data):
    def unique(pairs):
        obj = {}
        for key, value in pairs:
            require(key not in obj, 'duplicate JSON key')
            obj[key] = value
        return obj
    def bad_constant(_):
        raise ValueError('nonfinite JSON number')
    return json.loads(data, object_pairs_hook=unique, parse_constant=bad_constant)


def admit(binding, query):
    require(type(binding) is dict and binding.get('schema') ==
            'openergy.metadata-reader-command.v1', 'binding schema')
    purpose = binding.get('purpose')
    argv = binding.get('argv')
    require(type(argv) is list and 1 <= len(argv) <= 64, 'argv count')
    require(all(type(a) is str and '\0' not in a for a in argv), 'argv strings')
    require(sum(len(a.encode('utf-8')) + 1 for a in argv) <= 65536, 'local argv cap')
    require(os.path.isabs(argv[0]), 'explicit local executable required')
    if purpose == 'local-reader-verification':
        require(set(binding) == {'schema', 'purpose', 'argv'}, 'local binding keys')
        return argv, []
    require(purpose == 'windows-installed-metadata', 'purpose')
    require(set(binding) == {'schema', 'purpose', 'argv', 'query_sha256',
                            'selector_binding_sha256', 'command_layers'}, 'binding keys')
    require(binding['query_sha256'] == digest(query), 'query binding')
    require(hex64(binding['selector_binding_sha256']), 'private selector binding required')
    encoded = base64.b64encode(query.decode('utf-8').encode('utf-16le')).decode('ascii')
    native = ['powershell.exe', '-NoLogo', '-NoProfile', '-NonInteractive',
              '-EncodedCommand', encoded]
    prefix = ['/usr/bin/ssh', *SSH_OPTIONS]
    require(len(argv) == len(prefix) + 1 + len(native) and argv[:len(prefix)] == prefix,
            'existing SSH argv shape')
    selector = argv[len(prefix)]
    require(selector and not selector.startswith('-') and
            not any(c.isspace() for c in selector), 'pre-bound selector')
    require(argv[len(prefix) + 1:] == native, 'exact uncompressed query argv')
    line = subprocess.list2cmdline(native)
    sizes = [{'construction': 'submitted_native', 'units_including_nul':
              len(line.encode('utf-16le')) // 2 + 1, 'effective_limit': 'UNVERIFIED'}]
    layers = binding['command_layers']
    require(type(layers) is list and 1 <= len(layers) <= 4, 'layer count')
    for layer in layers:
        require(type(layer) is dict, 'layer object')
        kind = layer.get('construction')
        if kind == 'literal_native':
            require(set(layer) == {'construction', 'prefix', 'suffix',
                                   'effective_limit_including_nul'}, 'literal keys')
            require(type(layer['prefix']) is str and type(layer['suffix']) is str,
                    'literal strings')
            complete = layer['prefix'] + line + layer['suffix']
        elif kind == 'resolved_executable':
            require(set(layer) == {'construction', 'executable',
                                   'effective_limit_including_nul'}, 'resolved keys')
            exe = layer['executable']
            require(type(exe) is str, 'executable string')
            path = PureWindowsPath(exe)
            require(len(path.drive) == 2 and path.drive[0].isascii() and
                    path.drive[0].isalpha() and path.drive[1] == ':' and
                    path.root == '\\' and exe.count(':') == 1 and
                    ntpath.normpath(exe) == exe and
                    not any(c in exe for c in '"\0\r\n') and
                    path.name.casefold() == 'powershell.exe', 'resolved path')
            complete = '"' + exe + '"' + line[len(native[0]):]
        else:
            raise ValueError('layer construction')
        require('\0' not in complete, 'layer NUL')
        units = len(complete.encode('utf-16le')) // 2 + 1
        limit = layer['effective_limit_including_nul']
        require(limit is None or (type(limit) is int and 1 <= limit <= 32767),
                'explicit limit or unknown required')
        require(units <= 32767 and (limit is None or units <= limit), 'layer cap')
        sizes.append({'construction': kind, 'units_including_nul': units,
                      'effective_limit': limit if limit is not None else 'UNVERIFIED'})
    require(sizes[0]['units_including_nul'] <= 32767, 'native cap')
    return argv, sizes


def metadata_complete(raw):
    # Run only after raw bytes are durable and the local child is settled.
    require(raw.endswith(b'\n'), 'unterminated metadata')
    lines = raw.splitlines()
    require(len(lines) == 8 and all(len(x) <= 262144 for x in lines), 'record count/cap')
    records = [strict_json(x.decode('utf-8')) for x in lines]
    require(records[0] == {'schema': 'openergy.windows-installed-metadata.ndjson.v1',
                          'kind': 'begin', 'metadata_only': True, 'files_expected': 6}, 'begin')
    require(type(records[0]['metadata_only']) is bool and
            type(records[0]['files_expected']) is int, 'begin value types')
    for index, (path, item) in enumerate(zip(PATHS, records[1:7]), 1):
        require(type(item) is dict and set(item) ==
                {'kind', 'index', 'requested_path', 'path', 'length', 'file_version',
                 'product_version', 'sha256', 'runtime_config'}, 'file keys')
        require(item['kind'] == 'file' and type(item['index']) is int and
                item['index'] == index and item['requested_path'] == path, 'file order/path')
        require(type(item['path']) is str and item['path'].casefold() == path.casefold(),
                'resolved file path')
        require(type(item['length']) is int and 0 <= item['length'] <= 268435456 and
                hex64(item['sha256']), 'full hash/length')
        require(all(item[k] is None or (type(item[k]) is str and len(item[k]) <= 4096)
                    for k in ('file_version', 'product_version')), 'version fields')
        require((type(item['runtime_config']) is str and item['length'] <= 65536)
                if index == 6 else item['runtime_config'] is None, 'runtimeconfig field')
    end = records[7]
    require(type(end) is dict and set(end) == {'kind', 'files', 'metadata_only',
                                             'priority', 'session', 'bytes_hashed'}, 'end keys')
    require(end['kind'] == 'complete' and type(end['files']) is int and end['files'] == 6
            and end['metadata_only'] is True and end['priority'] == 'BelowNormal'
            and type(end['session']) is int and end['session'] >= 0
            and type(end['bytes_hashed']) is int
            and end['bytes_hashed'] == sum(x['length'] for x in records[1:7])
            and end['bytes_hashed'] <= 536870912, 'complete record')


def cli():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--binding', required=True)
    parser.add_argument('--binding-sha256', required=True)
    parser.add_argument('--run-dir', required=True)
    for name in ('work-seconds', 'cleanup-seconds', 'finalize-seconds', 'total-seconds'):
        parser.add_argument('--' + name, type=float, required=True)
    for name in ('stdout-cap', 'stderr-cap', 'aggregate-cap'):
        parser.add_argument('--' + name, type=int, required=True)
    args = parser.parse_args()
    numbers = (args.work_seconds, args.cleanup_seconds, args.finalize_seconds, args.total_seconds)
    require(os.name == 'posix' and all(math.isfinite(x) and x > 0 for x in numbers),
            'POSIX and finite positive budgets required')
    require(sum(numbers[:3]) <= args.total_seconds <= 120, 'absolute budget allocation')
    require(all(1 <= x <= STREAM_CEILING for x in
                (args.stdout_cap, args.stderr_cap, args.aggregate_cap)), 'capture policy cap')
    require(hex64(args.binding_sha256), 'binding hash')
    require(Path(args.binding).is_absolute(), 'absolute binding path')
    return args


def capture(args):
    started = time.monotonic()
    absolute = started + args.total_seconds
    work_end = started + args.work_seconds
    settle_end = absolute - args.finalize_seconds
    report = {'schema': 'openergy.metadata-reader-receipt.v1', 'execution': True,
              'status': 'INCOMPLETE', 'first_error': None, 'secondary_error_count': 0,
              'local_child_started': False, 'local_spawn_attempted': False,
              'local_child_returncode': None, 'local_exit_observed': False,
              'local_kill_called': False, 'remote_query_exit_and_current_state': 'UNVERIFIED',
              'remote_cleanup_proved': False, 'effective_limits': 'UNVERIFIED',
              'receipt_requires_cli_exit_zero': True,
              'hard_wall_clock_enforced': False, 'external_envelope_required': True,
              'metadata_complete': False, 'retries': 0,
              'budgets_seconds': {'work': args.work_seconds, 'cleanup': args.cleanup_seconds,
                                 'finalize': args.finalize_seconds, 'total': args.total_seconds},
              'caps': {'stdout': args.stdout_cap, 'stderr': args.stderr_cap,
                       'aggregate': args.aggregate_cap},
              'streams': {name: {'bytes': 0, 'bytes_read': 0, 'eof': False,
                                 'overflow_observed': False}
                          for name in ('stdout', 'stderr')}}
    proc = None
    mux = selectors.DefaultSelector()
    sinks = {}
    hashes = {name: hashlib.sha256() for name in ('stdout', 'stderr')}
    receipt = None
    phase = 'prepare'

    def fail(error):
        if report['first_error'] is None:
            # Do not copy exception text: it can contain connection/account data.
            report['first_error'] = {'phase': phase, 'class': type(error).__name__}
        else:
            report['secondary_error_count'] += 1

    def before(deadline):
        if time.monotonic() >= deadline:
            raise TimeoutError('absolute deadline')

    def checkpoint():
        before(absolute)
        report['elapsed_seconds'] = time.monotonic() - started
        for name, entry in report['streams'].items():
            entry['sha256'] = hashes[name].hexdigest()
            entry['bytes_not_preserved'] = entry['bytes_read'] - entry['bytes']
        data = (json.dumps(report, sort_keys=True, allow_nan=False) + '\n').encode()
        require(len(data) <= RECEIPT_CAP, 'receipt cap')
        # Fixed allocated control file. No temporary-file allocation after launch.
        os.lseek(receipt, 0, os.SEEK_SET)
        pending = memoryview(data + b' ' * (RECEIPT_CAP - len(data)))
        while pending:
            before(absolute)
            count = os.write(receipt, pending)
            require(count > 0, 'receipt write')
            pending = pending[count:]
        os.fsync(receipt)
        before(absolute)

    def sync_raw():
        for fd in sinks.values():
            before(absolute)
            os.fsync(fd)

    def observe():
        if proc is not None:
            code = proc.poll()  # WNOHANG; retain exact SSH exit or negative POSIX signal.
            if code is not None:
                report['local_child_returncode'] = code
                report['local_exit_observed'] = True

    def close_pipes():
        for key in list(mux.get_map().values()):
            try:
                mux.unregister(key.fileobj)
                key.fileobj.close()
            except BaseException as error:
                fail(error)

    def pump(deadline):
        before(deadline)
        observe()
        events = mux.select(min(0.05, max(0, deadline - time.monotonic())))
        for key, _ in events:
            before(deadline)
            name = key.data
            entry = report['streams'][name]
            total = sum(x['bytes'] for x in report['streams'].values())
            left = min(report['caps'][name] - entry['bytes'], args.aggregate_cap - total)
            # One-byte overflow sentinel. Accepted bytes never exceed either cap.
            size = min(CHUNK, left) if left else 1
            try:
                data = os.read(key.fd, size)
            except BlockingIOError:
                continue
            if not data:
                entry['eof'] = True
                mux.unregister(key.fileobj)
                key.fileobj.close()
                continue
            entry['bytes_read'] += len(data)
            if not left:
                entry['overflow_observed'] = True
                raise OverflowError('capture cap')
            pending = memoryview(data)
            while pending:
                before(deadline)
                count = os.write(sinks[name], pending)
                require(count > 0, 'raw write')
                hashes[name].update(pending[:count])
                entry['bytes'] += count
                pending = pending[count:]
        observe()
        before(deadline)  # A late final poll cannot consume settlement time as capture.

    try:
        run = Path(args.run_dir)
        require(run.is_absolute() and not run.exists(), 'new absolute private run directory')
        info = run.parent.lstat()
        require(stat.S_ISDIR(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o700
                and info.st_uid == os.getuid(), 'owned private parent mode 0700')
        run.mkdir(mode=0o700)
        for name in ('stdout', 'stderr'):
            sinks[name] = os.open(run / (name + '.bin'), os.O_WRONLY | os.O_CREAT |
                                  os.O_EXCL | os.O_NOFOLLOW, 0o600)
        receipt = os.open(run / 'receipt.json', os.O_RDWR | os.O_CREAT | os.O_EXCL |
                          os.O_NOFOLLOW, 0o600)
        checkpoint()  # Allocate control capacity and persist INCOMPLETE before spawn.
        directory = os.open(run, os.O_RDONLY | os.O_DIRECTORY)
        try:
            os.fsync(directory)
        finally:
            os.close(directory)
        before(work_end)
        report['reader_sha256'] = digest(read_bound(Path(__file__), 32768))
        binding_bytes = read_bound(args.binding, INPUT_CAP)
        require(digest(binding_bytes) == args.binding_sha256, 'exact command binding')
        binding = strict_json(binding_bytes)
        query = (read_bound(Path(__file__).with_name('metadata-query.ps1.txt'), 12224)
                 if binding.get('purpose') == 'windows-installed-metadata' else b'')
        argv, sizes = admit(binding, query)
        report['purpose'] = binding['purpose']
        report['binding_sha256'] = digest(binding_bytes)
        report['query_sha256'] = digest(query) if query else None
        report['command_sizes'] = sizes
        before(work_end)
        phase = 'spawn'
        report['local_spawn_attempted'] = True
        checkpoint()
        proc = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                                stderr=subprocess.PIPE, shell=False, bufsize=0,
                                close_fds=True)
        report['local_child_started'] = True
        for name in ('stdout', 'stderr'):
            stream = getattr(proc, name)
            os.set_blocking(stream.fileno(), False)
            mux.register(stream, selectors.EVENT_READ, name)
        phase = 'capture'
        while mux.get_map() or not report['local_exit_observed']:
            pump(work_end)
        report['work_completion_observed_monotonic'] = time.monotonic()
        before(work_end)  # Recheck after all capture/exit observations.
    except BaseException as error:
        fail(error)
        if any(x['bytes_read'] != x['bytes'] for x in report['streams'].values()):
            # A partial write or overflow sentinel must not create a later gap.
            close_pipes()
    finally:
        # Raw chunks are already on disk. Persist the first failure before local settlement.
        try:
            sync_raw()
            if receipt is not None:
                checkpoint()
        except BaseException as error:
            fail(error)
        phase = 'local-settlement'
        cleanup_end = min(settle_end, time.monotonic() + args.cleanup_seconds)
        try:
            observe()
            if proc is not None and not report['local_exit_observed']:
                before(cleanup_end)
                try:
                    proc.kill()  # Only this Popen child; no process group or remote command.
                    report['local_kill_called'] = True
                except ProcessLookupError:
                    pass
            try:
                while proc is not None and (mux.get_map() or not report['local_exit_observed']):
                    pump(cleanup_end)
            except BaseException as error:
                fail(error)
                close_pipes()
            # A full/broken stream must not prevent bounded local exit observation.
            while proc is not None and not report['local_exit_observed']:
                before(cleanup_end)
                observe()
                if not report['local_exit_observed']:
                    time.sleep(min(0.01, max(0, cleanup_end - time.monotonic())))
        except BaseException as error:
            fail(error)
        finally:
            # Never call communicate(), wait() without a remaining bound, or Popen.__exit__.
            close_pipes()
            for stream in (() if proc is None else (proc.stdout, proc.stderr)):
                try:
                    stream.close()
                except BaseException as error:
                    fail(error)
            mux.close()
            try:
                observe()
            except BaseException as error:
                fail(error)
        phase = 'finalize'
        try:
            before(absolute)
            sync_raw()
            for name in hashes:
                report['streams'][name]['sha256'] = hashes[name].hexdigest()
            ready = (report['first_error'] is None and report['local_exit_observed']
                     and report['local_child_returncode'] == 0
                     and all(x['eof'] and not x['overflow_observed']
                             for x in report['streams'].values()))
            if ready and report['purpose'] == 'windows-installed-metadata':
                require(report['streams']['stderr']['bytes'] == 0, 'metadata stderr')
                before(absolute)
                metadata_complete(read_bound(Path(args.run_dir) / 'stdout.bin', args.stdout_cap))
                before(absolute)
                report['metadata_complete'] = True
            for name, fd in list(sinks.items()):
                os.close(fd)
                del sinks[name]
            report['status'] = 'COMPLETE' if ready else 'INCOMPLETE'
            report['deadline_observation_before_final_write'] = time.monotonic() < absolute
            if receipt is not None:
                checkpoint()
        except BaseException as error:
            fail(error)
            report['status'] = 'INCOMPLETE'
            try:
                if receipt is not None:
                    checkpoint()
            except BaseException:
                pass  # Initial INCOMPLETE receipt and raw prefix survive if storage permits.
        finally:
            for fd in (*sinks.values(), *(() if receipt is None else (receipt,))):
                try:
                    os.close(fd)
                except BaseException:
                    report['status'] = 'INCOMPLETE'
    return 0 if report['status'] == 'COMPLETE' and time.monotonic() < absolute else 1


if __name__ == '__main__':
    # No raw argv, exception text, selector or output is printed to a public terminal.
    try:
        exit_code = capture(cli())
    except SystemExit:
        raise
    except BaseException:
        exit_code = 1
    raise SystemExit(exit_code)
