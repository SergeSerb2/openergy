def canonical(v):
    return (json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=True,allow_nan=False)+'\n').encode('ascii')

def stamp(s):
    return (s.st_dev,s.st_ino,s.st_mode,s.st_uid,s.st_gid,s.st_nlink,s.st_size,s.st_mtime_ns,s.st_ctime_ns)

def read_bound(path,size,wanted,mode):
    fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
    try:
        old=os.fstat(fd)
        assert stat.S_ISREG(old.st_mode) and old.st_uid==os.getuid() and old.st_nlink==1
        assert stat.S_IMODE(old.st_mode)==mode and old.st_size==size and size<=1048576
        parts=[];seen=0
        while True:
            chunk=os.read(fd,min(65536,size+1-seen))
            if not chunk:break
            seen+=len(chunk);assert seen<=size;parts.append(chunk)
        raw=b''.join(parts)
        assert len(raw)==size and hashlib.sha256(raw).hexdigest()==wanted
        assert stamp(old)==stamp(os.fstat(fd))==stamp(os.stat(path,follow_symlinks=False))
        return raw
    finally:
        os.close(fd)

def save(name,value,end):

    try:
        assert time.monotonic_ns()<end
        assert os.write(fd,raw)==len(raw)
        assert time.monotonic_ns()<end
        os.fsync(fd)
        assert time.monotonic_ns()<end
    finally:
        os.close(fd)
    d=os.open(B,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW)
    try:os.fsync(d)
    finally:os.close(d)
    assert time.monotonic_ns()<end
    return {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
